     1→'use client';
     2→
     3→import { useState, useEffect, useCallback, useMemo } from 'react';
     4→import {
     5→  ArrowLeft, User, Bookmark, Clock, LogOut, Pencil, Check, X, Trash2,
     6→  Loader2, Film, Tv, Users, Calendar, Shield, Palette, Heart, Camera,
     7→  ChevronDown, ShieldOff,
     8→} from 'lucide-react';
     9→import { motion, AnimatePresence } from 'framer-motion';
    10→import { useAuthStore } from '@/store/auth-store';
    11→import { useAppStore } from '@/store/app-store';
    12→import { Input } from '@/components/ui/input';
    13→import { AuthModal } from '@/components/auth/AuthModal';
    14→import {
    15→  ProfileAvatar, AVATAR_DEFS, ACCENT_COLORS, GENRE_OPTIONS,
    16→  getAvatarDef, type AvatarDef,
    17→} from '@/lib/avatars';
    18→import { getImageUrl } from '@/lib/tmdb';
    19→import { getLocalBrowseHistory, deleteLocalHistoryItem, clearLocalHistory } from '@/lib/useRecordHistory';
    20→import type { BrowseHistoryEntry } from '@/lib/useRecordHistory';
    21→import type { Movie } from '@/lib/types';
    22→
    23→interface BrowseHistoryItem {
    24→  id: string;
    25→  tmdbId: number;
    26→  title: string;
    27→  posterPath: string | null;
    28→  mediaType: 'movie' | 'tv' | 'person';
    29→  subtitle: string | null;
    30→  visitedAt: string;
    31→}
    32→
    33→type HistoryFilter = 'all' | 'movie' | 'tv' | 'person';
    34→
    35→type Tab = 'profile' | 'watchlist' | 'history';
    36→
    37→type EditStep = 'main' | 'avatar' | 'color' | 'genres';
    38→
    39→export function ProfilePage() {
    40→  const { user, loading: authLoading, updateProfile, logout } = useAuthStore();
    41→  const { goHome, watchlist, selectMovie, selectTv, selectPerson } = useAppStore();
    42→
    43→  const [tab, setTab] = useState<Tab>('profile');
    44→  const [authModalOpen, setAuthModalOpen] = useState(false);
    45→
    46→  // Profile editing
    47→  const [editing, setEditing] = useState(false);
    48→  const [editStep, setEditStep] = useState<EditStep>('main');
    49→  const [editName, setEditName] = useState('');
    50→  const [editBio, setEditBio] = useState('');
    51→  const [selectedAvatar, setSelectedAvatar] = useState('hero');
    52→  const [selectedColor, setSelectedColor] = useState('#e11d48');
    53→  const [selectedGenres, setSelectedGenres] = useState<string[]>([]);
    54→  const [saving, setSaving] = useState(false);
    55→  const [saveError, setSaveError] = useState('');
    56→
    57→  // Browse history
    58→  const [history, setHistory] = useState<BrowseHistoryItem[]>([]);
    59→  const [historyLoading, setHistoryLoading] = useState(false);
    60→  const [historyFilter, setHistoryFilter] = useState<HistoryFilter>('all');
    61→  const [historyPage, setHistoryPage] = useState(1);
    62→  const [historyTotal, setHistoryTotal] = useState(0);
    63→  const [historyLoadMore, setHistoryLoadMore] = useState(false);
    64→  const [deletingId, setDeletingId] = useState<string | null>(null);
    65→
    66→  // Profile stats from API
    67→  const [profileStats, setProfileStats] = useState<{ watchHistoryCount: number } | null>(null);
    68→
    69→  // Computed accent color for UI
    70→  const accentColor = useMemo(() => {
    71→    if (user?.accentColor) return user.accentColor;
    72→    const def = getAvatarDef(user?.avatar);
    73→    return def.color;
    74→  }, [user?.avatar, user?.accentColor]);
    75→
    76→  const fetchProfile = useCallback(async () => {
    77→    if (!user) return;
    78→    try {
    79→      const res = await fetch('/api/profile');
    80→      const data = await res.json();
    81→      if (res.ok) setProfileStats(data.stats);
    82→    } catch { /* ignore */ }
    83→  }, [user]);
    84→
    85→  const fetchHistory = useCallback(async (filter: HistoryFilter, page: number, append: boolean) => {
    86→    // Non-authenticated users: use localStorage
    87→    if (!user) {
    88→      setHistoryLoading(true);
    89→      let local = getLocalBrowseHistory();
    90→      if (filter !== 'all') local = local.filter((i) => i.mediaType === filter);
    91→      setHistory(local);
    92→      setHistoryTotal(local.length);
    93→      setHistoryLoading(false);
    94→      return;
    95→    }
    96→    // Authenticated: use server API
    97→    if (append) { setHistoryLoadMore(true); } else { setHistoryLoading(true); }
    98→    try {
    99→      const params = new URLSearchParams({ limit: '50', offset: String((page - 1) * 50) });
   100→      if (filter !== 'all') params.set('type', filter);
   101→      const res = await fetch(`/api/history?${params}`);
   102→      const data = await res.json();
   103→      if (res.ok) {
   104→        setHistory(prev => append ? [...prev, ...data.items] : data.items);
   105→        setHistoryTotal(data.total);
   106→        if (!append) setHistoryPage(1);
   107→      }
   108→    } catch { /* ignore */ }
   109→    setHistoryLoading(false);
   110→    setHistoryLoadMore(false);
   111→  }, [user]);
   112→
   113→  useEffect(() => {
   114→    if (tab === 'profile') fetchProfile();
   115→    if (tab === 'history') fetchHistory(historyFilter, 1, false);
   116→  }, [tab, historyFilter, user, fetchHistory, fetchProfile]);
   117→
   118→  useEffect(() => {
   119→    window.scrollTo({ top: 0, behavior: 'smooth' });
   120→  }, []);
   121→
   122→  const startEdit = () => {
   123→    setEditName(user?.name || '');
   124→    setEditBio(user?.bio || '');
   125→    setSelectedAvatar(user?.avatar || 'hero');
   126→    setSelectedColor(user?.accentColor || getAvatarDef(user?.avatar).color);
   127→    setSelectedGenres(user?.favoriteGenres || []);
   128→    setEditStep('main');
   129→    setEditing(true);
   130→    setSaveError('');
   131→  };
   132→
   133→  const saveProfile = async () => {
   134→    setSaving(true);
   135→    setSaveError('');
   136→    const result = await updateProfile({
   137→      name: editName,
   138→      bio: editBio,
   139→      avatar: selectedAvatar,
   140→      accentColor: selectedColor,
   141→      favoriteGenres: selectedGenres,
   142→    });
   143→    setSaving(false);
   144→    if (result.error) {
   145→      setSaveError(result.error);
   146→    } else {
   147→      setEditing(false);
   148→    }
   149→  };
   150→
   151→  const toggleGenre = (genre: string) => {
   152→    setSelectedGenres(prev =>
   153→      prev.includes(genre) ? prev.filter(g => g !== genre) : [...prev, genre]
   154→    );
   155→  };
   156→
   157→  const clearHistory = async () => {
   158→    if (!confirm('Clear all browsing history? This cannot be undone.')) return;
   159→    if (user) {
   160→      await fetch('/api/history', { method: 'DELETE' });
   161→    } else {
   162→      clearLocalHistory();
   163→    }
   164→    setHistory([]);
   165→    setHistoryTotal(0);
   166→  };
   167→
   168→  const handleDeleteHistoryItem = async (id: string, e: React.MouseEvent) => {
   169→    e.stopPropagation();
   170→    setDeletingId(id);
   171→    if (user) {
   172→      await fetch(`/api/history?id=${id}`, { method: 'DELETE' });
   173→    } else {
   174→      deleteLocalHistoryItem(id);
   175→    }
   176→    setHistory(prev => prev.filter(item => item.id !== id));
   177→    setHistoryTotal(prev => prev - 1);
   178→    setDeletingId(null);
   179→  };
   180→
   181→  const handleHistoryClick = (item: BrowseHistoryItem) => {
   182→    if (item.mediaType === 'person') {
   183→      selectPerson({ id: item.tmdbId, name: item.title, profilePath: item.posterPath });
   184→      return;
   185→    }
   186→    const media: Movie = {
   187→      id: item.tmdbId, title: item.title, name: item.title,
   188→      poster_path: item.posterPath, media_type: item.mediaType,
   189→      vote_average: 0, genre_ids: [], overview: '', popularity: 0,
   190→      release_date: '', first_air_date: '', backdrop_path: null, original_language: '',
   191→    };
   192→    if (item.mediaType === 'tv') selectTv(media); else selectMovie(media);
   193→  };
   194→
   195→  const handleHistoryFilterChange = (f: HistoryFilter) => {
   196→    if (f === historyFilter) return;
   197→    setHistoryFilter(f);
   198→  };
   199→
   200→  const loadMoreHistory = () => {
   201→    const nextPage = historyPage + 1;
   202→    setHistoryPage(nextPage);
   203→    fetchHistory(historyFilter, nextPage, true);
   204→  };
   205→
   206→  const handleWatchlistClick = (tmdbId: number, mediaType: string) => {
   207→    const item: Movie = {
   208→      id: tmdbId, title: '', name: '', poster_path: null,
   209→      media_type: mediaType as 'movie' | 'tv', vote_average: 0, genre_ids: [],
   210→      overview: '', popularity: 0, release_date: '', first_air_date: '',
   211→      backdrop_path: null, original_language: '',
   212→    };
   213→    if (mediaType === 'tv') selectTv(item); else selectMovie(item);
   214→  };
   215→
   216→  // Not logged in — but still show the page with limited tabs
   217→  const isLoggedIn = !!user && !authLoading;
   218→
   219→  const memberSince = user ? new Date(user.createdAt || Date.now()).toLocaleDateString('en-US', { month: 'long', year: 'numeric' }) : '';
   220→  const avatarDef = getAvatarDef(user?.avatar);
   221→
   222→  // Default to history tab when not logged in
   223→  useEffect(() => {
   224→    if (!isLoggedIn && tab === 'profile') setTab('history');
   225→  }, [isLoggedIn]);
   226→
   227→  const tabsList: { key: Tab; label: string; icon: React.ReactNode }[] = [
   228→    ...(isLoggedIn ? [{ key: 'profile' as Tab, label: 'Profile', icon: <User className="w-4 h-4" /> }] : []),
   229→    ...(isLoggedIn ? [{ key: 'watchlist' as Tab, label: `Watchlist (${watchlist.length})`, icon: <Bookmark className="w-4 h-4" /> }] : []),
   230→    { key: 'history', label: `History${historyTotal > 0 ? ` (${historyTotal})` : ''}`, icon: <Clock className="w-4 h-4" /> },
   231→  ];
   232→
   233→  return (
   234→    <div className="min-h-screen pb-10">
   235→      {/* Back button */}
   236→      <div className="pt-20 px-4 md:px-8">
   237→        <button onClick={goHome} className="flex items-center gap-2 text-white/50 hover:text-white mb-6 transition-colors group">
   238→          <div className="w-8 h-8 rounded-full bg-white/[0.06] flex items-center justify-center group-hover:bg-white/10 transition-colors">
   239→            <ArrowLeft className="w-4 h-4" />
   240→          </div>
   241→          <span className="text-sm font-medium">Back</span>
   242→        </button>
   243→
   244→        {/* ── Profile Header (logged in only) ── */}
   245→        {isLoggedIn && (
   246→        <div className="flex flex-col sm:flex-row items-center sm:items-start gap-6 mb-8">
   247→          {/* Avatar */}
   248→          <motion.div className="relative" initial={{ scale: 0.8, opacity: 0 }} animate={{ scale: 1, opacity: 1 }} transition={{ type: 'spring', stiffness: 300, damping: 20 }}>
   249→            <div className="w-28 h-28 rounded-2xl overflow-hidden shadow-2xl" style={{ boxShadow: `0 12px 40px ${accentColor}44` }}>
   250→              <ProfileAvatar slug={user?.avatar} size={112} />
   251→            </div>
   252→            {!editing && (
   253→              <button onClick={startEdit} className="absolute -bottom-1 -right-1 w-9 h-9 rounded-xl bg-white/10 backdrop-blur-sm border border-white/20 flex items-center justify-center hover:bg-white/20 transition-colors" style={{ boxShadow: `0 4px 12px ${accentColor}22` }}>
   254→                <Pencil className="w-3.5 h-3.5 text-white" />
   255→              </button>
   256→            )}
   257→            {/* Online indicator */}
   258→            <div className="absolute -bottom-0.5 -left-0.5 w-4 h-4 rounded-full border-2 border-zinc-950" style={{ backgroundColor: '#22c55e' }} />
   259→          </motion.div>
   260→
   261→          {/* Info */}
   262→          <div className="text-center sm:text-left flex-1">
   263→            <h1 className="text-2xl md:text-3xl font-bold text-white">{user?.name || 'StreamVault User'}</h1>
   264→            <p className="text-white/40 text-sm mt-1">{user?.email}</p>
   265→            {user?.bio && !editing && <p className="text-white/60 text-sm mt-2 max-w-md">{user.bio}</p>}
   266→            <div className="flex flex-wrap items-center justify-center sm:justify-start gap-3 mt-3 text-xs text-white/30">
   267→              <span className="flex items-center gap-1.5"><Calendar className="w-3.5 h-3.5" />Joined {memberSince}</span>
   268→              <span className="flex items-center gap-1.5"><Bookmark className="w-3.5 h-3.5" />{watchlist.length} saved</span>
   269→              {profileStats && <span className="flex items-center gap-1.5"><Clock className="w-3.5 h-3.5" />{profileStats.watchHistoryCount} watched</span>}
   270→            </div>
   271→            {/* Favorite genres badges */}
   272→            {user?.favoriteGenres && user.favoriteGenres.length > 0 && !editing && (
   273→              <div className="flex flex-wrap gap-1.5 mt-3 justify-center sm:justify-start">
   274→                {user.favoriteGenres.slice(0, 5).map(g => (
   275→                  <span key={g} className="px-2.5 py-1 rounded-lg text-[10px] font-semibold uppercase tracking-wider text-white/80" style={{ backgroundColor: `${accentColor}22`, border: `1px solid ${accentColor}33` }}>{g}</span>
   276→                ))}
   277→                {user.favoriteGenres.length > 5 && (
   278→                  <span className="px-2.5 py-1 rounded-lg text-[10px] font-semibold text-white/40 bg-white/[0.06]">+{user.favoriteGenres.length - 5}</span>
   279→                )}
   280→              </div>
   281→            )}
   282→          </div>
   283→
   284→          {/* Actions */}
   285→          <div className="sm:ml-auto flex gap-2">
   286→            <button onClick={startEdit} className="flex items-center gap-2 px-4 py-2.5 rounded-xl text-white/80 text-sm font-medium transition-all hover:brightness-110" style={{ backgroundColor: `${accentColor}22`, border: `1px solid ${accentColor}33` }}>
   287→              <Pencil className="w-4 h-4" /><span className="hidden sm:inline">Edit Profile</span>
   288→            </button>
   289→            <button onClick={async () => { await logout(); goHome(); }} className="flex items-center gap-2 px-4 py-2 rounded-xl bg-white/[0.06] hover:bg-white/10 text-white/60 hover:text-white text-sm transition-all border border-white/[0.06]">
   290→              <LogOut className="w-4 h-4" /><span className="hidden sm:inline">Sign Out</span>
   291→            </button>
   292→          </div>
   293→        </div>
   294→        )}
   295→        {!isLoggedIn && (
   296→          <div className="mb-8">
   297→            <h1 className="text-2xl md:text-3xl font-bold text-white">Browsing History</h1>
   298→            <p className="text-white/40 text-sm mt-1">Your recently viewed content is saved locally. <button onClick={() => setAuthModalOpen(true)} className="text-white/60 hover:text-white underline underline-offset-2 transition-colors">Sign in</button> to sync across devices.</p>
   299→          </div>
   300→        )}
   301→
   302→        {/* Tabs */}
   303→        <div className="flex gap-1 mb-6 border-b border-white/[0.06]">
   304→          {tabsList.map((t) => (
   305→            <button key={t.key} onClick={() => setTab(t.key)} className={`flex items-center gap-2 px-4 py-3 text-sm font-medium transition-all relative border-b-2 -mb-px ${tab === t.key ? 'text-white' : 'text-white/40 border-transparent hover:text-white/70'}`} style={tab === t.key ? { borderColor: accentColor } : undefined}>
   306→              {t.icon}
   307→              <span className="hidden sm:inline">{t.label}</span>
   308→              <span className="sm:hidden">{t.key === 'watchlist' ? watchlist.length : ''}</span>
   309→            </button>
   310→          ))}
   311→        </div>
   312→
   313→        {/* Tab Content */}
   314→        <AnimatePresence mode="wait">
   315→          {tab === 'profile' && (
   316→            <motion.div key="profile-tab" initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -8 }} className="max-w-2xl">
   317→              {editing ? (
   318→                <EditPanel
   319→                  editStep={editStep} setEditStep={setEditStep}
   320→                  editName={editName} setEditName={setEditName}
   321→                  editBio={editBio} setEditBio={setEditBio}
   322→                  selectedAvatar={selectedAvatar} setSelectedAvatar={setSelectedAvatar}
   323→                  selectedColor={selectedColor} setSelectedColor={setSelectedColor}
   324→                  selectedGenres={selectedGenres} toggleGenre={toggleGenre}
   325→                  accentColor={accentColor}
   326→                  saving={saving} saveError={saveError}
   327→                  onSave={saveProfile} onCancel={() => setEditing(false)}
   328→                />
   329→              ) : (
   330→                <div className="space-y-4">
   331→                  <div className="bg-white/[0.03] rounded-2xl border border-white/[0.06] p-5 space-y-4">
   332→                    <h3 className="text-sm font-semibold text-white/70 flex items-center gap-2"><Shield className="w-4 h-4 text-white/30" />Account Info</h3>
   333→                    <div className="space-y-3">
   334→                      <InfoRow label="Name" value={user?.name || '\u2014'} />
   335→                      <InfoRow label="Email" value={user?.email} />
   336→                      <InfoRow label="Avatar" value={avatarDef.name} />
   337→                      <InfoRow label="Bio" value={user?.bio || '\u2014'} truncate />
   338→                      <InfoRow label="Theme" value={user?.accentColor ? `Custom (${ACCENT_COLORS.find(c => c.value === user.accentColor)?.name || 'Custom'})` : `Avatar-matched (${avatarDef.name})`} />
   339→                      <InfoRow label="Favorite Genres" value={user?.favoriteGenres?.length ? user.favoriteGenres.join(', ') : '\u2014'} />
   340→                      <InfoRow label="Member Since" value={memberSince} />
   341→                    </div>
   342→                  </div>
   343→
   344→                  {/* Content Settings */}
   345→                  <div className="bg-white/[0.03] rounded-2xl border border-white/[0.06] p-5 space-y-4">
   346→                    <h3 className="text-sm font-semibold text-white/70 flex items-center gap-2"><ShieldOff className="w-4 h-4 text-white/30" />Content Settings</h3>
   347→                    <div className="flex items-center justify-between">
   348→                      <div>
   349→                        <p className="text-sm text-white/80 font-medium">Adult Content</p>
   350→                        <p className="text-xs text-white/35 mt-0.5">Show 18+ rated titles in a dedicated section</p>
   351→                      </div>
   352→                      <button
   353→                        onClick={async () => {
   354→                          const newVal = !user?.adultEnabled;
   355→                          await updateProfile({ adultEnabled: newVal });
   356→                        }}
   357→                        className={`relative w-11 h-6 rounded-full transition-colors duration-200 ${user?.adultEnabled ? 'bg-red-600' : 'bg-white/15'}`}
   358→                      >
   359→                        <div className={`absolute top-0.5 left-0.5 w-5 h-5 rounded-full bg-white shadow-sm transition-transform duration-200 ${user?.adultEnabled ? 'translate-x-5' : 'translate-x-0'}`} />
   360→                      </button>
   361→                    </div>
   362→                  </div>
   363→                </div>
   364→              )}
   365→            </motion.div>
   366→          )}
   367→
   368→          {tab === 'watchlist' && (
   369→            <motion.div key="watchlist-tab" initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -8 }}>
   370→              {watchlist.length === 0 ? (
   371→                <EmptyState icon={<Bookmark className="w-9 h-9 text-white/10" />} title="No items in watchlist" description="Browse movies and shows, and tap the heart icon to save them here." actionLabel="Browse Content" onAction={goHome} />
   372→              ) : (
   373→                <div className="grid grid-cols-3 sm:grid-cols-4 md:grid-cols-5 lg:grid-cols-6 gap-3 md:gap-4">
   374→                  {watchlist.map((id) => (<WatchlistCard key={id} tmdbId={id} onClick={handleWatchlistClick} />))}
   375→                </div>
   376→              )}
   377→            </motion.div>
   378→          )}
   379→
   380→          {tab === 'history' && (
   381→            <motion.div key="history-tab" initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -8 }}>
   382→              {/* Filter tabs */}
   383→              <div className="flex gap-2 mb-5 flex-wrap">
   384→                {([
   385→                  { key: 'all' as const, label: 'All', icon: <Clock className="w-3.5 h-3.5" /> },
   386→                  { key: 'movie' as const, label: 'Movies', icon: <Film className="w-3.5 h-3.5" /> },
   387→                  { key: 'tv' as const, label: 'TV Shows', icon: <Tv className="w-3.5 h-3.5" /> },
   388→                  { key: 'person' as const, label: 'People', icon: <Users className="w-3.5 h-3.5" /> },
   389→                ]).map(f => {
   390→                  const active = historyFilter === f.key;
   391→                  return (
   392→                    <button
   393→                      key={f.key}
   394→                      onClick={() => handleHistoryFilterChange(f.key)}
   395→                      className={`flex items-center gap-1.5 px-4 py-1.5 rounded-full text-xs font-medium transition-all ${
   396→                        active
   397→                          ? 'text-white border'
   398→                          : 'bg-white/[0.06] text-white/50 hover:text-white/80 border border-transparent hover:border-white/10'
   399→                      }`}
   400→                      style={active ? { backgroundColor: `${accentColor}22`, borderColor: `${accentColor}44`, color: accentColor } : undefined}
   401→                    >
   402→                      {f.icon}
   403→                      {f.label}
   404→                    </button>
   405→                  );
   406→                })}
   407→              </div>
   408→
   409→              {historyLoading ? (
   410→                <div className="flex items-center justify-center py-20"><Loader2 className="w-6 h-6 text-white/30 animate-spin" /></div>
   411→              ) : history.length === 0 ? (
   412→                <EmptyState icon={<Clock className="w-9 h-9 text-white/10" />} title="No browsing history" description="Movies, shows, and people you view will appear here." />
   413→              ) : (
   414→                <>
   415→                  <div className="flex items-center justify-between mb-4">
   416→                    <p className="text-sm text-white/40">{history.length} of {historyTotal} items</p>
   417→                    <button onClick={clearHistory} className="flex items-center gap-1.5 text-sm text-white/30 hover:text-red-400 transition-colors"><Trash2 className="w-3.5 h-3.5" />Clear All</button>
   418→                  </div>
   419→                  <div className="space-y-2">
   420→                    <AnimatePresence mode="popLayout">
   421→                      {history.map((item) => {
   422→                        const isPerson = item.mediaType === 'person';
   423→                        const imgSize = isPerson ? 'w185' : 'w92';
   424→                        const aspectClass = isPerson ? 'aspect-[3/4]' : 'aspect-[2/3]';
   425→                        const imgUrl = item.posterPath ? getImageUrl(item.posterPath, imgSize) : null;
   426→                        const dateStr = new Date(item.visitedAt).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
   427→
   428→                        return (
   429→                          <motion.div
   430→                            key={item.id}
   431→                            layout
   432→                            initial={{ opacity: 0, y: 8 }}
