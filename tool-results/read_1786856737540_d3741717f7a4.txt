   337→   337→                </div>
   338→   338→              )}
   339→   339→            </motion.div>
   340→   340→          )}
   341→   341→        </AnimatePresence>
   342→   342→      </div>
   343→   343→      <AuthModal open={authModalOpen} onClose={() => setAuthModalOpen(false)} />
   344→   344→    </div>
   345→   345→  );
   346→   346→}

   348→   348→// ── Edit Panel with step navigation ──

   351→   351→  selectedAvatar, setSelectedAvatar, selectedColor, setSelectedColor,
   352→   352→  selectedGenres, toggleGenre, accentColor, saving, saveError, onSave, onCancel,
   353→   353→}: {
   354→   354→  editStep: EditStep; setEditStep: (s: EditStep) => void;
   355→   355→  editName: string; setEditName: (v: string) => void;
   356→   356→  editBio: string; setEditBio: (v: string) => void;
   357→   357→  selectedAvatar: string; setSelectedAvatar: (v: string) => void;
   358→   358→  selectedColor: string; setSelectedColor: (v: string) => void;
