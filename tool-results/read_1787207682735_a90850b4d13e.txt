    35→/* ------------------------------------------------------------------ */
    36→/*  iOS native fullscreen detection                                    */
    37→/* ------------------------------------------------------------------ */
    38→function useIOSFullscreenDetect(onReturn: () => void) {
    39→  useEffect(() => {
    40→    const handleEnd = () => onReturn();
    41→    document.addEventListener('webkitendfullscreen', handleEnd);
    42→    return () => document.removeEventListener('webkitendfullscreen', handleEnd);
    43→  }, [onReturn]);
    44→}
    45→
    46→/* ------------------------------------------------------------------ */
    47→/*  Score badge helper                                                 */
    48→/* ------------------------------------------------------------------ */
    49→function scoreBadge(score: number) {
    50→  if (score >= 80) return { label: 'Top Pick', className: 'bg-emerald-500/20 text-emerald-400' };
    51→  if (score >= 40) return { label: 'Popular', className: 'bg-amber-500/20 text-amber-400' };
    52→  if (score > 0)   return { label: 'Used', className: 'bg-zinc-500/20 text-zinc-400' };
    53→  return null;
    54→}

   259→        >

   261→          <div className="flex items-center justify-between px-5 py-4 border-b border-white/[0.06]">

   300→                    style={{ backgroundColor: p.color + '18', color: p.color }}

   324→                      >
   325→                        {badge.label}

   327→                    )}
   328→                  </div>
   329→
   330→                  {/* Status indicator */}
   331→                  <div className="flex items-center shrink-0">
   332→                    {isSwitching ? (
   333→                      <Loader2 className="w-4 h-4 animate-spin text-white/40" />

   337→                  </div>

   340→            })}
   341→          </div>
   342→
   343→          {/* Panel footer hint */}
   344→          <div className="px-5 py-3 border-t border-white/[0.06]">
   345→            <p className="text-white/25 text-[10px] text-center">
   346→              Rankings based on community usage

   348→          </div>

   364→            animate={{ opacity: 1 }}
   365→            exit={{ opacity: 0 }}
   366→            className="absolute inset-0 bg-black/60 z-20"

   371→            initial={{ y: '100%' }}
   372→            animate={{ y: 0 }}

   374→            transition={{ type: 'spring', stiffness: 300, damping: 30 }}
   375→            className="absolute bottom-0 left-0 right-0 max-h-[70vh] bg-[#111] rounded-t-2xl z-30 flex flex-col"
   376→            style={{ paddingBottom: isIOS ? 'env(safe-area-inset-bottom, 16px)' : 16 }}

   382→
   383→            {/* Header */}
   384→            <div className="flex items-center justify-between px-5 pb-3">
   385→              <div className="flex items-center gap-2">
   386→                <Server className="w-4 h-4 text-white/50" />
   387→                <h4 className="text-white/90 text-sm font-semibold">Servers</h4>
   388→              </div>
   389→              <button
   390→                onClick={() => setMobileSheetOpen(false)}
   391→                className="w-7 h-7 rounded-lg flex items-center justify-center text-white/40 hover:text-white/80 hover:bg-white/[0.06] transition-colors"
   392→                aria-label="Close server sheet"

   402→                const score = getServerScore(p.id);
   403→                const badge = scoreBadge(score);
   404→                const isTop = rankedIds[0] === p.id;
   405→                const isSwitching = switchingTo === p.id;
   406→
   407→                return (
   408→                  <button
   409→                    key={p.id}
   410→                    onClick={() => switchProvider(p.id)}
   411→                    disabled={isSwitching}
   412→                    className={
   413→                      'w-full flex items-center gap-3 px-3 py-3 rounded-xl text-left transition-colors ' +
   414→                      (isActive
   415→                        ? 'bg-white/[0.06]'
   416→                        : 'hover:bg-white/[0.03]')
   417→                    }
   418→                  >
   419→                    <div
   420→                      className="w-9 h-9 rounded-lg flex items-center justify-center text-sm font-bold shrink-0"
   421→                      style={{ backgroundColor: p.color + '18', color: p.color }}
   422→                    >
   423→                      {p.icon}
   424→                    </div>
   425→                    <div className="flex-1 min-w-0">
   426→                      <div className="flex items-center gap-1.5">
   427→                        <span className="text-white/90 text-sm font-medium truncate">
   428→                          {p.name}
   429→                        </span>
   430→                        {isTop && <ShieldCheck className="w-3.5 h-3.5 text-emerald-400 shrink-0" />}
   431→                      </div>
   432→                      <p className="text-white/40 text-[11px] truncate mt-0.5">{p.description}</p>

   434→                    <div className="flex flex-col items-end gap-1 shrink-0">
   435→                      {badge && (
   436→                        <span
   437→                          className={
   438→                            'px-1.5 py-0.5 rounded text-[9px] font-semibold ' +
   439→                            badge.className
   440→                          }
   441→                        >
   442→                          {badge.label}

   459→  );
   460→
   461→  // ------------------------------------------------------------------
   462→  // Loading state with pulsing ring

   464→  const LoadingState = (
   465→    <motion.div
   466→      initial={{ opacity: 0 }}
   467→      animate={{ opacity: 1 }}

   471→      {/* Pulsing ring */}

   474→          className="absolute inset-0 rounded-full animate-ping opacity-30"
   475→          style={{ backgroundColor: activeProvider.color }}

   477→        <div
   478→          className="absolute inset-1 rounded-full animate-pulse opacity-60"
   479→          style={{ backgroundColor: activeProvider.color }}

   481→        <div
   482→          className="relative w-10 h-10 rounded-full flex items-center justify-center"
   483→          style={{ backgroundColor: activeProvider.color + '25' }}
   484→        >
   485→          <Loader2
   486→            className="w-5 h-5 animate-spin"
   487→            style={{ color: activeProvider.color }}
   488→          />

   491→      <p className="text-white/50 text-xs font-medium">
   492→        Connecting to {activeProvider.name}...
   493→      </p>
   494→    </motion.div>
   495→  );
   496→
   497→  // ------------------------------------------------------------------
   498→  // Error state
   499→  // ------------------------------------------------------------------
   500→  const ErrorState = (
   501→    <div className="absolute inset-0 flex flex-col items-center justify-center gap-4 z-10 bg-black">
   502→      <div className="w-14 h-14 rounded-2xl bg-white/[0.06] flex items-center justify-center">

   504→      </div>

   766→                style={{ backgroundColor: activeProvider.color }}

   768→              <span className="text-white/60 text-xs font-medium">

   770→              </span>
   771→              {activeBadge && (

   773→                  className={
   774→                    'px-1.5 py-0.5 rounded text-[9px] font-semibold ' +

   798→        {/* ==================== iOS BOTTOM SAFE AREA ==================== */}
   799→        {isIOS && !isMobile && (
