    28→    28→function useIOSFullscreenDetect(onReturn: () => void) {
    29→    29→  useEffect(() => {
    30→    30→    const h = () => onReturn();
    31→    31→    document.addEventListener('webkitendfullscreen', h);
    32→    32→    return () => document.removeEventListener('webkitendfullscreen', h);
    33→    33→  }, [onReturn]);
    34→    34→}

    36→    36→/* Animated equalizer bars for "now playing" indicator */
    37→    37→function PlayingBars({ color, size = 'sm' }: { color: string; size?: 'sm' | 'md' }) {
    38→    38→  const h = size === 'sm' ? 'h-3' : 'h-4';
    39→    39→  const gap = size === 'sm' ? 'gap-[2px]' : 'gap-[3px]';
    40→    40→  const barW = size === 'sm' ? 'w-[3px]' : 'w-[4px]';
    41→    41→  return (
    42→    42→    <div className={`flex items-end ${gap} ${h}`}>
    43→    43→      {[0, 1, 2].map((i) => (
    44→    44→        <motion.div
    45→    45→          key={i}
    46→    46→          className={`${barW} rounded-full`}
    47→    47→          style={{ backgroundColor: color }}
    48→    48→          animate={{ height: ['40%', '100%', '60%', '100%', '40%'] }}
    49→    49→          transition={{
    50→    50→            duration: 1.2,
    51→    51→            repeat: Infinity,
    52→    52→            delay: i * 0.2,
    53→    53→            ease: 'easeInOut',
    54→    54→          }}
    55→    55→        />
    56→    56→      ))}
    57→    57→    </div>
    58→    58→  );
    59→    59→}
