    99→function proxyCover(url?: string): string {
   100→  if (!url) return '';
   101→  if (url.includes('gutenberg.org')) return `/api/novels/proxy?url=${encodeURIComponent(url)}`;
   102→  return `/api/manga/proxy?url=${encodeURIComponent(url)}`;
   103→}

   116→/* ═══════════════════════════════════════════════════ */
   117→export function ReadPage() {
   118→  const { goHome, selectManga, selectNovel, selectComic } = useAppStore();
   119→  const [mainTab, setMainTab] = useState<MainTab>('manga');
   120→
   121→  // Manga state
   122→  const [mangaList, setMangaList] = useState<MangaItem[]>([]);
   123→  const [mangaSubTab, setMangaSubTab] = useState('all');
   124→  const [mangaLoading, setMangaLoading] = useState(false);
   125→  const [mangaError, setMangaError] = useState(false);
   126→  const [mangaSearch, setMangaSearch] = useState('');
   127→  const [mangaHasMore, setMangaHasMore] = useState(false);
   128→  const [mangaTotal, setMangaTotal] = useState(0);
   129→  const [mangaOffset, setMangaOffset] = useState(0);
   130→  const mangaDebounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   131→  const mangaAbortRef = useRef<AbortController | null>(null);
   132→
   133→  // Comics state
   134→  const [comicList, setComicList] = useState<any[]>([]);
   135→  const [comicPublisher, setComicPublisher] = useState('All');
   136→  const [comicGenre, setComicGenre] = useState('All');
   137→  const [comicSort, setComicSort] = useState('popular');
   138→  const [comicSearch, setComicSearch] = useState('');
   139→  const [comicLoading, setComicLoading] = useState(true);
   140→  const [comicError, setComicError] = useState(false);
   141→  const [comicTotal, setComicTotal] = useState(0);
   142→  const [availablePublishers, setAvailablePublishers] = useState<string[]>([]);
   143→  const [availableGenres, setAvailableGenres] = useState<string[]>([]);
   144→  const [showSortDropdown, setShowSortDropdown] = useState(false);
   145→  const comicDebounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   146→
   147→  // Novels state
   148→  const [novelList, setNovelList] = useState<NovelItem[]>([]);
   149→  const [novelGenre, setNovelGenre] = useState('All');
   150→  const [novelSearch, setNovelSearch] = useState('');
   151→  const [novelLoading, setNovelLoading] = useState(true);
   152→  const novelDebounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   153→
   154→  /* ── Manga fetch ── */
   155→  const fetchManga = useCallback(async (tab: string, search: string, offset: number, append = false) => {
   156→    mangaAbortRef.current?.abort();
   157→    const controller = new AbortController();
   158→    mangaAbortRef.current = controller;
   159→    if (!append) { setMangaLoading(true); setMangaError(false); }
   160→
   161→    try {
   162→      const params = new URLSearchParams();
   163→      if (search.trim()) {
   164→        params.set('q', search.trim());
   165→        if (offset > 0) params.set('page', String(Math.floor(offset / 20) + 1));
   166→        const res = await fetch(`/api/manga/search?${params}`, { signal: controller.signal });
   167→        if (!res.ok) throw new Error();
   168→        const data = await res.json();
   169→        if (controller.signal.aborted) return;
   170→        const items: MangaItem[] = (data.results || []).map((m: any) => ({
   171→          id: String(m.id), title: String(m.title || 'Untitled'),
   172→          coverUrl: String(m.coverUrl || ''), author: String(m.author || ''),
   173→          tags: Array.isArray(m.tags) ? m.tags.filter(Boolean).map(String) : [],
   174→        }));
   175→        setMangaList(append ? (prev) => [...prev, ...items] : items);
   176→        setMangaHasMore(!!data.hasMore);
   177→        setMangaTotal(data.total || 0);
   178→      } else {
   179→        if (tab === 'manga' || tab === 'manhwa' || tab === 'manhua') params.set('type', tab);
   180→        if (offset > 0) params.set('offset', String(offset));
   181→        const url = `/api/manga/trending${params.toString() ? `?${params}` : ''}`;
   182→        const res = await fetch(url, { signal: controller.signal });
   183→        if (!res.ok) throw new Error();
   184→        const data = await res.json();
   185→        if (controller.signal.aborted) return;
   186→        const items: MangaItem[] = (data.results || []).map((m: any) => ({
   187→          id: String(m.id), title: String(m.title || 'Untitled'),
   188→          coverUrl: String(m.coverUrl || ''), author: String(m.author || ''),
   189→          tags: Array.isArray(m.tags) ? m.tags.filter(Boolean).map(String) : [],
   190→        }));
   191→        setMangaList(append ? (prev) => [...prev, ...items] : items);
   192→        setMangaHasMore(!!data.hasMore);
   193→        setMangaTotal(data.total || 0);
   194→      }
   195→    } catch {
   196→      if (!append) setMangaError(true);
   197→    } finally {
   198→      if (!controller.signal.aborted) setMangaLoading(false);
   199→    }
   200→  }, []);
   201→
   202→  useEffect(() => {
   203→    setMangaOffset(0); setMangaSearch('');
   204→    fetchManga(mangaSubTab, '', 0);
   205→    return () => { mangaAbortRef.current?.abort(); };
   206→  }, [mangaSubTab, fetchManga]);
   207→
   208→  const handleMangaSearch = useCallback((value: string) => {
   209→    setMangaSearch(value);
   210→    if (mangaDebounceRef.current) clearTimeout(mangaDebounceRef.current);
   211→    if (!value.trim()) { setMangaOffset(0); fetchManga(mangaSubTab, '', 0); return; }
   212→    mangaDebounceRef.current = setTimeout(() => { setMangaOffset(0); fetchManga(mangaSubTab, value, 0); }, 300);
   213→  }, [mangaSubTab, fetchManga]);
   214→
   215→  /* ── Comics fetch ── */
   216→  const fetchComics = useCallback(async (search: string) => {
   217→    setComicLoading(true);
   218→    setComicError(false);
   219→    try {
   220→      const params = new URLSearchParams();
   221→      if (comicPublisher !== 'All') params.set('publisher', comicPublisher);
   222→      if (comicGenre !== 'All') params.set('genre', comicGenre);
   223→      if (comicSort) params.set('sort', comicSort);
   224→      if (search.trim()) params.set('q', search.trim());
   225→      const res = await fetch(`/api/comics/trending?${params}`);
   226→      if (!res.ok) throw new Error();
   227→      const data = await res.json();
   228→      setComicList(data.results || []);
   229→      setComicTotal(data.total || 0);
   230→      if (data.publishers) setAvailablePublishers(data.publishers);
   231→      if (data.genres) setAvailableGenres(data.genres);
   232→    } catch {
   233→      setComicError(true);
   234→    } finally {
   235→      setComicLoading(false);
   236→    }
   237→  }, [comicPublisher, comicGenre, comicSort]);
   238→
   239→  useEffect(() => {
   240→    fetchComics(comicSearch);
   241→  }, [fetchComics]);
   242→
   243→  const handleComicSearch = useCallback((value: string) => {
   244→    setComicSearch(value);
   245→    if (comicDebounceRef.current) clearTimeout(comicDebounceRef.current);
   246→    comicDebounceRef.current = setTimeout(() => {
   247→      fetchComics(value);
   248→    }, 300);
   249→  }, [fetchComics]);
   250→
   251→  /* ── Novels fetch ── */
   252→  useEffect(() => {
   253→    setNovelLoading(true);
   254→    fetch('/novels-data.json')
   255→      .then((r) => r.json())
   256→      .then((data: NovelItem[]) => setNovelList(data))
   257→      .catch(() => {})
   258→      .finally(() => setNovelLoading(false));
   259→  }, []);
   260→
   261→  const filteredNovels = useMemo(() => {
   262→    let list = novelList;
   263→    if (novelGenre !== 'All') list = list.filter((n) => n.genre.includes(novelGenre));
   264→    if (novelSearch.trim()) {
   265→      const q = novelSearch.toLowerCase();
   266→      list = list.filter(
   267→        (n) =>
   268→          n.title.toLowerCase().includes(q) ||
   269→          n.author.toLowerCase().includes(q) ||
   270→          n.description.toLowerCase().includes(q)
   271→      );
   272→    }
   273→    return list;
   274→  }, [novelList, novelGenre, novelSearch]);
   275→
   276→  const handleNovelSearch = useCallback((value: string) => {
   277→    setNovelSearch(value);
   278→    if (novelDebounceRef.current) clearTimeout(novelDebounceRef.current);
   279→  }, []);
   280→
   281→  /* ── Shared search input ── */
   282→  const searchPlaceholder = mainTab === 'manga'
   283→    ? 'Search manga, manhwa, manhua...'
   284→    : mainTab === 'comics'
   285→    ? 'Search Marvel, DC, Image comics...'
   286→    : 'Search classic novels...';
   287→
   288→  const currentSearch = mainTab === 'manga' ? mangaSearch : mainTab === 'comics' ? comicSearch : novelSearch;
   289→  const handleSearch = mainTab === 'manga' ? handleMangaSearch : mainTab === 'comics' ? handleComicSearch : handleNovelSearch;
   290→
   291→  /* ── Active color ── */
   292→  const activeColor = mainTab === 'manga' ? 'amber' : mainTab === 'comics' ? 'red' : 'emerald';
   293→
   294→  /* ── Render sub-tabs ── */
   295→  const subTabs: SubTab[] = mainTab === 'manga' ? MANGA_TABS : [];
   296→  const activeSubTab = mainTab === 'manga' ? mangaSubTab : '';
   297→  const setActiveSubTab = mainTab === 'manga' ? setMangaSubTab : () => {};
   298→
   299→  /* ── Item count ── */
   300→  const itemCount =
   301→    mainTab === 'manga' ? mangaTotal :
   302→    mainTab === 'comics' ? comicTotal :
   303→    filteredNovels.length;
   304→
   305→  /* ── Publisher icon helper ── */
   306→  const getPublisherIcon = (publisher: string) => {
   307→    switch (publisher) {
   308→      case 'Marvel': return <StarIcon className="w-3 h-3" />;
   309→      case 'DC': return <Crown className="w-3 h-3" />;
   310→      case 'Image': return <Eye className="w-3 h-3" />;
   311→      case 'Dark Horse': return <Skull className="w-3 h-3" />;
   312→      case 'IDW': return <Zap className="w-3 h-3" />;
   313→      default: return <Shield className="w-3 h-3" />;
   314→    }
   315→  };
   316→
   317→  return (
   318→    <div className="min-h-screen pb-10">
   319→      {/* Mobile back button */}
   320→      <button
   321→        onClick={goHome}
   322→        className="md:hidden fixed z-[90] flex items-center gap-1.5 text-white/60 active:text-white transition-colors"
   323→        style={{ top: 'max(env(safe-area-inset-top, 0px) + 8px, 8px)', left: 12 }}
   324→        aria-label="Go home"
   325→      >
   326→        <ArrowLeft className="w-5 h-5" />
   327→      </button>
   328→
   329→      <div className="pt-16 md:pt-8 pb-6 px-4 md:px-8">
   330→        {/* Desktop back */}
   331→        <button
   332→          onClick={goHome}
   333→          className="hidden md:flex items-center gap-2 text-white/70 hover:text-white mb-4 transition-colors"
   334→        >
   335→          <ArrowLeft className="w-4 h-4" />
   336→          <span className="text-sm">Back to Home</span>
   337→        </button>
   338→
   339→        {/* ── Hero header ── */}
   340→        <div className="relative mb-6">
   341→          <div className={`relative h-[20vh] min-h-[160px] md:min-h-[200px] rounded-2xl overflow-hidden`}>
   342→            <div className={`absolute inset-0 bg-gradient-to-br ${MAIN_TABS.find(t => t.key === mainTab)?.color || 'from-amber-500 to-orange-600'}`} />
   343→            <div className="absolute inset-0 opacity-20" style={{
   344→              backgroundImage: 'radial-gradient(circle at 20% 80%, rgba(255,255,255,0.15) 0%, transparent 50%), radial-gradient(circle at 80% 20%, rgba(255,255,255,0.1) 0%, transparent 50%)',
   345→            }} />
   346→            <div className="absolute inset-0 bg-gradient-to-t from-black/60 via-transparent to-black/20" />
   347→            <div className="relative h-full flex items-center justify-center text-center px-4">
   348→              <div>
   349→                <div className={`w-14 h-14 rounded-2xl bg-gradient-to-br ${MAIN_TABS.find(t => t.key === mainTab)?.color || ''} flex items-center justify-center shadow-lg mb-3`}>
   350→                  <BookOpen className="w-7 h-7 text-white" />
   351→                </div>
   352→                <h1 className="text-3xl md:text-4xl font-black text-white mb-1">
   353→                  Read
   354→                </h1>
   355→                <p className="text-white/50 text-xs md:text-sm max-w-md mx-auto">
   356→                  {mainTab === 'manga' && 'Manga, Manhwa & Manhua'}
   357→                  {mainTab === 'comics' && 'Marvel, DC, Image & More'}
   358→                  {mainTab === 'novels' && 'Classic Novels & Literature'}
   359→                  {' \u00B7 '}{itemCount > 0 && !mangaLoading && !comicLoading && !novelLoading ? `${itemCount.toLocaleString()} titles` : 'Free to read'}
   360→                </p>
   361→              </div>
   362→            </div>
   363→          </div>
   364→        </div>
   365→
   366→        {/* ── Main tabs ── */}
   367→        <div className="flex gap-1 mb-5">
   368→          {MAIN_TABS.map((tab) => {
   369→            const IconComp = tab.icon;
   370→            return (
   371→              <button
   372→                key={tab.key}
   373→                onClick={() => setMainTab(tab.key)}
   374→                className={`shrink-0 px-4 py-2.5 rounded-xl text-sm font-semibold transition-all relative ${
   375→                  mainTab === tab.key
   376→                    ? `bg-${activeColor}-500/15 text-${activeColor}-400 border border-${activeColor}-500/30`
   377→                    : 'bg-white/5 text-white/50 hover:bg-white/10 hover:text-white/70 border border-white/[0.08]'
   378→                }`}
   379→              >
   380→                <IconComp className="w-4 h-4" />
   381→                <span className="ml-1.5">{tab.label}</span>
   382→                {mainTab === tab.key && (
   383→                  <motion.div
   384→                    layoutId="read-main-tab"
   385→                    className={`absolute inset-0 rounded-xl bg-${activeColor}-500/15 border border-${activeColor}-500/30`}
   386→                    transition={{ type: 'spring', stiffness: 400, damping: 30 }}
   387→                  />
   388→                )}
   389→              </button>
   390→            );
   391→          })}
   392→        </div>
   393→
   394→        {/* ── Search ── */}
   395→        {mainTab !== 'novels' && (
   396→          <div className="relative mb-5">
   397→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-white/40" />
   398→            <input
   399→              type="text"
   400→              placeholder={searchPlaceholder}
   401→              value={currentSearch}
   402→              onChange={(e) => handleSearch(e.target.value)}
   403→              className={`w-full bg-white/5 border border-white/10 rounded-xl pl-10 pr-4 py-2.5 text-sm text-white placeholder:text-white/50 focus:outline-none focus:border-${activeColor}-500/50 focus:ring-1 focus:ring-${activeColor}-500/30 transition-colors`}
   404→            />
   405→          </div>
   406→        )}
   407→
   408→        {/* ── Sub-tabs (manga only) ── */}
   409→        {subTabs.length > 0 && (
   410→          <div className="flex gap-1 overflow-x-auto scrollbar-none pb-3 -mx-1 px-1">
   411→            {subTabs.map((tab) => (
   412→              <button
   413→                key={tab.key}
   414→                onClick={() => setActiveSubTab(tab.key)}
   415→                className={`shrink-0 px-3 py-1.5 rounded-lg text-xs font-medium transition-colors relative whitespace-nowrap ${
   416→                  activeSubTab === tab.key
   417→                    ? `text-${activeColor}-400`
   418→                    : 'text-white/40 hover:text-white/60'
   419→                }`}
   420→              >
   421→                {tab.flag && <span className="mr-1">{tab.flag}</span>}
   422→                {tab.label}
   423→                {activeSubTab === tab.key && (
   424→                  <motion.div
   425→                    layoutId="read-sub-tab"
   426→                    className={`absolute bottom-0 left-0 right-0 h-0.5 bg-${activeColor}-500 rounded-full`}
   427→                    transition={{ type: 'spring', stiffness: 400, damping: 30 }}
   428→                  />
   429→                )}
   430→              </button>
   431→            ))}
   432→          </div>
   433→        )}
   434→
   435→        {/* ── Comic publisher / genre / sort filters ── */}
   436→        {mainTab === 'comics' && (
   437→          <div className="space-y-3 mb-5">
   438→            {/* Publisher filter pills */}
   439→            <div className="flex gap-1.5 overflow-x-auto scrollbar-none -mx-1 px-1">
   440→              {COMIC_PUBLISHERS.map((pub) => (
   441→                <button
   442→                  key={pub}
   443→                  onClick={() => setComicPublisher(pub)}
   444→                  className={`shrink-0 flex items-center gap-1.5 px-3 py-1.5 rounded-full text-xs font-semibold transition-all border ${
   445→                    comicPublisher === pub
   446→                      ? (PUBLISHER_COLORS[pub] || 'bg-white/15 text-white/90 border-white/20')
   447→                      : 'bg-white/[0.04] text-white/40 border-white/[0.08] hover:bg-white/[0.08] hover:text-white/70'
   448→                  }`}
   449→                >
   450→                  {getPublisherIcon(pub)}
   451→                  {pub}
   452→                </button>
   453→              ))}
   454→            </div>
   455→
   456→            {/* Genre filter row */}
   457→            <div className="flex gap-1.5 overflow-x-auto scrollbar-none -mx-1 px-1">
   458→              {COMIC_GENRES.map((genre) => (
   459→                <button
   460→                  key={genre}
   461→                  onClick={() => setComicGenre(genre)}
   462→                  className={`shrink-0 px-2.5 py-1 rounded-full text-[11px] font-medium transition-all border ${
   463→                    comicGenre === genre
   464→                      ? 'bg-red-500/15 text-red-300 border-red-500/30'
