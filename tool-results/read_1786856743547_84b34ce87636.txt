   341→        </AnimatePresence>
   342→      </div>
   343→      <AuthModal open={authModalOpen} onClose={() => setAuthModalOpen(false)} />
   344→    </div>

'bg-white/[0.08] ring-2' : 'hover:bg-white/[0.04]'}`} style={selectedColor === c.value ? { ringColor: c.value, boxShadow: `0 0 20px ${c.value}44` } : {}}>

   456→              </div>

   458→              <button onClick={() => setSelectedColor(getAvatarDef(selectedAvatar).color)} className="w-full py-2.5 rounded-xl bg-white/[0.04] border border-white/[0.06] text-white/50 hover:text-white text-xs font-medium transition-all hover:bg-white/[0.06]">

'text-white shadow-lg' : 'text-white/50 bg-white/[0.04] hover:bg-white/[0.06] border border-white/[0.06]'}`} style={selectedGenres.includes(genre) ? { backgroundColor: `${selectedColor}cc`, boxShadow: `0 4px 16px ${selectedColor}44` } : {}}>

   479→              {selectedGenres.length > 0 && (
   480→                <p className="text-xs text-white/30">Selected: {selectedGenres.join(', ')}</p>
   481→              )}
   482→            </motion.div>
   483→          )}
   484→        </AnimatePresence>
   485→
   486→        {/* Error */}
   487→        {saveError && <p className="text-sm text-red-400 bg-red-500/10 border border-red-500/20 rounded-lg px-3 py-2">{saveError}</p>}
   488→
   489→        {/* Buttons */}
   490→        <div className="flex gap-3 pt-2">
   491→          <button onClick={onSave} disabled={saving} className="flex items-center gap-2 px-5 py-2.5 text-white text-sm font-semibold rounded-xl transition-all hover:brightness-110 disabled:opacity-50" style={{ backgroundColor: accentColor, boxShadow: `0 4px 16px ${accentColor}44` }}>
   492→            {saving ? <Loader2 className="w-4 h-4 animate-spin" /> : <Check className="w-4 h-4" />}Save
   493→          </button>
   494→          <button onClick={onCancel} className="flex items-center gap-2 px-5 py-2.5 bg-white/[0.06] hover:bg-white/10 text-white/60 text-sm font-medium rounded-xl transition-all">
   495→            <X className="w-4 h-4" />Cancel
   496→          </button>
   497→        </div>
   498→      </div>
   499→    </div>
   500→  );
   501→}

   527→function WatchlistCard({ tmdbId, onClick }: { tmdbId: number; onClick: (id: number, type: string) => void }) {
   528→  const [data, setData] = useState<{ title: string; posterPath: string | null; mediaType: string } | null>(null);
   529→

   531→    Promise.any([
   532→      fetch(`/api/tmdb/preview?id=${tmdbId}&type=movie`).then(r => r.json()),
   533→      fetch(`/api/tmdb/preview?id=${tmdbId}&type=tv`).then(r => r.json()),
   534→    ]).then(d => {
   535→      if (d.title || d.name) setData({ title: d.title || d.name, posterPath: d.poster_path, mediaType: d.media_type || (d.first_air_date ? 'tv' : 'movie') });
   536→    }).catch(() => {});
   537→  }, [tmdbId]);
   538→
   539→  return (
   540→    <button onClick={() => data && onClick(tmdbId, data.mediaType)} className="group text-left">
   541→      <div className="aspect-[2/3] rounded-xl bg-white/[0.06] overflow-hidden mb-2 relative">
   542→        {data?.posterPath ? (
   543→          <img src={`https://image.tmdb.org/t/p/w342${data.posterPath}`} alt={data.title} className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-300" />
   544→        ) : (
   545→          <div className="w-full h-full flex items-center justify-center"><Film className="w-8 h-8 text-white/10" /></div>
   546→        )}
   547→      </div>

   549→    </button>
   550→  );
