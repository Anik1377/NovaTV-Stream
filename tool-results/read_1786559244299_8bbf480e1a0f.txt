    53→    53→function formatDuration(iso: string): string {
    54→    54→  const m = iso.match(/PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?/);
    55→    55→  if (!m) return '0:00';
    56→    56→  const h = parseInt(m[1] || '0');
    57→    57→  const min = parseInt(m[2] || '0');
    58→    58→  const s = parseInt(m[3] || '0');
    59→    59→  if (h > 0) return `${h}:${min.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    60→    60→  return `${min}:${s.toString().padStart(2, '0')}`;
    61→    61→}
    62→    62→
    63→    63→function formatDurationSeconds(iso: string): number {
    64→    64→  const m = iso.match(/PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?/);
    65→    65→  if (!m) return 0;
    66→    66→  const h = parseInt(m[1] || '0');
    67→    67→  const min = parseInt(m[2] || '0');
    68→    68→  const s = parseInt(m[3] || '0');
    69→    69→  return h * 3600 + min * 60 + s;
    70→    70→}
    71→    71→
    72→    72→function formatTime(sec: number): string {
    73→    73→  const m = Math.floor(sec / 60);
    74→    74→  const s = Math.floor(sec % 60);
    75→    75→  return `${m}:${s.toString().padStart(2, '0')}`;
    76→    76→}
    77→    77→
    78→    78→function formatViews(n: number): string {
    79→    79→  if (n >= 1_000_000_000) return `${(n / 1_000_000_000).toFixed(1)}B`;
    80→    80→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
    81→    81→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`;
    82→    82→  return String(n);
    83→    83→}

   475→   475→function MiniPlayer({
   476→   476→  track,
   477→   477→  onExpand,
   478→   478→  onNext,
   479→   479→  onPrev,
