    53→function formatDuration(iso: string): string {
    54→  const m = iso.match(/PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?/);
    55→  if (!m) return '0:00';
    56→  const h = parseInt(m[1] || '0');
    57→  const min = parseInt(m[2] || '0');
    58→  const s = parseInt(m[3] || '0');
    59→  if (h > 0) return `${h}:${min.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    60→  return `${min}:${s.toString().padStart(2, '0')}`;
    61→}
    62→
    63→function formatDurationSeconds(iso: string): number {
    64→  const m = iso.match(/PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?/);
    65→  if (!m) return 0;
    66→  const h = parseInt(m[1] || '0');
    67→  const min = parseInt(m[2] || '0');
    68→  const s = parseInt(m[3] || '0');
    69→  return h * 3600 + min * 60 + s;
    70→}
    71→
    72→function formatTime(sec: number): string {
    73→  const m = Math.floor(sec / 60);
    74→  const s = Math.floor(sec % 60);
    75→  return `${m}:${s.toString().padStart(2, '0')}`;
    76→}
    77→
    78→function formatViews(n: number): string {
    79→  if (n >= 1_000_000_000) return `${(n / 1_000_000_000).toFixed(1)}B`;
    80→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
    81→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`;
    82→  return String(n);
    83→}

    99→function useAudioPlayer(videoId: string | null, trackMeta: TrackMeta | null) {
   100→  const audioRef = useRef<HTMLAudioElement | null>(null);
   101→  const [state, setState] = useState<PlayerState>({ isPlaying: false, currentTime: 0, duration: 0 });
   102→  const onEndRef = useRef<() => void>(() => {});
   103→  const onNextRef = useRef<() => void>(() => {});
   104→  const onPrevRef = useRef<() => void>(() => {});
   105→
   106→  // Lazy-create the audio element once
   107→  const getAudio = useCallback(() => {
   108→    if (!audioRef.current) {
   109→      audioRef.current = new Audio();
   110→      audioRef.current.preload = 'auto';
   111→    }
   112→    return audioRef.current;
   113→  }, []);
   114→
   115→  // Fetch audio URL and play when videoId changes
   116→  useEffect(() => {
   117→    const audio = getAudio();
   118→
   119→    if (!videoId) {
   120→      audio.pause();
   121→      audio.src = '';
   122→      return;
   123→    }
   124→
   125→    let cancelled = false;
   126→
   127→    const loadAndPlay = async () => {
   128→      try {
   129→        const res = await fetch(`/api/youtube/audio?videoId=${videoId}`);
   130→        const data = await res.json();
   131→        if (cancelled || !data.url) return;
   132→
   133→        audio.src = data.url;
   134→        if (data.duration) {
   135→          setState((s) => ({ ...s, duration: data.duration }));
   136→        }
   137→
   138→        await audio.play();
   139→      } catch (err) {
   140→        console.error('Failed to load audio:', err);
   141→      }
   142→    };
   143→
   144→    loadAndPlay();
   145→
   146→    // Native audio events
   147→    const onTimeUpdate = () => {
   148→      if (audio.duration && isFinite(audio.duration)) {
   149→        setState((s) => ({ ...s, currentTime: audio.currentTime, duration: audio.duration }));
   150→      }
   151→    };
   152→    const onPlay = () => setState((s) => ({ ...s, isPlaying: true }));
   153→    const onPause = () => setState((s) => ({ ...s, isPlaying: false }));
   154→    const onEnded = () => {
   155→      setState((s) => ({ ...s, isPlaying: false }));
   156→      onEndRef.current();
   157→    };
   158→    const onDurationChange = () => {
   159→      if (audio.duration && isFinite(audio.duration)) {
   160→        setState((s) => ({ ...s, duration: audio.duration }));
   161→      }
   162→    };
   163→
   164→    audio.addEventListener('timeupdate', onTimeUpdate);
   165→    audio.addEventListener('play', onPlay);
   166→    audio.addEventListener('pause', onPause);
   167→    audio.addEventListener('ended', onEnded);
   168→    audio.addEventListener('durationchange', onDurationChange);
   169→
   170→    // Update Media Session position state periodically
   171→    const posIv = setInterval(() => {
   172→      if ('mediaSession' in navigator && audio.duration && isFinite(audio.duration) && !audio.paused) {
   173→        try {
   174→          navigator.mediaSession.setPositionState({
   175→            duration: audio.duration,
   176→            playbackRate: audio.playbackRate,
   177→            position: audio.currentTime,
   178→          });
   179→        } catch { /* ignore */ }
   180→      }
   181→    }, 1000);
   182→
   183→    return () => {
   184→      cancelled = true;
   185→      clearInterval(posIv);
   186→      audio.removeEventListener('timeupdate', onTimeUpdate);
   187→      audio.removeEventListener('play', onPlay);
   188→      audio.removeEventListener('pause', onPause);
   189→      audio.removeEventListener('ended', onEnded);
   190→      audio.removeEventListener('durationchange', onDurationChange);
   191→      audio.pause();
   192→      audio.src = '';
   193→    };
   194→  }, [videoId, getAudio]);
   195→
   196→  // Media Session metadata & action handlers
   197→  useEffect(() => {
   198→    if (typeof navigator === 'undefined' || !('mediaSession' in navigator)) return;
   199→    const audio = getAudio();
   200→
   201→    if (trackMeta) {
   202→      navigator.mediaSession.metadata = new MediaMetadata({
   203→        title: trackMeta.title,
   204→        artist: trackMeta.artist,
   205→        album: 'StreamVault Music',
   206→        artwork: [
   207→          { src: trackMeta.artwork, sizes: '512x512', type: 'image/jpeg' },
   208→          { src: trackMeta.artwork, sizes: '192x192', type: 'image/jpeg' },
   209→        ],
   210→      });
   211→    }
   212→
   213→    if (navigator.mediaSession.setActionHandler) {
   214→      navigator.mediaSession.setActionHandler('play', () => { audio.play(); });
   215→      navigator.mediaSession.setActionHandler('pause', () => { audio.pause(); });
   216→      navigator.mediaSession.setActionHandler('previoustrack', () => onPrevRef.current());
   217→      navigator.mediaSession.setActionHandler('nexttrack', () => onNextRef.current());
   218→      navigator.mediaSession.setActionHandler('seekto', (details) => {
   219→        if (details && details.seekTime !== undefined) {
   220→          audio.currentTime = details.seekTime;
   221→        }
   222→      });
   223→      navigator.mediaSession.setActionHandler('stop', () => { audio.pause(); });
   224→    }
   225→  }, [trackMeta, getAudio]);
   226→
   227→  const togglePlay = useCallback(() => {
   228→    const audio = getAudio();
   229→    if (audio.paused) audio.play();
   230→    else audio.pause();
   231→  }, [getAudio]);
   232→
   233→  const seekTo = useCallback((seconds: number) => {
   234→    getAudio().currentTime = seconds;
   235→  }, [getAudio]);
   236→
   237→  const setOnEnd = useCallback((cb: () => void) => { onEndRef.current = cb; }, []);
   238→  const setOnNext = useCallback((cb: () => void) => { onNextRef.current = cb; }, []);
   239→  const setOnPrev = useCallback((cb: () => void) => { onPrevRef.current = cb; }, []);
   240→
   241→  return { ...state, togglePlay, seekTo, setOnEnd, setOnNext, setOnPrev };
   242→}

   491→function FullScreenPlayer({
   492→  track,
   493→  onClose,
   494→  onNext,
   495→  onPrev,
   496→  isPlaying,
   497→  currentTime,
   498→  duration,
   499→  onTogglePlay,

   504→  onNext: () => void;
   505→  onPrev: () => void;

   507→  currentTime: number;
   508→  duration: number;
   509→  onTogglePlay: () => void;
   510→  onSeek: (sec: number) => void;
   511→}) {
   512→  const [imgError, setImgError] = useState(false);
   513→  const [liked, setLiked] = useState(false);
