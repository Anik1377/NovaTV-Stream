    44→function formatDuration(iso: string): string {
    45→  const m = iso.match(/PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?/);
    46→  if (!m) return '0:00';
    47→  const h = parseInt(m[1] || '0');
    48→  const min = parseInt(m[2] || '0');
    49→  const s = parseInt(m[3] || '0');
    50→  if (h > 0) return `${h}:${min.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    51→  return `${min}:${s.toString().padStart(2, '0')}`;
    52→}
    53→
    54→function formatViewCount(count: number): string {
    55→  if (count >= 1_000_000_000) return `${(count / 1_000_000_000).toFixed(1)}B`;
    56→  if (count >= 1_000_000) return `${(count / 1_000_000).toFixed(1)}M`;
    57→  if (count >= 1_000) return `${(count / 1_000).toFixed(1)}K`;
    58→  return `${count}`;
    59→}
    60→
    61→function formatDurationSeconds(iso: string): number {
    62→  const m = iso.match(/PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?/);
    63→  if (!m) return 0;
    64→  const h = parseInt(m[1] || '0');
    65→  const min = parseInt(m[2] || '0');
    66→  const s = parseInt(m[3] || '0');
    67→  return h * 3600 + min * 60 + s;
    68→}
    69→
    70→function formatTime(sec: number): string {
    71→  const m = Math.floor(sec / 60);
    72→  const s = Math.floor(sec % 60);
    73→  return `${m}:${s.toString().padStart(2, '0')}`;
    74→}

   254→function FullScreenPlayer({
   255→  track,
   256→  onClose,
   257→  onNext,
   258→  onPrev,
   259→}: {
   260→  track: YoutubeVideoResult;
   261→  onClose: () => void;
   262→  onNext: () => void;
   263→  onPrev: () => void;
   264→}) {
   265→  const [imgError, setImgError] = useState(false);
   266→  const [progress, setProgress] = useState(0);
   267→  const totalSec = formatDurationSeconds(track.duration);
   268→  const [liked, setLiked] = useState(false);
   269→
   270→  // Simulated progress
   271→  useEffect(() => {
   272→    setProgress(0);
   273→    const interval = setInterval(() => {
   274→      setProgress((p) => {
   275→        if (p >= 100) return 100;
   276→        return p + (100 / Math.max(totalSec, 1)) * 0.5;
   277→      });
   278→    }, 500);
   279→    return () => clearInterval(interval);
   280→  }, [track.videoId, totalSec]);
   281→
   282→  const currentSec = (progress / 100) * totalSec;
   283→
   284→  return (
   285→    <motion.div
   286→      initial={{ y: '100%' }}
   287→      animate={{ y: 0 }}
   288→      exit={{ y: '100%' }}
   289→      transition={{ type: 'spring', stiffness: 300, damping: 30 }}
   290→      className="fixed inset-0 z-[60] flex flex-col"
   291→    >
   292→      {/* Blurred background */}
   293→      <div className="absolute inset-0 z-0">
   294→        {!imgError ? (
   295→          <img
   296→            src={track.thumbnail}
   297→            alt=""
   298→            className="w-full h-full object-cover scale-150 blur-3xl opacity-40"
   299→            onError={() => setImgError(true)}
   300→          />
   301→        ) : (
   302→          <div className="w-full h-full bg-gradient-to-b from-amber-900 to-black" />
   303→        )}
   304→        <div className="absolute inset-0 bg-black/40" />
   305→        <div className="absolute inset-0 bg-gradient-to-b from-black/60 via-transparent to-black/80" />
   306→      </div>
   307→
   308→      {/* Content */}
   309→      <div className="relative z-10 flex flex-col h-full">
   310→        {/* Top bar */}
   311→        <div className="flex items-center justify-between px-4 pt-[env(safe-area-inset-top,12px)] pb-2">
   312→          <Button
   313→            variant="ghost"
   314→            size="icon"
   315→            className="text-white/70 hover:text-white hover:bg-white/10"
   316→            onClick={onClose}
   317→          >
   318→            <ChevronDown className="w-6 h-6" />
   319→          </Button>
   320→          <div className="text-center">
   321→            <p className="text-[11px] font-medium text-white/40 uppercase tracking-wider">Playing from</p>
   322→            <p className="text-xs text-white/60">YouTube Music</p>
   323→          </div>
   324→          <Button
   325→            variant="ghost"
   326→            size="icon"
   327→            className="text-white/70 hover:text-white hover:bg-white/10"
   328→            onClick={() => setLiked(!liked)}
   329→          >
   330→            <Heart className={`w-5 h-5 transition-colors ${liked ? 'text-rose-400 fill-rose-400' : ''}`} />
   331→          </Button>
   332→        </div>
   333→
   334→        {/* Album art */}
   335→        <div className="flex-1 flex items-center justify-center px-8 py-4">
   336→          <motion.div
   337→            initial={{ scale: 0.8, opacity: 0 }}
   338→            animate={{ scale: 1, opacity: 1 }}
   339→            transition={{ delay: 0.1, type: 'spring', stiffness: 200 }}
   340→            className="w-full max-w-[300px] md:max-w-[360px] aspect-square rounded-2xl overflow-hidden shadow-2xl shadow-black/50"
   341→          >
   342→            {!imgError ? (
   343→              <img
   344→                src={track.thumbnail}
   345→                alt={track.title}
   346→                className="w-full h-full object-cover"
   347→                onError={() => setImgError(true)}
   348→              />
   349→            ) : (
   350→              <div className="w-full h-full bg-gradient-to-br from-amber-900/60 to-orange-900/60 flex items-center justify-center">
   351→                <Music className="w-16 h-16 text-white/30" />
   352→              </div>
   353→            )}
   354→          </motion.div>
   355→        </div>
   356→
   357→        {/* Track info + controls */}
   358→        <div className="px-6 pb-[env(safe-area-inset-bottom,16px)]">
   359→          {/* Title & artist */}
   360→          <div className="flex items-start justify-between gap-3 mb-4">
   361→            <div className="min-w-0 flex-1">

   363→              <p className="text-sm text-white/50 truncate mt-0.5">{track.channelTitle}</p>
   364→            </div>

   441→            />
   442→          </div>
   443→        </div>
   444→      </div>
   445→    </motion.div>
   446→  );
   447→}

   549→export function MusicPage() {
   550→  const [trending, setTrending] = useState<YoutubeVideoResult[]>([]);
   551→  const [searchResults, setSearchResults] = useState<YoutubeVideoResult[]>([]);
   552→  const [relatedResults, setRelatedResults] = useState<YoutubeVideoResult[]>([]);
   553→  const [loading, setLoading] = useState(true);
   554→  const [searchLoading, setSearchLoading] = useState(false);
   555→  const [searchQuery, setSearchQuery] = useState('');
   556→  const [activeMood, setActiveMood] = useState('');
   557→  const [currentTrack, setCurrentTrack] = useState<YoutubeVideoResult | null>(null);
   558→  const [showFullScreen, setShowFullScreen] = useState(false);
   559→  const [queue, setQueue] = useState<YoutubeVideoResult[]>([]);
   560→  const [queueIndex, setQueueIndex] = useState(0);
   561→  const [showQueue, setShowQueue] = useState(false);
   562→  const searchTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   563→  const searchInputRef = useRef<HTMLInputElement>(null);
   564→
   565→  // Fetch trending on mount
   566→  useEffect(() => {
   567→    async function fetchTrending() {
   568→      try {
   569→        const res = await fetch('/api/youtube/trending');
   570→        const data: ApiResponse = await res.json();
   571→        setTrending(data.results || []);
   572→      } catch (err) {
   573→        console.error('Failed to fetch trending:', err);
   574→      } finally {
   575→        setLoading(false);
   576→      }
   577→    }
   578→    fetchTrending();
   579→  }, []);
   580→
   581→  // Search with debounce
   582→  const handleSearchInput = useCallback((value: string) => {
   583→    setSearchQuery(value);
   584→    setActiveMood('');
   585→    if (searchTimerRef.current) clearTimeout(searchTimerRef.current);
   586→    if (!value.trim()) {
   587→      setSearchResults([]);
   588→      setRelatedResults([]);
   589→      return;
   590→    }
   591→    searchTimerRef.current = setTimeout(async () => {
   592→      setSearchLoading(true);
   593→      try {
   594→        const res = await fetch(`/api/youtube/search?query=${encodeURIComponent(value)}`);
   595→        const data: ApiResponse = await res.json();
   596→        setSearchResults(data.results || []);
   597→        if (data.results?.[0]) {
   598→          const relatedRes = await fetch(`/api/youtube/related?videoId=${data.results[0].videoId}`);
   599→          const relatedData: ApiResponse = await relatedRes.json();
   600→          setRelatedResults(relatedData.results || []);
   601→        } else {
   602→          setRelatedResults([]);
   603→        }
   604→      } catch {
   605→        setSearchResults([]);
   606→        setRelatedResults([]);
   607→      } finally {
   608→        setSearchLoading(false);
   609→      }
   610→    }, 500);
   611→  }, []);
   612→
   613→  // Mood card click
   614→  const handleMoodClick = useCallback((mood: typeof MOOD_CARDS[number]) => {
   615→    const isCurrentlyActive = activeMood === mood.label;
   616→    setActiveMood(isCurrentlyActive ? '' : mood.label);
   617→    if (!isCurrentlyActive) {
   618→      handleSearchInput(mood.query);
   619→      if (searchInputRef.current) searchInputRef.current.value = mood.query;
   620→    } else {
   621→      setSearchQuery('');
   622→      setSearchResults([]);
   623→      setRelatedResults([]);
   624→      if (searchInputRef.current) searchInputRef.current.value = '';
   625→    }
   626→  }, [activeMood, handleSearchInput]);
   627→
   628→  // Play a track
   629→  const handlePlay = useCallback((track: YoutubeVideoResult, trackList?: YoutubeVideoResult[]) => {
   630→    setCurrentTrack(track);
   631→    setShowFullScreen(true);
   632→    if (trackList) {
   633→      const idx = trackList.findIndex((t) => t.videoId === track.videoId);
   634→      setQueue(trackList);
   635→      setQueueIndex(idx >= 0 ? idx : 0);
   636→    }
   637→  }, []);
   638→
   639→  // Next/Prev
   640→  const handleNext = useCallback(() => {
   641→    if (queue.length === 0) return;
   642→    const nextIdx = (queueIndex + 1) % queue.length;
   643→    setQueueIndex(nextIdx);
   644→    setCurrentTrack(queue[nextIdx]);
   645→  }, [queue, queueIndex]);
   646→
   647→  const handlePrev = useCallback(() => {
   648→    if (queue.length === 0) return;
   649→    const prevIdx = (queueIndex - 1 + queue.length) % queue.length;
   650→    setQueueIndex(prevIdx);
   651→    setCurrentTrack(queue[prevIdx]);
   652→  }, [queue, queueIndex]);
   653→
   654→  // Close
   655→  const handleClosePlayer = useCallback(() => {
   656→    setShowFullScreen(false);
   657→  }, []);
   658→
   659→  const handleStopCompletely = useCallback(() => {
   660→    setCurrentTrack(null);
   661→    setShowFullScreen(false);
   662→  }, []);
   663→
   664→  const isSearching = searchQuery.trim().length > 0;
   665→  const displayList = isSearching ? searchResults : trending;
   666→
   667→  const hasPlayer = currentTrack !== null;
   668→
   669→  return (
   670→    <div className={`${hasPlayer && !showFullScreen ? 'pb-[7.5rem]' : 'pb-24'}`}>
   671→      {/* ── Search Bar (sticky) ── */}
   672→      <div className="sticky top-[calc(3rem+env(safe-area-inset-top,0px))] z-30 bg-black/80 backdrop-blur-xl px-4 py-3">
   673→        <div className="relative">
   674→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-white/30" />
   675→          <Input
   676→            ref={searchInputRef}
   677→            type="text"
   678→            placeholder="Search songs, artists..."
   679→            defaultValue={searchQuery}
   680→            onChange={(e) => handleSearchInput(e.target.value)}
   681→            className="pl-10 pr-10 bg-white/10 border-white/10 text-white placeholder:text-white/30 focus:border-amber-500/50 h-10 rounded-xl text-sm"
   682→          />
   683→          {searchQuery && (
   684→            <button
   685→              onClick={() => {
   686→                handleSearchInput('');
   687→                if (searchInputRef.current) searchInputRef.current.value = '';
   688→              }}
   689→              className="absolute right-3 top-1/2 -translate-y-1/2 text-white/30 hover:text-white transition-colors"
   690→            >
   691→              <X className="w-4 h-4" />
   692→            </button>
   693→          )}
   694→        </div>
   695→      </div>
   696→
   697→      {/* ── Mood / Genre Cards ── */}
   698→      {!isSearching && (
   699→        <div className="px-4 mb-6">
   700→          <SectionHeader title="Browse Moods" icon={<span className="text-lg">🎵</span>} />
   701→          <div className="flex gap-3 overflow-x-auto pb-2 -mx-4 px-4 content-scroll">
   702→            {MOOD_CARDS.map((mood) => (
   703→              <MoodCard
   704→                key={mood.label}
   705→                label={mood.label}
   706→                gradient={mood.gradient}
   707→                emoji={mood.emoji}
   708→                onClick={() => handleMoodClick(mood)}
   709→                isActive={activeMood === mood.label}
   710→              />
   711→            ))}
   712→          </div>
   713→        </div>
   714→      )}
   715→
   716→      {/* ── Genre Pills (when searching) ── */}
   717→      {isSearching && (
   718→        <div className="px-4 mb-4">
   719→          <div className="flex gap-2 overflow-x-auto pb-1 -mx-4 px-4 content-scroll">
   720→            {GENRE_PILLS.map((genre) => (
   721→              <button
   722→                key={genre}
   723→                onClick={() => {
   724→                  handleSearchInput(genre);
   725→                  if (searchInputRef.current) searchInputRef.current.value = genre;
   726→                }}
   727→                className="shrink-0 px-3 py-1 rounded-full text-xs font-medium bg-white/5 text-white/50 border border-white/5 active:bg-white/10 active:text-white transition-colors"
   728→              >
   729→                {genre}
   730→              </button>

   732→          </div>
   733→        </div>
   734→      )}
   735→
   736→      {/* ── Content ── */}
