    76→/* ── Helpers ── */
    77→function formatDuration(seconds: number): string {
    78→  const m = Math.floor(seconds / 60);
    79→  const s = seconds % 60;
    80→  return `${m}:${s.toString().padStart(2, '0')}`;
    81→}
    82→
    83→function formatPlayCount(count: number): string {
    84→  if (count >= 1000000) return `${(count / 1000000).toFixed(1)}M`;
    85→  if (count >= 1000) return `${(count / 1000).toFixed(1)}K`;
    86→  return String(count);
    87→}

    89→function useDebounce<T>(value: T, delay: number): T {
    90→  const [debounced, setDebounced] = useState(value);
    91→  useEffect(() => {
    92→    const timer = setTimeout(() => setDebounced(value), delay);
    93→    return () => clearTimeout(timer);
    94→  }, [value, delay]);
    95→  return debounced;
    96→}

   122→/* ════════════════════════════════════════════════════════

   125→export function MusicPage() {
   126→  const { goHome } = useAppStore();
   127→
   128→  /* ── Search state ── */
   129→  const [query, setQuery] = useState('');
   130→  const debouncedQuery = useDebounce(query, 400);
   131→  const [activeLang, setActiveLang] = useState('all');
   132→
   133→  /* ── Results state ── */
   134→  const [songs, setSongs] = useState<Song[]>([]);
   135→  const [total, setTotal] = useState(0);
   136→  const [page, setPage] = useState(1);
   137→  const [loading, setLoading] = useState(false);
   138→  const [error, setError] = useState('');
   139→  const [hasSearched, setHasSearched] = useState(false);
   140→  const abortRef = useRef<AbortController | null>(null);
   141→
   142→  /* ── Player state ── */
   143→  const [currentSong, setCurrentSong] = useState<Song | null>(null);
   144→  const [isPlaying, setIsPlaying] = useState(false);
   145→  const [progress, setProgress] = useState(0);
   146→  const [currentTime, setCurrentTime] = useState(0);
   147→  const audioRef = useRef<HTMLAudioElement>(null);
   148→  const progressInterval = useRef<ReturnType<typeof setInterval> | null>(null);
   149→
   150→  /* ── Filtered songs ── */
   151→  const filteredSongs = activeLang === 'all'
   152→    ? songs
   153→    : songs.filter((s) => s.language === activeLang);
   154→
   155→  /* ── Fetch songs ── */
   156→  const fetchSongs = useCallback(async (q: string, p: number) => {
   157→    if (!q.trim()) {
   158→      setSongs([]);
   159→      setTotal(0);
   160→      setLoading(false);
   161→      return;
   162→    }
   163→
   164→    // Abort previous request
   165→    if (abortRef.current) abortRef.current.abort();
   166→    const controller = new AbortController();
   167→    abortRef.current = controller;
   168→
   169→    setLoading(true);
   170→    setError('');
   171→
   172→    try {
   173→      const res = await fetch(
   174→        `/api/music/search?query=${encodeURIComponent(q)}&page=${p}&limit=20`,
   175→        { signal: controller.signal }
   176→      );
   177→      const data: SearchResponse = await res.json();
   178→
   179→      if (controller.signal.aborted) return;
   180→
   181→      if (data.error) {
   182→        setError(data.error);
   183→        setSongs([]);
   184→      } else {
   185→        if (p === 1) {
   186→          setSongs(data.songs);
   187→        } else {
   188→          setSongs((prev) => {
   189→            const existingIds = new Set(prev.map((s) => s.id));
   190→            const newSongs = data.songs.filter((s) => !existingIds.has(s.id));
   191→            return [...prev, ...newSongs];
   192→          });
   193→        }
   194→        setTotal(data.total);
   195→      }
   196→    } catch (err) {
   197→      if (err instanceof DOMException && err.name === 'AbortError') return;
   198→      setError('Failed to search. Please try again.');
   199→      setSongs([]);
   200→    } finally {
   201→      if (!controller.signal.aborted) {
   202→        setLoading(false);
   203→      }
   204→    }
   205→  }, []);
   206→
   207→  /* ── Search effect ── */
   208→  useEffect(() => {
   209→    setPage(1);
   210→    setHasSearched(true);
   211→    fetchSongs(debouncedQuery, 1);
   212→  }, [debouncedQuery, fetchSongs]);
   213→
   214→  /* ── Load more ── */
   215→  const loadMore = useCallback(() => {
   216→    const nextPage = page + 1;
   217→    setPage(nextPage);
   218→    fetchSongs(debouncedQuery, nextPage);
   219→  }, [page, debouncedQuery, fetchSongs]);
   220→
   221→  /* ── Audio player ── */
   222→  const playSong = useCallback((song: Song) => {
   223→    if (currentSong?.id === song.id) {
   224→      // Toggle play/pause for same song
   225→      if (isPlaying) {
   226→        audioRef.current?.pause();
   227→        setIsPlaying(false);
   228→      } else {
   229→        audioRef.current?.play().catch(() => {});
   230→        setIsPlaying(true);
   231→      }
   232→      return;
   233→    }
   234→
   235→    setCurrentSong(song);
   236→    setIsPlaying(true);
   237→    setProgress(0);
   238→    setCurrentTime(0);
   239→
   240→    // Play will be triggered by the useEffect below when currentSong changes
   241→  }, [currentSong, isPlaying]);
   242→
   243→  const pauseSong = useCallback(() => {
   244→    audioRef.current?.pause();
   245→    setIsPlaying(false);
   246→  }, []);
   247→
   248→  const resumeSong = useCallback(() => {
   249→    audioRef.current?.play().catch(() => {});
   250→    setIsPlaying(true);
   251→  }, []);
   252→
   253→  const closePlayer = useCallback(() => {
   254→    if (audioRef.current) {
   255→      audioRef.current.pause();
   256→      audioRef.current.src = '';
   257→    }
   258→    setCurrentSong(null);
   259→    setIsPlaying(false);
   260→    setProgress(0);
   261→    setCurrentTime(0);
   262→  }, []);
   263→
   264→  /* ── Load song into audio element ── */
   265→  useEffect(() => {
   266→    const audio = audioRef.current;
   267→    if (!audio || !currentSong) return;
   268→
   269→    audio.src = currentSong.previewUrl;
   270→    audio.load();
   271→    audio.play().catch(() => {
   272→      setIsPlaying(false);
   273→    });
   274→  }, [currentSong]);
   275→
   276→  /* ── Progress tracking ── */
   277→  useEffect(() => {
   278→    const audio = audioRef.current;
   279→    if (!audio) return;
   280→
   281→    const updateTime = () => {
   282→      if (audio.duration) {
   283→        setCurrentTime(audio.currentTime);
   284→        setProgress((audio.currentTime / audio.duration) * 100);
   285→      }
   286→    };
   287→
   288→    const onEnded = () => {
   289→      setIsPlaying(false);
   290→      setProgress(100);
   291→      // Auto-play next song
   292→      if (currentSong) {
   293→        const idx = filteredSongs.findIndex((s) => s.id === currentSong.id);
   294→        if (idx >= 0 && idx < filteredSongs.length - 1) {
   295→          playSong(filteredSongs[idx + 1]);
   296→        }
   297→      }
   298→    };
   299→
   300→    const onPlay = () => setIsPlaying(true);
   301→    const onPause = () => setIsPlaying(false);
   302→
   303→    audio.addEventListener('timeupdate', updateTime);
   304→    audio.addEventListener('ended', onEnded);
   305→    audio.addEventListener('play', onPlay);
   306→    audio.addEventListener('pause', onPause);
   307→
   308→    return () => {
   309→      audio.removeEventListener('timeupdate', updateTime);
   310→      audio.removeEventListener('ended', onEnded);
   311→      audio.removeEventListener('play', onPlay);
   312→      audio.removeEventListener('pause', onPause);
   313→    };
   314→  }, [currentSong, filteredSongs, playSong]);
   315→
   316→  /* ── Seek handler ── */
   317→  const handleSeek = useCallback((e: React.MouseEvent<HTMLDivElement>) => {
   318→    const audio = audioRef.current;
   319→    if (!audio || !audio.duration) return;
   320→    const rect = e.currentTarget.getBoundingClientRect();
   321→    const ratio = Math.max(0, Math.min(1, (e.clientX - rect.left) / rect.width));
   322→    audio.currentTime = ratio * audio.duration;
   323→  }, []);
   324→
   325→  /* ── Keyboard shortcut ── */
   326→  useEffect(() => {
   327→    const handleKey = (e: KeyboardEvent) => {
   328→      if (e.target instanceof HTMLInputElement) return;
   329→      if (e.code === 'Space' && currentSong) {
   330→        e.preventDefault();
   331→        if (isPlaying) pauseSong(); else resumeSong();
   332→      }
   333→    };
   334→    window.addEventListener('keydown', handleKey);
   335→    return () => window.removeEventListener('keydown', handleKey);
   336→  }, [currentSong, isPlaying, pauseSong, resumeSong]);
   337→
   338→  /* ── Genre card click ── */
   339→  const handleGenreClick = useCallback((genreQuery: string) => {
   340→    setQuery(genreQuery);
   341→  }, []);
   342→
   343→  /* ── Render ── */
   344→  return (
   345→    <div className="min-h-screen">
   346→      {/* Hidden audio element */}
   347→      <audio ref={audioRef} preload="auto" />
   348→
   349→      {/* Mobile back button */}
   350→      <button
   351→        onClick={goHome}
   352→        className="md:hidden fixed z-[90] flex items-center gap-1.5 text-white/60 active:text-white transition-colors"
   353→        style={{ top: 'max(env(safe-area-inset-top, 0px) + 8px, 8px)', left: 12 }}
   354→        aria-label="Go home"
   355→      >
   356→        <Home className="w-5 h-5" />
   357→      </button>
   358→
   359→      {/* ═══ Header ═══ */}
   360→      <div className="pt-16 md:pt-8 pb-4 px-4 md:px-8">
   361→        {/* Desktop back button */}
   362→        <button
   363→          onClick={goHome}
   364→          className="hidden md:flex items-center gap-2 text-white/70 hover:text-white mb-4 transition-colors"
   365→        >
   366→          <Home className="w-4 h-4" />
   367→          <span className="text-sm">Back to Home</span>
   368→        </button>
   369→
   370→        {/* Title section */}
   371→        <div className="flex items-center gap-3 mb-6">
   372→          <div className="w-12 h-12 rounded-2xl bg-gradient-to-br from-purple-500 to-pink-500 flex items-center justify-center shadow-lg shadow-purple-500/25">
   373→            <Music className="w-6 h-6 text-white" />
   374→          </div>
   375→          <div>
   376→            <h1 className="text-2xl md:text-3xl font-bold text-white">Music</h1>
   377→            <p className="text-white/50 text-sm">Discover Hindi, Bengali & regional songs</p>
   378→          </div>
   379→        </div>
   380→
   381→        {/* ═══ Search bar ═══ */}
   382→        <div className="relative mb-4">
   383→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-white/40" />
   384→          <Input
   385→            value={query}
   386→            onChange={(e) => setQuery(e.target.value)}
   387→            placeholder="Search songs, artists, albums..."
   388→            className="pl-10 pr-10 h-11 bg-white/[0.07] border-white/10 text-white placeholder:text-white/30 rounded-xl focus-visible:border-purple-500/50 focus-visible:ring-purple-500/20"
   389→          />
   390→          {query && (
   391→            <button
   392→              onClick={() => setQuery('')}
   393→              className="absolute right-3 top-1/2 -translate-y-1/2 text-white/40 hover:text-white/70 transition-colors"
   394→            >
   395→              <X className="w-4 h-4" />
   396→            </button>
   397→          )}
   398→        </div>
   399→
   400→        {/* ═══ Language filter chips ═══ */}
   401→        <div className="flex gap-2 overflow-x-auto scrollbar-none pb-1">
   402→          {LANGUAGES.map((lang) => {
   403→            const isActive = activeLang === lang.key;
   404→            return (
   405→              <button
   406→                key={lang.key}
   407→                onClick={() => setActiveLang(lang.key)}
   408→                className={`shrink-0 px-4 py-1.5 rounded-full text-sm font-medium transition-all duration-200 border ${
   409→                  isActive
   410→                    ? 'bg-purple-500/20 border-purple-500/40 text-purple-300 shadow-sm shadow-purple-500/10'
   411→                    : 'bg-white/5 border-white/5 text-white/50 hover:text-white/70 hover:bg-white/10'
   412→                }`}
   413→              >
   414→                {lang.label}
   415→              </button>
   416→            );
   417→          })}
   418→        </div>
   419→      </div>
   420→
   421→      {/* ═══ Content area ═══ */}
   422→      <div className="px-4 md:px-8 pb-10">
   423→        <p className="text-white/40 text-center py-8">Music content area</p>

   508→                            <motion.div
   509→                              className="w-[3px] bg-purple-400 rounded-full"
   510→                              animate={{ height: ['70%', '30%', '90%'] }}
   511→                              transition={{ repeat: Infinity, duration: 0.7, ease: 'easeInOut', delay: 0.1 }}

   515→                              animate={{ height: ['50%', '80%', '40%'] }}

   519→                        ) : (
   520→                          <span className={`text-xs ${isActive ? 'text-purple-400' : 'text-white/30 group-hover:text-white/50'}`}>
   521→                            {index + 1}
   522→                          </span>
   523→                        )}
   524→                      </div>
   525→
   526→                      {/* Album art */}
   527→                      <div className="relative w-12 h-12 rounded-lg overflow-hidden shrink-0 bg-white/[0.06]">
   528→                        {song.image ? (
   529→                          <img
   530→                            src={song.image}
   531→                            alt={song.title}
   532→                            className="w-full h-full object-cover"
   533→                            loading="lazy"
   534→                          />
   535→                        ) : (
   536→                          <div className="w-full h-full flex items-center justify-center">
   537→                            <Music className="w-5 h-5 text-white/20" />
   538→                          </div>
   539→                        )}
   540→                        {/* Play overlay */}
   541→                        <div
   542→                          className={`absolute inset-0 flex items-center justify-center bg-black/40 transition-opacity ${
   543→                            isActive && isPlaying ? 'opacity-0' : 'opacity-0 group-hover:opacity-100'
   544→                          }`}
   545→                        >
   546→                          <Play className="w-5 h-5 text-white" fill="white" />
   547→                        </div>
   548→                      </div>
   549→
   550→                      {/* Song info */}
   551→                      <div className="flex-1 min-w-0">
   552→                        <p className={`text-sm font-medium truncate ${isActive ? 'text-purple-300' : 'text-white/90'}`}>
   553→                          {song.title}
   554→                        </p>
   555→                        <p className="text-xs text-white/40 truncate mt-0.5">
   556→                          {song.artists}
   557→                          {song.album && <span className="text-white/20"> · {song.album}</span>}
   558→                        </p>
   559→                      </div>
   560→
   561→                      {/* Language badge */}
   562→                      <Badge
   563→                        className={`hidden sm:inline-flex shrink-0 text-[10px] px-1.5 py-0 border ${
   564→                          LANG_BADGE_COLORS[song.language] || 'bg-white/10 text-white/50 border-white/10'
   565→                        }`}
   566→                      >
   567→                        {song.language}
   568→                      </Badge>
   569→
   570→                      {/* Play count */}
   571→                      <div className="hidden md:flex items-center gap-1 text-white/30 shrink-0">
   572→                        <Volume2 className="w-3 h-3" />
   573→                        <span className="text-xs">{formatPlayCount(song.playCount)}</span>
   574→                      </div>
   575→
   576→                      {/* Duration */}
   577→                      <span className="text-xs text-white/30 shrink-0 tabular-nums">
   578→                        {formatDuration(song.duration)}
   579→                      </span>
   580→                    </motion.div>
   581→                  );
   582→                })}
   583→              </AnimatePresence>
   584→            </motion.div>
   585→
   586→            {/* Load more */}
   587→            {filteredSongs.length < total && debouncedQuery && (
   588→              <div className="mt-6 flex justify-center">
   589→                <button
   590→                  onClick={loadMore}
   591→                  disabled={loading}
   592→                  className="flex items-center gap-2 px-6 py-2.5 rounded-full bg-white/[0.07] hover:bg-white/[0.12] text-white/70 hover:text-white text-sm font-medium transition-colors border border-white/10 disabled:opacity-50"
   593→                >
   594→                  {loading ? (
   595→                    <Loader2 className="w-4 h-4 animate-spin" />
   596→                  ) : null}
   597→                  {loading ? 'Loading...' : 'Load More'}
   598→                </button>
   599→              </div>
   600→            )}
   601→          </>
   602→        )}
   603→
   604→        {/* ── Loading more indicator (at bottom) ── */}
   605→        {loading && songs.length > 0 && (
   606→          <div className="mt-4 flex justify-center">
   607→            <Loader2 className="w-5 h-5 text-purple-400 animate-spin" />
   608→          </div>
   609→        )}
   610→      </div>
   611→
   612→      {/* ═══ Bottom Audio Player Bar ═══ */}
   613→      <AnimatePresence>
   614→        {currentSong && (
   615→          <motion.div
   616→            initial={{ y: 100, opacity: 0 }}
   617→            animate={{ y: 0, opacity: 1 }}
   618→            exit={{ y: 100, opacity: 0 }}
   619→            transition={{ type: 'spring', damping: 30, stiffness: 300 }}
   620→            className="fixed bottom-[calc(3.5rem+env(safe-area-inset-bottom,0px))] md:bottom-0 left-0 right-0 z-50"
   621→          >
   622→            {/* Backdrop gradient */}
   623→            <div className="absolute inset-0 bg-gradient-to-t from-black via-black/95 to-black/80 backdrop-blur-xl" />
   624→            {/* Top accent line */}
   625→            <div className="absolute top-0 left-0 right-0 h-[2px] bg-gradient-to-r from-purple-500 via-pink-500 to-purple-500" />
   626→
   627→            <div className="relative max-w-5xl mx-auto px-4 py-3 md:py-3.5">
   628→              {/* Progress bar (thin line at top) */}
   629→              <div
   630→                className="absolute top-0 left-0 right-0 h-1 bg-white/10 cursor-pointer group"
   631→                onClick={handleSeek}
   632→              >
   633→                <motion.div
   634→                  className="h-full bg-gradient-to-r from-purple-500 to-pink-500 relative"
   635→                  style={{ width: `${progress}%` }}
   636→                  transition={{ duration: 0.1 }}
   637→                >
   638→                  <div className="absolute right-0 top-1/2 -translate-y-1/2 w-3 h-3 rounded-full bg-white shadow-md opacity-0 group-hover:opacity-100 transition-opacity" />
   639→                </motion.div>
   640→              </div>
   641→
   642→              <div className="flex items-center gap-3 md:gap-4">
   643→                {/* Song art */}
   644→                <div className="w-11 h-11 md:w-12 md:h-12 rounded-lg overflow-hidden shrink-0 bg-white/[0.06] shadow-lg">
   645→                  {currentSong.image ? (
   646→                    <img
   647→                      src={currentSong.image}
   648→                      alt={currentSong.title}
   649→                      className={`w-full h-full object-cover ${isPlaying ? 'animate-[spin_8s_linear_infinite]' : ''}`}
   650→                      style={{ animationPlayState: isPlaying ? 'running' : 'paused' }}
   651→                    />
   652→                  ) : (
   653→                    <div className="w-full h-full flex items-center justify-center">
   654→                      <Music className="w-5 h-5 text-white/20" />
   655→                    </div>
   656→                  )}
   657→                </div>
   658→
   659→                {/* Song info */}
   660→                <div className="flex-1 min-w-0">
   661→                  <p className="text-sm font-medium text-white truncate">{currentSong.title}</p>
   662→                  <p className="text-xs text-white/50 truncate">{currentSong.artists}</p>
   663→                </div>
   664→
   665→                {/* Time display (desktop) */}
   666→                <div className="hidden md:flex items-center gap-2 text-xs text-white/40 tabular-nums">
   667→                  <span>{formatDuration(Math.floor(currentTime))}</span>
   668→                  <span>/</span>
   669→                  <span>{formatDuration(currentSong.duration)}</span>
   670→                </div>
   671→
   672→                {/* Controls */}
   673→                <div className="flex items-center gap-1 md:gap-2">
   674→                  {/* Prev song */}
   675→                  <button
   676→                    onClick={(e) => {
   677→                      e.stopPropagation();
   678→                      const idx = filteredSongs.findIndex((s) => s.id === currentSong.id);
   679→                      if (idx > 0) playSong(filteredSongs[idx - 1]);
   680→                    }}
   681→                    className="hidden md:flex p-2 rounded-full text-white/60 hover:text-white hover:bg-white/10 transition-colors"
   682→                  >
   683→                    <SkipBack className="w-4 h-4" />
   684→                  </button>
   685→
   686→                  {/* Play / Pause */}
   687→                  <button
   688→                    onClick={(e) => {
   689→                      e.stopPropagation();
   690→                      if (isPlaying) pauseSong(); else resumeSong();
   691→                    }}
   692→                    className="p-2.5 rounded-full bg-white/10 hover:bg-white/20 text-white transition-colors"
   693→                  >
   694→                    {isPlaying ? (
   695→                      <Pause className="w-5 h-5" fill="white" />
   696→                    ) : (
   697→                      <Play className="w-5 h-5" fill="white" />
   698→                    )}
   699→                  </button>
   700→
   701→                  {/* Next song */}
   702→                  <button
   703→                    onClick={(e) => {
   704→                      e.stopPropagation();
   705→                      const idx = filteredSongs.findIndex((s) => s.id === currentSong.id);
   706→                      if (idx < filteredSongs.length - 1) playSong(filteredSongs[idx + 1]);
   707→                    }}
   708→                    className="hidden md:flex p-2 rounded-full text-white/60 hover:text-white hover:bg-white/10 transition-colors"
   709→                  >
   710→                    <SkipForward className="w-4 h-4" />
   711→                  </button>
   712→
   713→                  {/* Close */}
   714→                  <button
   715→                    onClick={(e) => {
   716→                      e.stopPropagation();
   717→                      closePlayer();
   718→                    }}
   719→                    className="p-1.5 rounded-full text-white/40 hover:text-white/70 hover:bg-white/10 transition-colors"
   720→                  >
   721→                    <X className="w-4 h-4" />
   722→                  </button>
   723→                </div>
   724→              </div>
   725→            </div>
   726→          </motion.div>
   727→        )}
   728→      </AnimatePresence>
   729→    </div>
