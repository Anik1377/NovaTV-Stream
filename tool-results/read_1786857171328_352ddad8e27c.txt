     1→'use client';
     2→
     3→import { useState, useEffect, useCallback, useMemo } from 'react';
     4→import {
     5→  ArrowLeft, User, Bookmark, Clock, LogOut, Pencil, Check, X, Trash2,
     6→  Loader2, Film, Tv, Users, Calendar, Shield, Palette, Heart, Sparkles,
     7→  ChevronDown,
     8→} from 'lucide-react';
     9→import { motion, AnimatePresence } from 'framer-motion';
    10→import { useAuthStore } from '@/store/auth-store';
    11→import { useAppStore } from '@/store/app-store';
    12→import { Input } from '@/components/ui/input';
    13→import { AuthModal } from '@/components/auth/AuthModal';
    14→import {
    15→  ProfileAvatar, AVATAR_DEFS, ACCENT_COLORS, GENRE_OPTIONS,
    16→  getAvatarDef, type AvatarDef,
    17→} from '@/lib/avatars';
    18→import { getImageUrl } from '@/lib/tmdb';
    19→import type { Movie } from '@/lib/types';
    20→
    21→interface BrowseHistoryItem {
    22→  id: string;
    23→  tmdbId: number;
    24→  title: string;
    25→  posterPath: string | null;
    26→  mediaType: 'movie' | 'tv' | 'person';
    27→  subtitle: string | null;
    28→  visitedAt: string;
    29→}
    30→
    31→type HistoryFilter = 'all' | 'movie' | 'tv' | 'person';
    32→
    33→type Tab = 'profile' | 'watchlist' | 'history';
    34→
    35→type EditStep = 'main' | 'avatar' | 'color' | 'genres';
    36→
    37→export function ProfilePage() {
    38→  const { user, loading: authLoading, updateProfile, logout } = useAuthStore();
    39→  const { goHome, watchlist, selectMovie, selectTv, selectPerson } = useAppStore();
    40→
    41→  const [tab, setTab] = useState<Tab>('profile');
    42→  const [authModalOpen, setAuthModalOpen] = useState(false);
    43→
    44→  // Profile editing
    45→  const [editing, setEditing] = useState(false);
    46→  const [editStep, setEditStep] = useState<EditStep>('main');
    47→  const [editName, setEditName] = useState('');
    48→  const [editBio, setEditBio] = useState('');
    49→  const [selectedAvatar, setSelectedAvatar] = useState('hero');
    50→  const [selectedColor, setSelectedColor] = useState('#e11d48');
    51→  const [selectedGenres, setSelectedGenres] = useState<string[]>([]);
    52→  const [saving, setSaving] = useState(false);
    53→  const [saveError, setSaveError] = useState('');
    54→
    55→  // Browse history
    56→  const [history, setHistory] = useState<BrowseHistoryItem[]>([]);
    57→  const [historyLoading, setHistoryLoading] = useState(false);
    58→  const [historyFilter, setHistoryFilter] = useState<HistoryFilter>('all');
    59→  const [historyPage, setHistoryPage] = useState(1);
    60→  const [historyTotal, setHistoryTotal] = useState(0);
    61→  const [historyLoadMore, setHistoryLoadMore] = useState(false);
    62→  const [deletingId, setDeletingId] = useState<string | null>(null);
    63→
    64→  // Profile stats from API
    65→  const [profileStats, setProfileStats] = useState<{ watchHistoryCount: number } | null>(null);
    66→
    67→  // Computed accent color for UI
    68→  const accentColor = useMemo(() => {
    69→    if (user?.accentColor) return user.accentColor;
    70→    const def = getAvatarDef(user?.avatar);
    71→    return def.color;
    72→  }, [user?.avatar, user?.accentColor]);
    73→
    74→  const fetchProfile = useCallback(async () => {
    75→    if (!user) return;
    76→    try {
    77→      const res = await fetch('/api/profile');
    78→      const data = await res.json();
    79→      if (res.ok) setProfileStats(data.stats);
    80→    } catch { /* ignore */ }
    81→  }, [user]);
    82→
    83→  const fetchHistory = useCallback(async (filter: HistoryFilter, page: number, append: boolean) => {
    84→    if (!user) return;
    85→    if (append) { setHistoryLoadMore(true); } else { setHistoryLoading(true); }
    86→    try {
    87→      const params = new URLSearchParams({ limit: '50', offset: String((page - 1) * 50) });
    88→      if (filter !== 'all') params.set('type', filter);
    89→      const res = await fetch(`/api/history?${params}`);
    90→      const data = await res.json();
    91→      if (res.ok) {
    92→        setHistory(prev => append ? [...prev, ...data.items] : data.items);
    93→        setHistoryTotal(data.total);
    94→        if (!append) setHistoryPage(1);
    95→      }
    96→    } catch { /* ignore */ }
    97→    setHistoryLoading(false);
    98→    setHistoryLoadMore(false);
    99→  }, [user]);
   100→
   101→  useEffect(() => {
   102→    if (tab === 'profile') fetchProfile();
   103→    if (tab === 'history') fetchHistory(historyFilter, 1, false);
   104→  }, [tab, historyFilter]);
   105→
   106→  useEffect(() => {
   107→    window.scrollTo({ top: 0, behavior: 'smooth' });
   108→  }, []);
   109→
   110→  const startEdit = () => {
   111→    setEditName(user?.name || '');
   112→    setEditBio(user?.bio || '');
   113→    setSelectedAvatar(user?.avatar || 'hero');
   114→    setSelectedColor(user?.accentColor || getAvatarDef(user?.avatar).color);
   115→    setSelectedGenres(user?.favoriteGenres || []);
   116→    setEditStep('main');
   117→    setEditing(true);
   118→    setSaveError('');
   119→  };
   120→
   121→  const saveProfile = async () => {
   122→    setSaving(true);
   123→    setSaveError('');
   124→    const result = await updateProfile({
   125→      name: editName,
   126→      bio: editBio,
   127→      avatar: selectedAvatar,
   128→      accentColor: selectedColor,
   129→      favoriteGenres: selectedGenres,
   130→    });
   131→    setSaving(false);
   132→    if (result.error) {
   133→      setSaveError(result.error);
   134→    } else {
   135→      setEditing(false);
   136→    }
   137→  };
   138→
   139→  const toggleGenre = (genre: string) => {
   140→    setSelectedGenres(prev =>
   141→      prev.includes(genre) ? prev.filter(g => g !== genre) : [...prev, genre]
   142→    );
   143→  };
   144→
   145→  const clearHistory = async () => {
   146→    if (!confirm('Clear all browsing history? This cannot be undone.')) return;
   147→    await fetch('/api/history', { method: 'DELETE' });
   148→    setHistory([]);
   149→    setHistoryTotal(0);
   150→  };
   151→
   152→  const handleDeleteHistoryItem = async (id: string, e: React.MouseEvent) => {
   153→    e.stopPropagation();
   154→    setDeletingId(id);
   155→    await fetch(`/api/history?id=${id}`, { method: 'DELETE' });
   156→    setHistory(prev => prev.filter(item => item.id !== id));
   157→    setHistoryTotal(prev => prev - 1);
   158→    setDeletingId(null);
   159→  };
   160→
   161→  const handleHistoryClick = (item: BrowseHistoryItem) => {
   162→    if (item.mediaType === 'person') {
   163→      selectPerson({ id: item.tmdbId, name: item.title, profilePath: item.posterPath });
   164→      return;
   165→    }
   166→    const media: Movie = {
   167→      id: item.tmdbId, title: item.title, name: item.title,
   168→      poster_path: item.posterPath, media_type: item.mediaType,
   169→      vote_average: 0, genre_ids: [], overview: '', popularity: 0,
   170→      release_date: '', first_air_date: '', backdrop_path: null, original_language: '',
   171→    };
   172→    if (item.mediaType === 'tv') selectTv(media); else selectMovie(media);
   173→  };
   174→
   175→  const handleHistoryFilterChange = (f: HistoryFilter) => {
   176→    if (f === historyFilter) return;
   177→    setHistoryFilter(f);
   178→  };
   179→
   180→  const loadMoreHistory = () => {
   181→    const nextPage = historyPage + 1;
   182→    setHistoryPage(nextPage);
   183→    fetchHistory(historyFilter, nextPage, true);
   184→  };
   185→
   186→  const handleWatchlistClick = (tmdbId: number, mediaType: string) => {
   187→    const item: Movie = {
   188→      id: tmdbId, title: '', name: '', poster_path: null,
   189→      media_type: mediaType as 'movie' | 'tv', vote_average: 0, genre_ids: [],
   190→      overview: '', popularity: 0, release_date: '', first_air_date: '',
   191→      backdrop_path: null, original_language: '',
   192→    };
   193→    if (mediaType === 'tv') selectTv(item); else selectMovie(item);
   194→  };
   195→
   196→  // Not logged in
   197→  if (!authLoading && !user) {
   198→    return (
   199→      <div className="min-h-screen flex items-center justify-center">
   200→        <div className="text-center max-w-sm">
   201→          <div className="w-24 h-24 rounded-full bg-white/[0.06] flex items-center justify-center mx-auto mb-6">
   202→            <User className="w-10 h-10 text-white/20" />
   203→          </div>
   204→          <h2 className="text-2xl font-bold text-white mb-2">Sign in to your Profile</h2>
   205→          <p className="text-white/40 text-sm mb-6">Track your watchlist and history across devices.</p>
   206→          <button onClick={() => setAuthModalOpen(true)} className="px-6 py-3 hover:brightness-110 text-white font-semibold rounded-xl transition-all shadow-lg" style={{ backgroundColor: accentColor, boxShadow: `0 10px 30px ${accentColor}33` }}>
   207→            Sign In
   208→          </button>
   209→          <button onClick={goHome} className="flex items-center gap-2 text-white/40 hover:text-white mx-auto mt-8 transition-colors">
   210→            <ArrowLeft className="w-4 h-4" /><span className="text-sm">Back to Home</span>
   211→          </button>
   212→        </div>
   213→        <AuthModal open={authModalOpen} onClose={() => setAuthModalOpen(false)} />
   214→      </div>
   215→    );
   216→  }
   217→
   218→  const memberSince = user ? new Date(user.createdAt || Date.now()).toLocaleDateString('en-US', { month: 'long', year: 'numeric' }) : '';
   219→  const avatarDef = getAvatarDef(user?.avatar);
   220→
   221→  const tabsList: { key: Tab; label: string; icon: React.ReactNode }[] = [
   222→    { key: 'profile', label: 'Profile', icon: <User className="w-4 h-4" /> },
   223→    { key: 'watchlist', label: `Watchlist (${watchlist.length})`, icon: <Bookmark className="w-4 h-4" /> },
   224→    { key: 'history', label: `History${historyTotal > 0 ? ` (${historyTotal})` : ''}`, icon: <Clock className="w-4 h-4" /> },
   225→  ];
   226→
   227→  return (
   228→    <div className="min-h-screen pb-10">
   229→      {/* Back button */}
   230→      <div className="pt-20 px-4 md:px-8">
   231→        <button onClick={goHome} className="flex items-center gap-2 text-white/50 hover:text-white mb-6 transition-colors group">
   232→          <div className="w-8 h-8 rounded-full bg-white/[0.06] flex items-center justify-center group-hover:bg-white/10 transition-colors">
   233→            <ArrowLeft className="w-4 h-4" />
   234→          </div>
   235→          <span className="text-sm font-medium">Back</span>
   236→        </button>
   237→
   238→        {/* ── Profile Header ── */}
   239→        <div className="flex flex-col sm:flex-row items-center sm:items-start gap-6 mb-8">
   240→          {/* Avatar */}
   241→          <motion.div className="relative" initial={{ scale: 0.8, opacity: 0 }} animate={{ scale: 1, opacity: 1 }} transition={{ type: 'spring', stiffness: 300, damping: 20 }}>
   242→            <div className="w-28 h-28 rounded-2xl overflow-hidden shadow-2xl" style={{ boxShadow: `0 12px 40px ${accentColor}44` }}>
   243→              <ProfileAvatar slug={user?.avatar} size={112} />
   244→            </div>
   245→            {!editing && (
   246→              <button onClick={startEdit} className="absolute -bottom-1 -right-1 w-9 h-9 rounded-xl bg-white/10 backdrop-blur-sm border border-white/20 flex items-center justify-center hover:bg-white/20 transition-colors" style={{ boxShadow: `0 4px 12px ${accentColor}22` }}>
   247→                <Pencil className="w-3.5 h-3.5 text-white" />
   248→              </button>
   249→            )}
   250→            {/* Online indicator */}
   251→            <div className="absolute -bottom-0.5 -left-0.5 w-4 h-4 rounded-full border-2 border-zinc-950" style={{ backgroundColor: '#22c55e' }} />
   252→          </motion.div>
   253→
   254→          {/* Info */}
   255→          <div className="text-center sm:text-left flex-1">
   256→            <h1 className="text-2xl md:text-3xl font-bold text-white">{user?.name || 'StreamVault User'}</h1>
   257→            <p className="text-white/40 text-sm mt-1">{user?.email}</p>
   258→            {user?.bio && !editing && <p className="text-white/60 text-sm mt-2 max-w-md">{user.bio}</p>}
   259→            <div className="flex flex-wrap items-center justify-center sm:justify-start gap-3 mt-3 text-xs text-white/30">
   260→              <span className="flex items-center gap-1.5"><Calendar className="w-3.5 h-3.5" />Joined {memberSince}</span>
   261→              <span className="flex items-center gap-1.5"><Bookmark className="w-3.5 h-3.5" />{watchlist.length} saved</span>
   262→              {profileStats && <span className="flex items-center gap-1.5"><Clock className="w-3.5 h-3.5" />{profileStats.watchHistoryCount} watched</span>}
   263→            </div>
   264→            {/* Favorite genres badges */}
   265→            {user?.favoriteGenres && user.favoriteGenres.length > 0 && !editing && (
   266→              <div className="flex flex-wrap gap-1.5 mt-3 justify-center sm:justify-start">
   267→                {user.favoriteGenres.slice(0, 5).map(g => (
   268→                  <span key={g} className="px-2.5 py-1 rounded-lg text-[10px] font-semibold uppercase tracking-wider text-white/80" style={{ backgroundColor: `${accentColor}22`, border: `1px solid ${accentColor}33` }}>{g}</span>
   269→                ))}
   270→                {user.favoriteGenres.length > 5 && (
   271→                  <span className="px-2.5 py-1 rounded-lg text-[10px] font-semibold text-white/40 bg-white/[0.06]">+{user.favoriteGenres.length - 5}</span>
   272→                )}
   273→              </div>
   274→            )}
   275→          </div>
   276→
   277→          {/* Actions */}
   278→          <div className="sm:ml-auto flex gap-2">
   279→            <button onClick={startEdit} className="flex items-center gap-2 px-4 py-2.5 rounded-xl text-white/80 text-sm font-medium transition-all hover:brightness-110" style={{ backgroundColor: `${accentColor}22`, border: `1px solid ${accentColor}33` }}>
   280→              <Pencil className="w-4 h-4" /><span className="hidden sm:inline">Edit Profile</span>
   281→            </button>
   282→            <button onClick={async () => { await logout(); goHome(); }} className="flex items-center gap-2 px-4 py-2 rounded-xl bg-white/[0.06] hover:bg-white/10 text-white/60 hover:text-white text-sm transition-all border border-white/[0.06]">
   283→              <LogOut className="w-4 h-4" /><span className="hidden sm:inline">Sign Out</span>
   284→            </button>
   285→          </div>
   286→        </div>
   287→
   288→        {/* Tabs */}
   289→        <div className="flex gap-1 mb-6 border-b border-white/[0.06]">
   290→          {tabsList.map((t) => (
   291→            <button key={t.key} onClick={() => setTab(t.key)} className={`flex items-center gap-2 px-4 py-3 text-sm font-medium transition-all relative border-b-2 -mb-px ${tab === t.key ? 'text-white' : 'text-white/40 border-transparent hover:text-white/70'}`} style={tab === t.key ? { borderColor: accentColor } : undefined}>
   292→              {t.icon}
   293→              <span className="hidden sm:inline">{t.label}</span>
   294→              <span className="sm:hidden">{t.key === 'watchlist' ? watchlist.length : ''}</span>
   295→            </button>
   296→          ))}
   297→        </div>
   298→
   299→        {/* Tab Content */}
   300→        <AnimatePresence mode="wait">
   301→          {tab === 'profile' && (
   302→            <motion.div key="profile-tab" initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -8 }} className="max-w-2xl">
   303→              {editing ? (
   304→                <EditPanel
   305→                  editStep={editStep} setEditStep={setEditStep}
   306→                  editName={editName} setEditName={setEditName}
   307→                  editBio={editBio} setEditBio={setEditBio}
   308→                  selectedAvatar={selectedAvatar} setSelectedAvatar={setSelectedAvatar}
   309→                  selectedColor={selectedColor} setSelectedColor={setSelectedColor}
   310→                  selectedGenres={selectedGenres} toggleGenre={toggleGenre}
   311→                  accentColor={accentColor}
   312→                  saving={saving} saveError={saveError}
   313→                  onSave={saveProfile} onCancel={() => setEditing(false)}
   314→                />
   315→              ) : (
   316→                <div className="space-y-4">
   317→                  <div className="bg-white/[0.03] rounded-2xl border border-white/[0.06] p-5 space-y-4">
   318→                    <h3 className="text-sm font-semibold text-white/70 flex items-center gap-2"><Shield className="w-4 h-4 text-white/30" />Account Info</h3>
   319→                    <div className="space-y-3">
   320→                      <InfoRow label="Name" value={user?.name || '\u2014'} />
   321→                      <InfoRow label="Email" value={user?.email} />
   322→                      <InfoRow label="Avatar" value={avatarDef.name} />
   323→                      <InfoRow label="Bio" value={user?.bio || '\u2014'} truncate />
   324→                      <InfoRow label="Theme" value={user?.accentColor ? `Custom (${ACCENT_COLORS.find(c => c.value === user.accentColor)?.name || 'Custom'})` : `Avatar-matched (${avatarDef.name})`} />
   325→                      <InfoRow label="Favorite Genres" value={user?.favoriteGenres?.length ? user.favoriteGenres.join(', ') : '\u2014'} />
   326→                      <InfoRow label="Member Since" value={memberSince} />
   327→                    </div>
   328→                  </div>
   329→                </div>
   330→              )}
   331→            </motion.div>
   332→          )}
   333→
   334→          {tab === 'watchlist' && (
   335→            <motion.div key="watchlist-tab" initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -8 }}>
   336→              {watchlist.length === 0 ? (
   337→                <EmptyState icon={<Bookmark className="w-9 h-9 text-white/10" />} title="No items in watchlist" description="Browse movies and shows, and tap the heart icon to save them here." actionLabel="Browse Content" onAction={goHome} />
   338→              ) : (
   339→                <div className="grid grid-cols-3 sm:grid-cols-4 md:grid-cols-5 lg:grid-cols-6 gap-3 md:gap-4">
   340→                  {watchlist.map((id) => (<WatchlistCard key={id} tmdbId={id} onClick={handleWatchlistClick} />))}
   341→                </div>
   342→              )}
   343→            </motion.div>
   344→          )}
   345→
   346→          {tab === 'history' && (
   347→            <motion.div key="history-tab" initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -8 }}>
   348→              {historyLoading ? (
   349→                <div className="flex items-center justify-center py-20"><Loader2 className="w-6 h-6 text-white/30 animate-spin" /></div>
   350→              ) : history.length === 0 ? (
   351→                <EmptyState icon={<Clock className="w-9 h-9 text-white/10" />} title="No watch history" description="Movies and shows you watch will appear here." />
   352→              ) : (
   353→                <div className="space-y-2">
   354→                  <div className="flex items-center justify-between mb-4">
   355→                    <p className="text-sm text-white/40">{history.length} items</p>
   356→                    <button onClick={clearHistory} className="flex items-center gap-1.5 text-sm text-white/30 hover:text-red-400 transition-colors"><Trash2 className="w-3.5 h-3.5" />Clear All</button>
   357→                  </div>
   358→                  {history.map((entry) => (
   359→                    <button key={entry.id} onClick={() => handleHistoryClick(entry)} className="w-full flex items-center gap-4 p-3 rounded-xl hover:bg-white/[0.04] transition-colors text-left group">
   360→                      <div className="w-12 h-16 rounded-lg bg-white/[0.06] overflow-hidden shrink-0">
   361→                        {entry.posterPath ? (
   362→                          <img src={`https://image.tmdb.org/t/p/w185${entry.posterPath}`} alt={entry.title} className="w-full h-full object-cover" />
   363→                        ) : (
   364→                          <div className="w-full h-full flex items-center justify-center">{entry.mediaType === 'tv' ? <Tv className="w-5 h-5 text-white/20" /> : <Film className="w-5 h-5 text-white/20" />}</div>
   365→                        )}
   366→                      </div>
   367→                      <div className="flex-1 min-w-0">
   368→                        <p className="text-sm font-medium text-white truncate group-hover:text-red-400 transition-colors">{entry.title}</p>
   369→                        <div className="flex items-center gap-2 mt-1">
   370→                          <span className="text-xs text-white/30 px-1.5 py-0.5 rounded bg-white/[0.06]">{entry.mediaType === 'tv' ? 'TV' : 'Movie'}</span>
   371→                          {entry.season && <span className="text-xs text-white/30">S{String(entry.season).padStart(2, '0')}{entry.episode ? `E${String(entry.episode).padStart(2, '0')}` : ''}</span>}
   372→                        </div>
   373→                      </div>
   374→                      <span className="text-xs text-white/20 shrink-0">{new Date(entry.watchedAt).toLocaleDateString()}</span>
   375→                    </button>
   376→                  ))}
   377→                </div>
   378→              )}
   379→            </motion.div>
   380→          )}
   381→        </AnimatePresence>
   382→      </div>
   383→      <AuthModal open={authModalOpen} onClose={() => setAuthModalOpen(false)} />
   384→    </div>
   385→  );
   386→}

   388→// ── Edit Panel with step navigation ──
   389→function EditPanel({
   390→  editStep, setEditStep, editName, setEditName, editBio, setEditBio,
   391→  selectedAvatar, setSelectedAvatar, selectedColor, setSelectedColor,
   392→  selectedGenres, toggleGenre, accentColor, saving, saveError, onSave, onCancel,
   393→}: {
   394→  editStep: EditStep; setEditStep: (s: EditStep) => void;
   395→  editName: string; setEditName: (v: string) => void;
   396→  editBio: string; setEditBio: (v: string) => void;
   397→  selectedAvatar: string; setSelectedAvatar: (v: string) => void;
   398→  selectedColor: string; setSelectedColor: (v: string) => void;
   399→  selectedGenres: string[]; toggleGenre: (g: string) => void;
   400→  accentColor: string; saving: boolean; saveError: string;
   401→  onSave: () => void; onCancel: () => void;
   402→}) {
   403→  const steps: { key: EditStep; label: string; icon: React.ReactNode }[] = [
   404→    { key: 'main', label: 'Info', icon: <User className="w-3.5 h-3.5" /> },
   405→    { key: 'avatar', label: 'Avatar', icon: <Sparkles className="w-3.5 h-3.5" /> },
   406→    { key: 'color', label: 'Theme', icon: <Palette className="w-3.5 h-3.5" /> },
   407→    { key: 'genres', label: 'Genres', icon: <Heart className="w-3.5 h-3.5" /> },
   408→  ];
   409→
   410→  return (
   411→    <div className="space-y-0 bg-white/[0.03] rounded-2xl border border-white/[0.06] overflow-hidden">
   412→      {/* Step navigation */}
   413→      <div className="flex border-b border-white/[0.06]">
   414→        {steps.map((s) => (

   536→          </button>

   567→function WatchlistCard({ tmdbId, onClick }: { tmdbId: number; onClick: (id: number, type: string) => void }) {
   568→  const [data, setData] = useState<{ title: string; posterPath: string | null; mediaType: string } | null>(null);
   569→
   570→  useEffect(() => {
   571→    Promise.any([
   572→      fetch(`/api/tmdb/preview?id=${tmdbId}&type=movie`).then(r => r.json()),
   573→      fetch(`/api/tmdb/preview?id=${tmdbId}&type=tv`).then(r => r.json()),
   574→    ]).then(d => {
   575→      if (d.title || d.name) setData({ title: d.title || d.name, posterPath: d.poster_path, mediaType: d.media_type || (d.first_air_date ? 'tv' : 'movie') });
   576→    }).catch(() => {});
   577→  }, [tmdbId]);
   578→
   579→  return (
   580→    <button onClick={() => data && onClick(tmdbId, data.mediaType)} className="group text-left">
   581→      <div className="aspect-[2/3] rounded-xl bg-white/[0.06] overflow-hidden mb-2 relative">
   582→        {data?.posterPath ? (
   583→          <img src={`https://image.tmdb.org/t/p/w342${data.posterPath}`} alt={data.title} className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-300" />
   584→        ) : (
   585→          <div className="w-full h-full flex items-center justify-center"><Film className="w-8 h-8 text-white/10" /></div>
   586→        )}
   587→      </div>
   588→      <p className="text-xs text-white/70 font-medium truncate group-hover:text-white transition-colors">{data?.title || 'Loading...'}</p>
   589→    </button>
   590→  );
   591→}
