     1→'use client';
     2→
     3→import { useEffect, useState, useRef, useCallback, useMemo } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import {
     6→  Loader2,
     7→  Search,
     8→  BookOpen,
     9→  ArrowLeft,
    10→  AlertCircle,
    11→  Gamepad2,
    12→  BookText,
    13→  Sparkles,
    14→  Flame,
    15→  TrendingUp,
    16→  Star,
    17→} from 'lucide-react';
    18→import { useAppStore } from '@/store/app-store';
    19→
    20→/* ── Types ── */
    21→interface MangaItem {
    22→  id: string;
    23→  title: string;
    24→  coverUrl: string;
    25→  author?: string;
    26→  tags?: string[];
    27→}
    28→
    29→interface NovelItem {
    30→  id: number;
    31→  title: string;
    32→  author: string;
    33→  genre: string[];
    34→  year: number;
    35→  coverUrl: string;
    36→  description: string;
    37→  downloadCount: number;
    38→}
    39→
    40→/* ── Tab config ── */
    41→type MainTab = 'manga' | 'comics' | 'novels';
    42→
    43→interface SubTab {
    44→  key: string;
    45→  label: string;
    46→  icon?: string;
    47→  flag?: string;
    48→}
    49→
    50→const MANGA_TABS: SubTab[] = [
    51→  { key: 'all', label: 'All' },
    52→  { key: 'manga', label: 'Manga', flag: '\u{1F1EF}\u{1F1F5}' },
    53→  { key: 'manhwa', label: 'Manhwa', flag: '\u{1F1F0}\u{1F1F7}' },
    54→  { key: 'manhua', label: 'Manhua', flag: '\u{1F1E8}\u{1F1F3}' },
    55→];
    56→
    57→const COMICS_TABS: SubTab[] = [
    58→  { key: 'all', label: 'All Comics', icon: 'Sparkles' },
    59→  { key: 'trending', label: 'Trending', icon: 'TrendingUp' },
    60→  { key: 'popular', label: 'Most Popular', icon: 'Flame' },
    61→];
    62→
    63→const NOVEL_GENRES = [
    64→  'All', 'Adventure', 'Classic', 'Comedy', 'Drama', 'Fantasy',
    65→  'Gothic', 'Historical Fiction', 'Horror', 'Mystery',
    66→  'Philosophy', 'Romance', 'Science Fiction', 'Thriller',
    67→];
    68→
    69→/* ── Main tab config ── */
    70→const MAIN_TABS: { key: MainTab; label: string; icon: typeof BookOpen; color: string }[] = [
    71→  { key: 'manga', label: 'Manga', icon: BookOpen, color: 'from-amber-500 to-orange-600' },
    72→  { key: 'comics', label: 'Comics', icon: Gamepad2, color: 'from-violet-500 to-purple-600' },
    73→  { key: 'novels', label: 'Novels', icon: BookText, color: 'from-emerald-500 to-teal-600' },
    74→];
    75→
    76→/* ── Helpers ── */
    77→function proxyCover(url?: string): string {
    78→  if (!url) return '';
    79→  if (url.includes('gutenberg.org')) return `/api/novels/proxy?url=${encodeURIComponent(url)}`;
    80→  return `/api/manga/proxy?url=${encodeURIComponent(url)}`;
    81→}
    82→
    83→/* ── Skeleton ── */
    84→function SkeletonCard() {
    85→  return (
    86→    <div className="space-y-2">
    87→      <div className="aspect-[2/3] bg-white/5 rounded-lg animate-pulse" />
    88→      <div className="h-3.5 bg-white/5 rounded w-3/4 animate-pulse" />
    89→      <div className="h-3 bg-white/5 rounded w-1/2 animate-pulse" />
    90→    </div>
    91→  );
    92→}
    93→
    94→/* ═══════════════════════════════════════════════════ */
    95→export function ReadPage() {
    96→  const { goHome, selectManga, selectNovel } = useAppStore();
    97→  const [mainTab, setMainTab] = useState<MainTab>('manga');
    98→
    99→  // Manga state
   100→  const [mangaList, setMangaList] = useState<MangaItem[]>([]);
   101→  const [mangaSubTab, setMangaSubTab] = useState('all');
   102→  const [mangaLoading, setMangaLoading] = useState(false);
   103→  const [mangaError, setMangaError] = useState(false);
   104→  const [mangaSearch, setMangaSearch] = useState('');
   105→  const [mangaHasMore, setMangaHasMore] = useState(false);
   106→  const [mangaTotal, setMangaTotal] = useState(0);
   107→  const [mangaOffset, setMangaOffset] = useState(0);
   108→  const mangaDebounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   109→  const mangaAbortRef = useRef<AbortController | null>(null);
   110→
   111→  // Comics state
   112→  const [comicList, setComicList] = useState<MangaItem[]>([]);
   113→  const [comicSubTab, setComicSubTab] = useState('trending');
   114→  const [comicLoading, setComicLoading] = useState(false);
   115→  const [comicError, setComicError] = useState(false);
   116→  const [comicTotal, setComicTotal] = useState(0);
   117→
   118→  // Novels state
   119→  const [novelList, setNovelList] = useState<NovelItem[]>([]);
   120→  const [novelGenre, setNovelGenre] = useState('All');
   121→  const [novelSearch, setNovelSearch] = useState('');
   122→  const [novelLoading, setNovelLoading] = useState(true);
   123→  const novelDebounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   124→
   125→  /* ── Manga fetch ── */
   126→  const fetchManga = useCallback(async (tab: string, search: string, offset: number, append = false) => {
   127→    mangaAbortRef.current?.abort();
   128→    const controller = new AbortController();
   129→    mangaAbortRef.current = controller;
   130→    if (!append) { setMangaLoading(true); setMangaError(false); }
   131→
   132→    try {
   133→      const params = new URLSearchParams();
   134→      if (search.trim()) {
   135→        params.set('q', search.trim());
   136→        if (offset > 0) params.set('page', String(Math.floor(offset / 20) + 1));
   137→        const res = await fetch(`/api/manga/search?${params}`, { signal: controller.signal });
   138→        if (!res.ok) throw new Error();
   139→        const data = await res.json();
   140→        if (controller.signal.aborted) return;
   141→        const items: MangaItem[] = (data.results || []).map((m: any) => ({
   142→          id: String(m.id), title: String(m.title || 'Untitled'),
   143→          coverUrl: String(m.coverUrl || ''), author: String(m.author || ''),
   144→          tags: Array.isArray(m.tags) ? m.tags.filter(Boolean).map(String) : [],
   145→        }));
   146→        setMangaList(append ? (prev) => [...prev, ...items] : items);
   147→        setMangaHasMore(!!data.hasMore);
   148→        setMangaTotal(data.total || 0);
   149→      } else {
   150→        if (tab === 'manga' || tab === 'manhwa' || tab === 'manhua') params.set('type', tab);
   151→        if (offset > 0) params.set('offset', String(offset));
   152→        const url = `/api/manga/trending${params.toString() ? `?${params}` : ''}`;
   153→        const res = await fetch(url, { signal: controller.signal });
   154→        if (!res.ok) throw new Error();
   155→        const data = await res.json();
   156→        if (controller.signal.aborted) return;
   157→        const items: MangaItem[] = (data.results || []).map((m: any) => ({
   158→          id: String(m.id), title: String(m.title || 'Untitled'),
   159→          coverUrl: String(m.coverUrl || ''), author: String(m.author || ''),
   160→          tags: Array.isArray(m.tags) ? m.tags.filter(Boolean).map(String) : [],
   161→        }));
   162→        setMangaList(append ? (prev) => [...prev, ...items] : items);
   163→        setMangaHasMore(!!data.hasMore);
   164→        setMangaTotal(data.total || 0);
   165→      }
   166→    } catch {
   167→      if (!append) setMangaError(true);
   168→    } finally {
   169→      if (!controller.signal.aborted) setMangaLoading(false);
   170→    }
   171→  }, []);
   172→
   173→  useEffect(() => {
   174→    setMangaOffset(0); setMangaSearch('');
   175→    fetchManga(mangaSubTab, '', 0);
   176→    return () => { mangaAbortRef.current?.abort(); };
   177→  }, [mangaSubTab, fetchManga]);
   178→
   179→  const handleMangaSearch = useCallback((value: string) => {
   180→    setMangaSearch(value);
   181→    if (mangaDebounceRef.current) clearTimeout(mangaDebounceRef.current);
   182→    if (!value.trim()) { setMangaOffset(0); fetchManga(mangaSubTab, '', 0); return; }
   183→    mangaDebounceRef.current = setTimeout(() => { setMangaOffset(0); fetchManga(mangaSubTab, value, 0); }, 300);
   184→  }, [mangaSubTab, fetchManga]);
   185→
   186→  /* ── Comics fetch ── */
   187→  const fetchComics = useCallback(async () => {
   188→    setComicLoading(true); setComicError(false);
   189→    try {
   190→      const params = new URLSearchParams();
   191→      if (comicSubTab === 'popular') params.set('order', 'popular');
   192→      params.set('limit', '20');
   193→      const res = await fetch(`/api/comics/trending?${params}`);
   194→      if (!res.ok) throw new Error();
   195→      const data = await res.json();
   196→      const items: MangaItem[] = (data.results || []).map((m: any) => ({
   197→        id: String(m.id), title: String(m.title || 'Untitled'),
   198→        coverUrl: String(m.coverUrl || ''), author: String(m.author || ''),
   199→        tags: Array.isArray(m.tags) ? m.tags.filter(Boolean).map(String) : [],
   200→      }));
   201→      setComicList(items);
   202→      setComicTotal(data.total || 0);
   203→    } catch { setComicError(true); }
   204→    finally { setComicLoading(false); }
   205→  }, [comicSubTab]);
   206→
   207→  useEffect(() => { fetchComics(); }, [fetchComics]);
   208→
   209→  /* ── Novels fetch ── */
   210→  useEffect(() => {
   211→    setNovelLoading(true);
   212→    fetch('/novels-data.json')
   213→      .then((r) => r.json())
   214→      .then((data: NovelItem[]) => setNovelList(data))
   215→      .catch(() => {})
   216→      .finally(() => setNovelLoading(false));
   217→  }, []);
   218→
   219→  const filteredNovels = useMemo(() => {
   220→    let list = novelList;
   221→    if (novelGenre !== 'All') list = list.filter((n) => n.genre.includes(novelGenre));
   222→    if (novelSearch.trim()) {
   223→      const q = novelSearch.toLowerCase();
   224→      list = list.filter(
   225→        (n) =>
   226→          n.title.toLowerCase().includes(q) ||
   227→          n.author.toLowerCase().includes(q) ||
   228→          n.description.toLowerCase().includes(q)
   229→      );
   230→    }
   231→    return list;
   232→  }, [novelList, novelGenre, novelSearch]);
   233→
   234→  const handleNovelSearch = useCallback((value: string) => {
   235→    setNovelSearch(value);
   236→    if (novelDebounceRef.current) clearTimeout(novelDebounceRef.current);
   237→  }, []);
   238→
   239→  /* ── Shared search input ── */
   240→  const searchPlaceholder = mainTab === 'manga'
   241→    ? 'Search manga, manhwa, manhua...'
   242→    : mainTab === 'comics'
   243→    ? 'Browse comics'
   244→    : 'Search classic novels...';
   245→
   246→  const currentSearch = mainTab === 'manga' ? mangaSearch : novelSearch;
   247→  const handleSearch = mainTab === 'manga' ? handleMangaSearch : handleNovelSearch;
   248→
   249→  /* ── Active color ── */
   250→  const activeColor = mainTab === 'manga' ? 'amber' : mainTab === 'comics' ? 'violet' : 'emerald';
   251→
   252→  /* ── Render sub-tabs ── */
   253→  const subTabs: SubTab[] =
   254→    mainTab === 'manga' ? MANGA_TABS : mainTab === 'comics' ? COMICS_TABS : [];
   255→  const activeSubTab = mainTab === 'manga' ? mangaSubTab : mainTab === 'comics' ? comicSubTab : '';
   256→  const setActiveSubTab = mainTab === 'manga' ? setMangaSubTab : setComicSubTab;
   257→
   258→  /* ── Item count ── */
   259→  const itemCount =
   260→    mainTab === 'manga' ? mangaTotal :
   261→    mainTab === 'comics' ? comicTotal :
   262→    filteredNovels.length;
   263→
   264→  return (
   265→    <div className="min-h-screen pb-10">
   266→      {/* Mobile back button */}
   267→      <button
   268→        onClick={goHome}
   269→        className="md:hidden fixed z-[90] flex items-center gap-1.5 text-white/60 active:text-white transition-colors"
   270→        style={{ top: 'max(env(safe-area-inset-top, 0px) + 8px, 8px)', left: 12 }}
   271→        aria-label="Go home"
   272→      >
   273→        <ArrowLeft className="w-5 h-5" />
   274→      </button>
   275→
   276→      <div className="pt-16 md:pt-8 pb-6 px-4 md:px-8">
   277→        {/* Desktop back */}
   278→        <button
   279→          onClick={goHome}
   280→          className="hidden md:flex items-center gap-2 text-white/70 hover:text-white mb-4 transition-colors"
   281→        >
   282→          <ArrowLeft className="w-4 h-4" />
   283→          <span className="text-sm">Back to Home</span>
   284→        </button>
   285→
   286→        {/* ── Hero header ── */}
   287→        <div className="relative mb-6">
   288→          <div className={`relative h-[20vh] min-h-[160px] md:min-h-[200px] rounded-2xl overflow-hidden`}>
   289→            <div className={`absolute inset-0 bg-gradient-to-br ${MAIN_TABS.find(t => t.key === mainTab)?.color || 'from-amber-500 to-orange-600'}`} />
   290→            <div className="absolute inset-0 opacity-20" style={{
   291→              backgroundImage: 'radial-gradient(circle at 20% 80%, rgba(255,255,255,0.15) 0%, transparent 50%), radial-gradient(circle at 80% 20%, rgba(255,255,255,0.1) 0%, transparent 50%)',
   292→            }} />
   293→            <div className="absolute inset-0 bg-gradient-to-t from-black/60 via-transparent to-black/20" />
   294→            <div className="relative h-full flex items-center justify-center text-center px-4">
   295→              <div>
   296→                <div className={`w-14 h-14 rounded-2xl bg-gradient-to-br ${MAIN_TABS.find(t => t.key === mainTab)?.color || ''} flex items-center justify-center shadow-lg mb-3`}>
   297→                  <BookOpen className="w-7 h-7 text-white" />
   298→                </div>
   299→                <h1 className="text-3xl md:text-4xl font-black text-white mb-1">
   300→                  Read
   301→                </h1>
   302→                <p className="text-white/50 text-xs md:text-sm max-w-md mx-auto">
   303→                  {mainTab === 'manga' && 'Manga, Manhwa & Manhua'}
   304→                  {mainTab === 'comics' && 'Webtoons, Manhwa & Comics'}
   305→                  {mainTab === 'novels' && 'Classic Novels & Literature'}
   306→                  &middot; {itemCount > 0 && !mangaLoading && !comicLoading && !novelLoading ? `${itemCount.toLocaleString()} titles` : 'Free to read'}
   307→                </p>
   308→              </div>
   309→            </div>
   310→          </div>
   311→        </div>
   312→
   313→        {/* ── Main tabs ── */}
   314→        <div className="flex gap-1 mb-5">
   315→          {MAIN_TABS.map((tab) => {
   316→            const IconComp = tab.icon;
   317→            return (
   318→              <button
   319→                key={tab.key}
   320→                onClick={() => setMainTab(tab.key)}
   321→                className={`shrink-0 px-4 py-2.5 rounded-xl text-sm font-semibold transition-all relative ${
   322→                  mainTab === tab.key
   323→                    ? `bg-${activeColor}-500/15 text-${activeColor}-400 border border-${activeColor}-500/30`
   324→                    : 'bg-white/5 text-white/50 hover:bg-white/10 hover:text-white/70 border border-white/[0.08]'
   325→                }`}
   326→              >
   327→                <IconComp className="w-4 h-4" />
   328→                <span className="ml-1.5">{tab.label}</span>
   329→                {mainTab === tab.key && (
   330→                  <motion.div
   331→                    layoutId="read-main-tab"
   332→                    className="absolute inset-0 rounded-xl bg-${activeColor}-500/15 border border-${activeColor}-500/30"
   333→                    transition={{ type: 'spring', stiffness: 400, damping: 30 }}
   334→                  />
   335→                )}
   336→              </button>
   337→            );
   338→          })}
   339→        </div>
   340→
   341→        {/* ── Search ── */}
   342→        {mainTab !== 'comics' && (
   343→          <div className="relative mb-5">
   344→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-white/40" />
   345→            <input
   346→              type="text"
   347→              placeholder={searchPlaceholder}
   348→              value={currentSearch}
   349→              onChange={(e) => handleSearch(e.target.value)}
   350→              className="w-full bg-white/5 border border-white/10 rounded-xl pl-10 pr-4 py-2.5 text-sm text-white placeholder:text-white/50 focus:outline-none focus:border-amber-500/50 focus:ring-1 focus:ring-amber-500/30 transition-colors"
   351→            />
   352→          </div>
   353→        )}
   354→
   355→        {/* ── Sub-tabs / filters ── */}
   356→        {subTabs.length > 0 && (
   357→          <div className="flex gap-1 overflow-x-auto scrollbar-none pb-3 -mx-1 px-1">
   358→            {subTabs.map((tab) => (
   359→              <button
   360→                key={tab.key}
   361→                onClick={() => setActiveSubTab(tab.key)}
   362→                className={`shrink-0 px-3 py-1.5 rounded-lg text-xs font-medium transition-colors relative whitespace-nowrap ${
   363→                  activeSubTab === tab.key
   364→                    ? `text-${activeColor}-400`
   365→                    : 'text-white/40 hover:text-white/60'
   366→                }`}
   367→              >
   368→                {tab.flag && <span className="mr-1">{tab.flag}</span>}
   369→                {tab.label}
   370→                {activeSubTab === tab.key && (
   371→                  <motion.div
   372→                    layoutId="read-sub-tab"
   373→                    className={`absolute bottom-0 left-0 right-0 h-0.5 bg-${activeColor}-500 rounded-full`}
   374→                    transition={{ type: 'spring', stiffness: 400, damping: 30 }}
   375→                  />
   376→                )}
   377→              </button>
   378→            ))}
   379→          </div>
   380→        )}
   381→
   382→        {/* ── Novel genre pills ── */}
   383→        {mainTab === 'novels' && (
   384→          <div className="flex gap-1.5 overflow-x-auto scrollbar-none pb-3 -mx-1 px-1">
   385→            {NOVEL_GENRES.map((genre) => (
   386→              <button
   387→                key={genre}
   388→                onClick={() => setNovelGenre(genre)}
   389→                className={`shrink-0 px-3 py-1.5 rounded-full text-[11px] font-semibold transition-all border ${
   390→                  novelGenre === genre
   391→                    ? `bg-emerald-500/15 text-emerald-300 border-emerald-500/30`
   392→                    : 'bg-white/[0.04] text-white/40 border-white/[0.08] hover:bg-white/[0.08] hover:text-white/70'
   393→                }`}
   394→              >
   395→                {genre}
   396→              </button>
   397→            ))}
   398→          </div>
   399→        )}
   400→      </div>
   401→
   402→      {/* ═══ Content area ═══ */}
   403→      <div className="px-4 md:px-8">
   404→        {/* ── MANGA ── */}
   405→        {mainTab === 'manga' && (
   406→          <>
   407→            {mangaLoading && (
   408→              <div className="grid grid-cols-3 sm:grid-cols-4 md:grid-cols-5 lg:grid-cols-6 xl:grid-cols-7 gap-3 md:gap-4">
   409→                {Array.from({ length: 14 }).map((_, i) => <SkeletonCard key={i} />)}
   410→              </div>
   411→            )}
   412→            {mangaError && (
   413→              <div className="flex flex-col items-center justify-center py-20 text-center">
   414→                <AlertCircle className="w-10 h-10 text-white/15 mb-3" />
   415→                <p className="text-white/40 text-sm">Failed to load manga</p>
   416→                <button onClick={() => fetchManga(mangaSubTab, mangaSearch, 0)} className="mt-4 text-amber-500 text-sm">Retry</button>
   417→              </div>
   418→            )}
   419→            {!mangaLoading && !mangaError && (
   420→              mangaList.length === 0 ? (
   421→                <div className="flex flex-col items-center justify-center py-20 text-center">
   422→                  <Search className="w-10 h-10 text-white/15 mb-3" />
   423→                  <p className="text-white/40 text-sm">{mangaSearch ? `No results for "${mangaSearch}"` : 'No manga found'}</p>
   424→                </div>
   425→              ) : (
   426→                <>
   427→                  <div className="grid grid-cols-3 sm:grid-cols-4 md:grid-cols-5 lg:grid-cols-6 xl:grid-cols-7 gap-3 md:gap-4">
   428→                    {mangaList.map((manga, i) => (
   429→                      <motion.div
   430→                        key={manga.id}
   431→                        initial={{ opacity: 0, y: 16 }}
   432→                        whileInView={{ opacity: 1, y: 0 }}
   433→                        viewport={{ once: true, margin: '-40px' }}
   434→                        transition={{ duration: 0.3, delay: Math.min(i * 0.03, 0.3) }}
   435→                        className="cursor-pointer group"
   436→                        onClick={() => selectManga({ id: manga.id, title: manga.title, coverUrl: manga.coverUrl })}
   437→                      >
   438→                        <div className="aspect-[2/3] rounded-lg overflow-hidden bg-white/5 mb-2 relative">
   439→                          <img
   440→                            src={proxyCover(manga.coverUrl)}
   441→                            alt={manga.title}
   442→                            className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-300"
   443→                            loading="lazy"
   444→                            onError={(e) => { (e.target as HTMLImageElement).style.display = 'none'; }}
   445→                          />
   446→                          <div className="absolute inset-0 bg-black/0 group-hover:bg-black/50 transition-colors flex items-center justify-center">
   447→                            <BookOpen className="w-10 h-10 text-amber-400 opacity-0 group-hover:opacity-100 transition-opacity" />
   448→                          </div>
   449→                        </div>
   450→                        <p className="text-sm font-medium text-white/90 truncate">{manga.title}</p>
   451→                        {manga.author && <p className="text-xs text-white/40 truncate">{manga.author}</p>}
   452→                      </motion.div>
   453→                    ))}
   454→                  </div>
   455→                  {mangaHasMore && (
   456→                    <div className="flex justify-center mt-8 mb-4">
   457→                      <button onClick={() => { const next = mangaOffset + 20; setMangaOffset(next); fetchManga(mangaSubTab, mangaSearch, next, true); }} className="flex items-center gap-2 px-6 py-2.5 rounded-xl bg-white/10 hover:bg-white/15 text-white/80 text-sm font-medium transition-colors">
   458→                        Load More
   459→                      </button>
   460→                    </div>
   461→                  )}
   462→                </>
   463→              )
   464→            )}
