    44→function formatDuration(iso: string): string {
    45→  const m = iso.match(/PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?/);
    46→  if (!m) return '0:00';
    47→  const h = parseInt(m[1] || '0');
    48→  const min = parseInt(m[2] || '0');
    49→  const s = parseInt(m[3] || '0');
    50→  if (h > 0) return `${h}:${min.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    51→  return `${min}:${s.toString().padStart(2, '0')}`;
    52→}
    53→
    54→
    55→function formatDurationSeconds(iso: string): number {
    56→  const m = iso.match(/PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?/);
    57→  if (!m) return 0;
    58→  const h = parseInt(m[1] || '0');
    59→  const min = parseInt(m[2] || '0');
    60→  const s = parseInt(m[3] || '0');
    61→  return h * 3600 + min * 60 + s;
    62→}
    63→
    64→function formatTime(sec: number): string {
    65→  const m = Math.floor(sec / 60);
    66→  const s = Math.floor(sec % 60);
    67→  return `${m}:${s.toString().padStart(2, '0')}`;
    68→}

   251→  onNext,
   252→  onPrev,

   263→

   266→    setProgress(0);
   267→    const interval = setInterval(() => {
   268→      setProgress((p) => {
   269→        if (p >= 100) return 100;
   270→        return p + (100 / Math.max(totalSec, 1)) * 0.5;
   271→      });
   272→    }, 500);
   273→    return () => clearInterval(interval);
   274→  }, [track.videoId, totalSec]);
   275→

   291→            alt=""
   292→            className="w-full h-full object-cover scale-150 blur-3xl opacity-40"
   293→            onError={() => setImgError(true)}
   294→          />
   295→        ) : (
   296→          <div className="w-full h-full bg-gradient-to-b from-amber-900 to-black" />

   536→      <h2 className="text-base font-bold text-white">{title}</h2>
   537→    </div>
   538→  );
   539→}

   542→
   543→export function MusicPage() {
   544→  const [trending, setTrending] = useState<YoutubeVideoResult[]>([]);
   545→  const [searchResults, setSearchResults] = useState<YoutubeVideoResult[]>([]);
   546→  const [relatedResults, setRelatedResults] = useState<YoutubeVideoResult[]>([]);
   547→  const [loading, setLoading] = useState(true);
   548→  const [searchLoading, setSearchLoading] = useState(false);
   549→  const [searchQuery, setSearchQuery] = useState('');
   550→  const [activeMood, setActiveMood] = useState('');
   551→  const [currentTrack, setCurrentTrack] = useState<YoutubeVideoResult | null>(null);
   552→  const [showFullScreen, setShowFullScreen] = useState(false);
   553→  const [queue, setQueue] = useState<YoutubeVideoResult[]>([]);
   554→  const [queueIndex, setQueueIndex] = useState(0);
   555→  const searchTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   556→  const searchInputRef = useRef<HTMLInputElement>(null);
   557→
   558→  // Fetch trending on mount
   559→  useEffect(() => {
   560→    async function fetchTrending() {
   561→      try {
   562→        const res = await fetch('/api/youtube/trending');
   563→        const data: ApiResponse = await res.json();
   564→        setTrending(data.results || []);
   565→      } catch (err) {
   566→        console.error('Failed to fetch trending:', err);
   567→      } finally {
   568→        setLoading(false);
   569→      }
   570→    }
   571→    fetchTrending();
   572→  }, []);
   573→
   574→  // Search with debounce
   575→  const handleSearchInput = useCallback((value: string) => {
   576→    setSearchQuery(value);
   577→    setActiveMood('');
   578→    if (searchTimerRef.current) clearTimeout(searchTimerRef.current);
   579→    if (!value.trim()) {
   580→      setSearchResults([]);
   581→      setRelatedResults([]);
   582→      return;
   583→    }
   584→    searchTimerRef.current = setTimeout(async () => {
   585→      setSearchLoading(true);
   586→      try {
   587→        const res = await fetch(`/api/youtube/search?query=${encodeURIComponent(value)}`);
   588→        const data: ApiResponse = await res.json();
   589→        setSearchResults(data.results || []);
   590→        if (data.results?.[0]) {
   591→          const relatedRes = await fetch(`/api/youtube/related?videoId=${data.results[0].videoId}`);
   592→          const relatedData: ApiResponse = await relatedRes.json();
   593→          setRelatedResults(relatedData.results || []);
   594→        } else {
   595→          setRelatedResults([]);
   596→        }
   597→      } catch {
   598→        setSearchResults([]);
   599→        setRelatedResults([]);
   600→      } finally {
   601→        setSearchLoading(false);
   602→      }
   603→    }, 500);
   604→  }, []);
   605→
   606→  // Mood card click
   607→  const handleMoodClick = useCallback((mood: typeof MOOD_CARDS[number]) => {
   608→    const isCurrentlyActive = activeMood === mood.label;
   609→    setActiveMood(isCurrentlyActive ? '' : mood.label);
   610→    if (!isCurrentlyActive) {
   611→      handleSearchInput(mood.query);
   612→      if (searchInputRef.current) searchInputRef.current.value = mood.query;
   613→    } else {
   614→      setSearchQuery('');
   615→      setSearchResults([]);
   616→      setRelatedResults([]);
   617→      if (searchInputRef.current) searchInputRef.current.value = '';
   618→    }
   619→  }, [activeMood, handleSearchInput]);
   620→
   621→  // Play a track
   622→  const handlePlay = useCallback((track: YoutubeVideoResult, trackList?: YoutubeVideoResult[]) => {
   623→    setCurrentTrack(track);
   624→    setShowFullScreen(true);
   625→    if (trackList) {
   626→      const idx = trackList.findIndex((t) => t.videoId === track.videoId);
   627→      setQueue(trackList);
   628→      setQueueIndex(idx >= 0 ? idx : 0);
   629→    }
   630→  }, []);
   631→
   632→  // Next/Prev
   633→  const handleNext = useCallback(() => {
   634→    if (queue.length === 0) return;
   635→    const nextIdx = (queueIndex + 1) % queue.length;
   636→    setQueueIndex(nextIdx);
   637→    setCurrentTrack(queue[nextIdx]);
   638→  }, [queue, queueIndex]);
   639→
   640→  const handlePrev = useCallback(() => {
   641→    if (queue.length === 0) return;
   642→    const prevIdx = (queueIndex - 1 + queue.length) % queue.length;
   643→    setQueueIndex(prevIdx);
   644→    setCurrentTrack(queue[prevIdx]);
   645→  }, [queue, queueIndex]);
   646→
   647→  // Close
   648→  const handleClosePlayer = useCallback(() => {
   649→    setShowFullScreen(false);
   650→  }, []);
   651→
   652→  const isSearching = searchQuery.trim().length > 0;
   653→  const hasPlayer = currentTrack !== null;
   654→
   655→  return (
   656→    <div className={`${hasPlayer && !showFullScreen ? 'pb-[7.5rem]' : 'pb-24'}`}>
   657→      {/* ── Search Bar (sticky) ── */}
   658→      <div className="sticky top-[calc(3rem+env(safe-area-inset-top,0px))] z-30 bg-black/80 backdrop-blur-xl px-4 py-3">
   659→        <div className="relative">
   660→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-white/30" />
   661→          <Input
   662→            ref={searchInputRef}
   663→            type="text"
   664→            placeholder="Search songs, artists..."
   665→            defaultValue={searchQuery}
   666→            onChange={(e) => handleSearchInput(e.target.value)}
   667→            className="pl-10 pr-10 bg-white/10 border-white/10 text-white placeholder:text-white/30 focus:border-amber-500/50 h-10 rounded-xl text-sm"
   668→          />
   669→          {searchQuery && (
   670→            <button
   671→              onClick={() => {
   672→                handleSearchInput('');
   673→                if (searchInputRef.current) searchInputRef.current.value = '';
   674→              }}
   675→              className="absolute right-3 top-1/2 -translate-y-1/2 text-white/30 hover:text-white transition-colors"
   676→            >
   677→              <X className="w-4 h-4" />
   678→            </button>
   679→          )}
   680→        </div>
   681→      </div>
   682→
   683→      {/* ── Mood / Genre Cards ── */}
   684→      {!isSearching && (
   685→        <div className="px-4 mb-6">
   686→          <SectionHeader title="Browse Moods" icon={<span className="text-lg">🎵</span>} />
   687→          <div className="flex gap-3 overflow-x-auto pb-2 -mx-4 px-4 content-scroll">
   688→            {MOOD_CARDS.map((mood) => (
   689→              <MoodCard
   690→                key={mood.label}
   691→                label={mood.label}
   692→                gradient={mood.gradient}
   693→                emoji={mood.emoji}
   694→                onClick={() => handleMoodClick(mood)}
   695→                isActive={activeMood === mood.label}
   696→              />
   697→            ))}
   698→          </div>
   699→        </div>
   700→      )}
   701→
   702→      {/* ── Genre Pills (when searching) ── */}
   703→      {isSearching && (
   704→        <div className="px-4 mb-4">
   705→          <div className="flex gap-2 overflow-x-auto pb-1 -mx-4 px-4 content-scroll">
   706→            {GENRE_PILLS.map((genre) => (
   707→              <button
   708→                key={genre}
   709→                onClick={() => {
   710→                  handleSearchInput(genre);
   711→                  if (searchInputRef.current) searchInputRef.current.value = genre;
   712→                }}
   713→                className="shrink-0 px-3 py-1 rounded-full text-xs font-medium bg-white/5 text-white/50 border border-white/5 active:bg-white/10 active:text-white transition-colors"
   714→              >
   715→                {genre}
   716→              </button>
   717→            ))}
   718→          </div>
   719→        </div>
   720→      )}
   721→
   722→      {/* ── Content ── */}
   723→      <div className="px-4">
   724→        {isSearching ? (
   725→          <>
   726→            {searchLoading ? (
   727→              <TrackGridSkeleton />
   728→            ) : searchResults.length > 0 ? (
   729→              <>
   730→                <SectionHeader
   731→                  icon={<Search className="w-4 h-4" />}
   732→                  title={searchQuery}
   733→                />
   734→                <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6 gap-3 md:gap-4 mb-6">
   735→                  {searchResults.map((track, i) => (
   736→                    <TrackCard
   737→                      key={track.videoId}
   738→                      track={track}
   739→                      onPlay={(t) => handlePlay(t, searchResults)}
   740→                      isActive={currentTrack?.videoId === track.videoId}
   741→                      index={i}
   742→                    />
   743→                  ))}
   744→                </div>
   745→              </>
   746→            ) : (
   747→              <div className="text-center py-20">
   748→                <Music className="w-10 h-10 text-white/10 mx-auto mb-3" />
   749→                <p className="text-white/30 text-sm">No results found</p>
   750→              </div>
   751→            )}
   752→
   753→            {/* Related */}
   754→            {relatedResults.length > 0 && (
   755→              <>
   756→                <SectionHeader
   757→                  icon={<TrendingUp className="w-4 h-4" />}
   758→                  title="You Might Also Like"
   759→                />
   760→                <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6 gap-3 md:gap-4 mb-6">
   761→                  {relatedResults.map((track, i) => (
   762→                    <TrackCard
   763→                      key={track.videoId}
   764→                      track={track}
   765→                      onPlay={(t) => handlePlay(t, relatedResults)}
   766→                      isActive={currentTrack?.videoId === track.videoId}
   767→                      index={i}
   768→                    />
   769→                  ))}
   770→                </div>
   771→              </>
   772→            )}
   773→          </>
   774→        ) : (
   775→          <>
   776→            {loading ? (
   777→              <>
   778→                {/* Mood skeleton */}
   779→                <div className="flex gap-3 overflow-x-auto pb-2 -mx-4 px-4 mb-6">
   780→                  {Array.from({ length: 4 }).map((_, i) => (
   781→                    <MoodCardSkeleton key={i} />
   782→                  ))}
   783→                </div>
   784→                <TrackGridSkeleton />
   785→              </>
   786→            ) : (
   787→              <>
   788→                <SectionHeader
   789→                  icon={<TrendingUp className="w-4 h-4" />}
   790→                  title="Trending Now"
   791→                />
   792→                <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6 gap-3 md:gap-4">
   793→                  {trending.map((track, i) => (
   794→                    <TrackCard
   795→                      key={track.videoId}
   796→                      track={track}
   797→                      onPlay={(t) => handlePlay(t, trending)}
   798→                      isActive={currentTrack?.videoId === track.videoId}
   799→                      index={i}
   800→                    />
   801→                  ))}
   802→                </div>
   803→              </>
   804→            )}
   805→          </>
   806→        )}
   807→      </div>
   808→
   809→      {/* Footer */}
   810→      <div className="mt-8 border-t border-white/5 pt-4 pb-2 px-4 text-center">
   811→        <p className="text-[10px] text-white/15">
   812→          StreamVault Music — Powered by YouTube
   813→        </p>
   814→      </div>
   815→
   816→      {/* ── Mini Player ── */}
   817→      <AnimatePresence>
   818→        {currentTrack && !showFullScreen && (
   819→          <MiniPlayer
   820→            track={currentTrack}
   821→            onExpand={() => setShowFullScreen(true)}
   822→            onNext={handleNext}
   823→            onPrev={handlePrev}
   824→          />
   825→        )}
   826→      </AnimatePresence>
   827→
   828→      {/* ── Full Screen Player ── */}
   829→      <AnimatePresence>
   830→        {currentTrack && showFullScreen && (
   831→          <FullScreenPlayer
   832→            track={currentTrack}
   833→            onClose={handleClosePlayer}
   834→            onNext={handleNext}
   835→            onPrev={handlePrev}
   836→          />
   837→        )}
   838→      </AnimatePresence>
   839→    </div>
   840→  );
   841→}
