    95→    95→export function ReadPage() {
    96→    96→  const { goHome, selectManga, selectNovel } = useAppStore();
    97→    97→  const [mainTab, setMainTab] = useState<MainTab>('manga');
    98→    98→
    99→    99→  // Manga state
   100→   100→  const [mangaList, setMangaList] = useState<MangaItem[]>([]);
   101→   101→  const [mangaSubTab, setMangaSubTab] = useState('all');
   102→   102→  const [mangaLoading, setMangaLoading] = useState(false);
   103→   103→  const [mangaError, setMangaError] = useState(false);
   104→   104→  const [mangaSearch, setMangaSearch] = useState('');
   105→   105→  const [mangaHasMore, setMangaHasMore] = useState(false);
   106→   106→  const [mangaTotal, setMangaTotal] = useState(0);
   107→   107→  const [mangaOffset, setMangaOffset] = useState(0);
   108→   108→  const mangaDebounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   109→   109→  const mangaAbortRef = useRef<AbortController | null>(null);
   110→   110→
   111→   111→  // Comics state
   112→   112→  const [comicList, setComicList] = useState<MangaItem[]>([]);
   113→   113→  const [comicSubTab, setComicSubTab] = useState('trending');
   114→   114→  const [comicLoading, setComicLoading] = useState(false);
   115→   115→  const [comicError, setComicError] = useState(false);
   116→   116→  const [comicTotal, setComicTotal] = useState(0);
   117→   117→
   118→   118→  // Novels state
   119→   119→  const [novelList, setNovelList] = useState<NovelItem[]>([]);
   120→   120→  const [novelGenre, setNovelGenre] = useState('All');
   121→   121→  const [novelSearch, setNovelSearch] = useState('');
   122→   122→  const [novelLoading, setNovelLoading] = useState(true);
   123→   123→  const novelDebounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   124→   124→
   125→   125→  /* ── Manga fetch ── */
   126→   126→  const fetchManga = useCallback(async (tab: string, search: string, offset: number, append = false) => {
   127→   127→    mangaAbortRef.current?.abort();
   128→   128→    const controller = new AbortController();
   129→   129→    mangaAbortRef.current = controller;
   130→   130→    if (!append) { setMangaLoading(true); setMangaError(false); }
   131→   131→
   132→   132→    try {
   133→   133→      const params = new URLSearchParams();
   134→   134→      if (search.trim()) {
   135→   135→        params.set('q', search.trim());
   136→   136→        if (offset > 0) params.set('page', String(Math.floor(offset / 20) + 1));
   137→   137→        const res = await fetch(`/api/manga/search?${params}`, { signal: controller.signal });
   138→   138→        if (!res.ok) throw new Error();
   139→   139→        const data = await res.json();
   140→   140→        if (controller.signal.aborted) return;
   141→   141→        const items: MangaItem[] = (data.results || []).map((m: any) => ({
   142→   142→          id: String(m.id), title: String(m.title || 'Untitled'),
   143→   143→          coverUrl: String(m.coverUrl || ''), author: String(m.author || ''),
   144→   144→          tags: Array.isArray(m.tags) ? m.tags.filter(Boolean).map(String) : [],
   145→   145→        }));
   146→   146→        setMangaList(append ? (prev) => [...prev, ...items] : items);
   147→   147→        setMangaHasMore(!!data.hasMore);
   148→   148→        setMangaTotal(data.total || 0);
   149→   149→      } else {
   150→   150→        if (tab === 'manga' || tab === 'manhwa' || tab === 'manhua') params.set('type', tab);
   151→   151→        if (offset > 0) params.set('offset', String(offset));
   152→   152→        const url = `/api/manga/trending${params.toString() ? `?${params}` : ''}`;
   153→   153→        const res = await fetch(url, { signal: controller.signal });
   154→   154→        if (!res.ok) throw new Error();
   155→   155→        const data = await res.json();
   156→   156→        if (controller.signal.aborted) return;
   157→   157→        const items: MangaItem[] = (data.results || []).map((m: any) => ({
   158→   158→          id: String(m.id), title: String(m.title || 'Untitled'),
   159→   159→          coverUrl: String(m.coverUrl || ''), author: String(m.author || ''),
   160→   160→          tags: Array.isArray(m.tags) ? m.tags.filter(Boolean).map(String) : [],
   161→   161→        }));
   162→   162→        setMangaList(append ? (prev) => [...prev, ...items] : items);
   163→   163→        setMangaHasMore(!!data.hasMore);
   164→   164→        setMangaTotal(data.total || 0);
   165→   165→      }
   166→   166→    } catch {
   167→   167→      if (!append) setMangaError(true);
   168→   168→    } finally {
   169→   169→      if (!controller.signal.aborted) setMangaLoading(false);
   170→   170→    }
   171→   171→  }, []);
   172→   172→
   173→   173→  useEffect(() => {
   174→   174→    setMangaOffset(0); setMangaSearch('');
   175→   175→    fetchManga(mangaSubTab, '', 0);
   176→   176→    return () => { mangaAbortRef.current?.abort(); };
   177→   177→  }, [mangaSubTab, fetchManga]);
   178→   178→
   179→   179→  const handleMangaSearch = useCallback((value: string) => {
   180→   180→    setMangaSearch(value);
   181→   181→    if (mangaDebounceRef.current) clearTimeout(mangaDebounceRef.current);
   182→   182→    if (!value.trim()) { setMangaOffset(0); fetchManga(mangaSubTab, '', 0); return; }
   183→   183→    mangaDebounceRef.current = setTimeout(() => { setMangaOffset(0); fetchManga(mangaSubTab, value, 0); }, 300);
   184→   184→  }, [mangaSubTab, fetchManga]);
   185→   185→
   186→   186→  /* ── Comics fetch ── */
   187→   187→  const fetchComics = useCallback(async () => {
   188→   188→    setComicLoading(true); setComicError(false);
   189→   189→    try {
   190→   190→      const params = new URLSearchParams();
   191→   191→      if (comicSubTab === 'popular') params.set('order', 'popular');
   192→   192→      params.set('limit', '20');
   193→   193→      const res = await fetch(`/api/comics/trending?${params}`);
   194→   194→      if (!res.ok) throw new Error();
   195→   195→      const data = await res.json();
   196→   196→      const items: MangaItem[] = (data.results || []).map((m: any) => ({
   197→   197→        id: String(m.id), title: String(m.title || 'Untitled'),
   198→   198→        coverUrl: String(m.coverUrl || ''), author: String(m.author || ''),
   199→   199→        tags: Array.isArray(m.tags) ? m.tags.filter(Boolean).map(String) : [],
   200→   200→      }));
   201→   201→      setComicList(items);
   202→   202→      setComicTotal(data.total || 0);
   203→   203→    } catch { setComicError(true); }
   204→   204→    finally { setComicLoading(false); }
   205→   205→  }, [comicSubTab]);
   206→   206→
   207→   207→  useEffect(() => { fetchComics(); }, [fetchComics]);
   208→   208→
   209→   209→  /* ── Novels fetch ── */
   210→   210→  useEffect(() => {
   211→   211→    setNovelLoading(true);
   212→   212→    fetch('/novels-data.json')
   213→   213→      .then((r) => r.json())
   214→   214→      .then((data: NovelItem[]) => setNovelList(data))
   215→   215→      .catch(() => {})
   216→   216→      .finally(() => setNovelLoading(false));
   217→   217→  }, []);
   218→   218→
   219→   219→  const filteredNovels = useMemo(() => {
   220→   220→    let list = novelList;
   221→   221→    if (novelGenre !== 'All') list = list.filter((n) => n.genre.includes(novelGenre));
   222→   222→    if (novelSearch.trim()) {
   223→   223→      const q = novelSearch.toLowerCase();
   224→   224→      list = list.filter(
   225→   225→        (n) =>
   226→   226→          n.title.toLowerCase().includes(q) ||
   227→   227→          n.author.toLowerCase().includes(q) ||
   228→   228→          n.description.toLowerCase().includes(q)
   229→   229→      );
   230→   230→    }
   231→   231→    return list;
   232→   232→  }, [novelList, novelGenre, novelSearch]);
   233→   233→
   234→   234→  const handleNovelSearch = useCallback((value: string) => {
   235→   235→    setNovelSearch(value);
   236→   236→    if (novelDebounceRef.current) clearTimeout(novelDebounceRef.current);
   237→   237→  }, []);
   238→   238→
   239→   239→  /* ── Shared search input ── */
   240→   240→  const searchPlaceholder = mainTab === 'manga'
   241→   241→    ? 'Search manga, manhwa, manhua...'
   242→   242→    : mainTab === 'comics'
   243→   243→    ? 'Browse comics'
   244→   244→    : 'Search classic novels...';
   245→   245→
   246→   246→  const currentSearch = mainTab === 'manga' ? mangaSearch : novelSearch;
   247→   247→  const handleSearch = mainTab === 'manga' ? handleMangaSearch : handleNovelSearch;
   248→   248→
   249→   249→  /* ── Active color ── */
   250→   250→  const activeColor = mainTab === 'manga' ? 'amber' : mainTab === 'comics' ? 'violet' : 'emerald';
   251→   251→
   252→   252→  /* ── Render sub-tabs ── */
   253→   253→  const subTabs: SubTab[] =
   254→   254→    mainTab === 'manga' ? MANGA_TABS : mainTab === 'comics' ? COMICS_TABS : [];
   255→   255→  const activeSubTab = mainTab === 'manga' ? mangaSubTab : mainTab === 'comics' ? comicSubTab : '';
   256→   256→  const setActiveSubTab = mainTab === 'manga' ? setMangaSubTab : setComicSubTab;
   257→   257→
   258→   258→  /* ── Item count ── */
   259→   259→  const itemCount =
   260→   260→    mainTab === 'manga' ? mangaTotal :
   261→   261→    mainTab === 'comics' ? comicTotal :
   262→   262→    filteredNovels.length;
   263→   263→
   264→   264→  return (
   265→   265→    <div className="min-h-screen pb-10">
   266→   266→      {/* Mobile back button */}
   267→   267→      <button
   268→   268→        onClick={goHome}
   269→   269→        className="md:hidden fixed z-[90] flex items-center gap-1.5 text-white/60 active:text-white transition-colors"
   270→   270→        style={{ top: 'max(env(safe-area-inset-top, 0px) + 8px, 8px)', left: 12 }}
   271→   271→        aria-label="Go home"
   272→   272→      >
   273→   273→        <ArrowLeft className="w-5 h-5" />
   274→   274→      </button>
   275→   275→
   276→   276→      <div className="pt-16 md:pt-8 pb-6 px-4 md:px-8">
   277→   277→        {/* Desktop back */}
   278→   278→        <button
   279→   279→          onClick={goHome}
   280→   280→          className="hidden md:flex items-center gap-2 text-white/70 hover:text-white mb-4 transition-colors"
   281→   281→        >
   282→   282→          <ArrowLeft className="w-4 h-4" />
   283→   283→          <span className="text-sm">Back to Home</span>
   284→   284→        </button>
   285→   285→
   286→   286→        {/* ── Hero header ── */}
   287→   287→        <div className="relative mb-6">
   288→   288→          <div className={`relative h-[20vh] min-h-[160px] md:min-h-[200px] rounded-2xl overflow-hidden`}>
   289→   289→            <div className={`absolute inset-0 bg-gradient-to-br ${MAIN_TABS.find(t => t.key === mainTab)?.color || 'from-amber-500 to-orange-600'}`} />
   290→   290→            <div className="absolute inset-0 opacity-20" style={{
   291→   291→              backgroundImage: 'radial-gradient(circle at 20% 80%, rgba(255,255,255,0.15) 0%, transparent 50%), radial-gradient(circle at 80% 20%, rgba(255,255,255,0.1) 0%, transparent 50%)',
   292→   292→            }} />
   293→   293→            <div className="absolute inset-0 bg-gradient-to-t from-black/60 via-transparent to-black/20" />
   294→   294→            <div className="relative h-full flex items-center justify-center text-center px-4">
   295→   295→              <div>
   296→   296→                <div className={`w-14 h-14 rounded-2xl bg-gradient-to-br ${MAIN_TABS.find(t => t.key === mainTab)?.color || ''} flex items-center justify-center shadow-lg mb-3`}>
   297→   297→                  <BookOpen className="w-7 h-7 text-white" />
   298→   298→                </div>
   299→   299→                <h1 className="text-3xl md:text-4xl font-black text-white mb-1">
   300→   300→                  Read
   301→   301→                </h1>
   302→   302→                <p className="text-white/50 text-xs md:text-sm max-w-md mx-auto">
   303→   303→                  {mainTab === 'manga' && 'Manga, Manhwa & Manhua'}
   304→   304→                  {mainTab === 'comics' && 'Webtoons, Manhwa & Comics'}
   305→   305→                  {mainTab === 'novels' && 'Classic Novels & Literature'}
   306→   306→                  &middot; {itemCount > 0 && !mangaLoading && !comicLoading && !novelLoading ? `${itemCount.toLocaleString()} titles` : 'Free to read'}
   307→   307→                </p>
   308→   308→              </div>
   309→   309→            </div>
   310→   310→          </div>
   311→   311→        </div>
   312→   312→
   313→   313→        {/* ── Main tabs ── */}
   314→   314→        <div className="flex gap-1 mb-5">
   315→   315→          {MAIN_TABS.map((tab) => {
   316→   316→            const IconComp = tab.icon;
   317→   317→            return (
   318→   318→              <button
   319→   319→                key={tab.key}
   320→   320→                onClick={() => setMainTab(tab.key)}
   321→   321→                className={`shrink-0 px-4 py-2.5 rounded-xl text-sm font-semibold transition-all relative ${
   322→   322→                  mainTab === tab.key
   323→   323→                    ? `bg-${activeColor}-500/15 text-${activeColor}-400 border border-${activeColor}-500/30`
   324→   324→                    : 'bg-white/5 text-white/50 hover:bg-white/10 hover:text-white/70 border border-white/[0.08]'
   325→   325→                }`}
   326→   326→              >
   327→   327→                <IconComp className="w-4 h-4" />
   328→   328→                <span className="ml-1.5">{tab.label}</span>
   329→   329→                {mainTab === tab.key && (
   330→   330→                  <motion.div
   331→   331→                    layoutId="read-main-tab"
   332→   332→                    className="absolute inset-0 rounded-xl bg-${activeColor}-500/15 border border-${activeColor}-500/30"
   333→   333→                    transition={{ type: 'spring', stiffness: 400, damping: 30 }}
   334→   334→                  />
   335→   335→                )}
   336→   336→              </button>
   337→   337→            );
   338→   338→          })}
   339→   339→        </div>
   340→   340→
   341→   341→        {/* ── Search ── */}
   342→   342→        {mainTab !== 'comics' && (
   343→   343→          <div className="relative mb-5">
   344→   344→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-white/40" />
   345→   345→            <input
   346→   346→              type="text"
   347→   347→              placeholder={searchPlaceholder}
   348→   348→              value={currentSearch}
   349→   349→              onChange={(e) => handleSearch(e.target.value)}
   350→   350→              className="w-full bg-white/5 border border-white/10 rounded-xl pl-10 pr-4 py-2.5 text-sm text-white placeholder:text-white/50 focus:outline-none focus:border-amber-500/50 focus:ring-1 focus:ring-amber-500/30 transition-colors"
   351→   351→            />
   352→   352→          </div>
   353→   353→        )}
   354→   354→
   355→   355→        {/* ── Sub-tabs / filters ── */}
   356→   356→        {subTabs.length > 0 && (
   357→   357→          <div className="flex gap-1 overflow-x-auto scrollbar-none pb-3 -mx-1 px-1">
   358→   358→            {subTabs.map((tab) => (
   359→   359→              <button
   360→   360→                key={tab.key}
   361→   361→                onClick={() => setActiveSubTab(tab.key)}
   362→   362→                className={`shrink-0 px-3 py-1.5 rounded-lg text-xs font-medium transition-colors relative whitespace-nowrap ${
   363→   363→                  activeSubTab === tab.key
   364→   364→                    ? `text-${activeColor}-400`
   365→   365→                    : 'text-white/40 hover:text-white/60'
   366→   366→                }`}
   367→   367→              >
   368→   368→                {tab.flag && <span className="mr-1">{tab.flag}</span>}
   369→   369→                {tab.label}
   370→   370→                {activeSubTab === tab.key && (
   371→   371→                  <motion.div
   372→   372→                    layoutId="read-sub-tab"
   373→   373→                    className={`absolute bottom-0 left-0 right-0 h-0.5 bg-${activeColor}-500 rounded-full`}
   374→   374→                    transition={{ type: 'spring', stiffness: 400, damping: 30 }}
   375→   375→                  />
   376→   376→                )}
   377→   377→              </button>
   378→   378→            ))}
   379→   379→          </div>
   380→   380→        )}
   381→   381→
   382→   382→        {/* ── Novel genre pills ── */}
   383→   383→        {mainTab === 'novels' && (
   384→   384→          <div className="flex gap-1.5 overflow-x-auto scrollbar-none pb-3 -mx-1 px-1">
   385→   385→            {NOVEL_GENRES.map((genre) => (
   386→   386→              <button
   387→   387→                key={genre}
   388→   388→                onClick={() => setNovelGenre(genre)}
   389→   389→                className={`shrink-0 px-3 py-1.5 rounded-full text-[11px] font-semibold transition-all border ${
   390→   390→                  novelGenre === genre
   391→   391→                    ? `bg-emerald-500/15 text-emerald-300 border-emerald-500/30`
   392→   392→                    : 'bg-white/[0.04] text-white/40 border-white/[0.08] hover:bg-white/[0.08] hover:text-white/70'
   393→   393→                }`}
   394→   394→              >
   395→   395→                {genre}
   396→   396→              </button>
   397→   397→            ))}
   398→   398→          </div>
   399→   399→        )}
   400→   400→      </div>
   401→   401→
   402→   402→      {/* ═══ Content area ═══ */}
