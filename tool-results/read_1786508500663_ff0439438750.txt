    44→function formatDuration(iso: string): string {
    45→  const m = iso.match(/PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?/);
    46→  if (!m) return '0:00';
    47→  const h = parseInt(m[1] || '0');
    48→  const min = parseInt(m[2] || '0');
    49→  const s = parseInt(m[3] || '0');
    50→  if (h > 0) return `${h}:${min.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    51→  return `${min}:${s.toString().padStart(2, '0')}`;
    52→}
    53→
    54→
    55→function formatDurationSeconds(iso: string): number {
    56→  const m = iso.match(/PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?/);
    57→  if (!m) return 0;
    58→  const h = parseInt(m[1] || '0');
    59→  const min = parseInt(m[2] || '0');
    60→  const s = parseInt(m[3] || '0');
    61→  return h * 3600 + min * 60 + s;
    62→}
    63→
    64→function formatTime(sec: number): string {
    65→  const m = Math.floor(sec / 60);
    66→  const s = Math.floor(sec % 60);
    67→  return `${m}:${s.toString().padStart(2, '0')}`;
    68→}

   269→

   271→    const interval = setInterval(() => {
   272→      setProgress((p) => {
   273→        if (p >= 100) return 100;

   275→      });
   276→    }, 500);
   277→    return () => clearInterval(interval);
   278→  }, [totalSec]);
   279→

   283→    <motion.div
   284→      initial={{ y: '100%' }}

   293→          <img
   294→            src={track.thumbnail}
   295→            alt=""
   296→            className="w-full h-full object-cover scale-150 blur-3xl opacity-40"
   297→            onError={() => setImgError(true)}
   298→          />
   299→        ) : (
   300→          <div className="w-full h-full bg-gradient-to-b from-amber-900 to-black" />

   544→// ─── Main Component ─────────────────────────────────────────────────────────
   545→
   546→export function MusicPage() {
   547→  const [trending, setTrending] = useState<YoutubeVideoResult[]>([]);
   548→  const [searchResults, setSearchResults] = useState<YoutubeVideoResult[]>([]);
   549→  const [relatedResults, setRelatedResults] = useState<YoutubeVideoResult[]>([]);
   550→  const [loading, setLoading] = useState(true);
   551→  const [searchLoading, setSearchLoading] = useState(false);
   552→  const [searchQuery, setSearchQuery] = useState('');
   553→  const [activeMood, setActiveMood] = useState('');
   554→  const [currentTrack, setCurrentTrack] = useState<YoutubeVideoResult | null>(null);
   555→  const [showFullScreen, setShowFullScreen] = useState(false);
   556→  const [queue, setQueue] = useState<YoutubeVideoResult[]>([]);
   557→  const [queueIndex, setQueueIndex] = useState(0);
   558→  const searchTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   559→  const searchInputRef = useRef<HTMLInputElement>(null);
   560→
   561→  // Fetch trending on mount
   562→  useEffect(() => {
   563→    async function fetchTrending() {
   564→      try {
   565→        const res = await fetch('/api/youtube/trending');
   566→        const data: ApiResponse = await res.json();
   567→        setTrending(data.results || []);
   568→      } catch (err) {
   569→        console.error('Failed to fetch trending:', err);
   570→      } finally {
   571→        setLoading(false);
   572→      }
   573→    }
   574→    fetchTrending();
   575→  }, []);
   576→
   577→  // Search with debounce
   578→  const handleSearchInput = useCallback((value: string) => {
   579→    setSearchQuery(value);
   580→    setActiveMood('');
   581→    if (searchTimerRef.current) clearTimeout(searchTimerRef.current);
   582→    if (!value.trim()) {
   583→      setSearchResults([]);
   584→      setRelatedResults([]);
   585→      return;
   586→    }
   587→    searchTimerRef.current = setTimeout(async () => {
   588→      setSearchLoading(true);
   589→      try {
   590→        const res = await fetch(`/api/youtube/search?query=${encodeURIComponent(value)}`);
   591→        const data: ApiResponse = await res.json();
   592→        setSearchResults(data.results || []);
   593→        if (data.results?.[0]) {
   594→          const relatedRes = await fetch(`/api/youtube/related?videoId=${data.results[0].videoId}`);
   595→          const relatedData: ApiResponse = await relatedRes.json();
   596→          setRelatedResults(relatedData.results || []);
   597→        } else {
   598→          setRelatedResults([]);
   599→        }
   600→      } catch {
   601→        setSearchResults([]);
   602→        setRelatedResults([]);
   603→      } finally {
   604→        setSearchLoading(false);
   605→      }
   606→    }, 500);
   607→  }, []);
   608→
   609→  // Mood card click
   610→  const handleMoodClick = useCallback((mood: typeof MOOD_CARDS[number]) => {
   611→    const isCurrentlyActive = activeMood === mood.label;
   612→    setActiveMood(isCurrentlyActive ? '' : mood.label);
   613→    if (!isCurrentlyActive) {
   614→      handleSearchInput(mood.query);
   615→      if (searchInputRef.current) searchInputRef.current.value = mood.query;
   616→    } else {
   617→      setSearchQuery('');
   618→      setSearchResults([]);
   619→      setRelatedResults([]);
   620→      if (searchInputRef.current) searchInputRef.current.value = '';
   621→    }
   622→  }, [activeMood, handleSearchInput]);
   623→
   624→  // Play a track
   625→  const handlePlay = useCallback((track: YoutubeVideoResult, trackList?: YoutubeVideoResult[]) => {
   626→    setCurrentTrack(track);
   627→    setShowFullScreen(true);
   628→    if (trackList) {
   629→      const idx = trackList.findIndex((t) => t.videoId === track.videoId);
   630→      setQueue(trackList);
   631→      setQueueIndex(idx >= 0 ? idx : 0);
   632→    }
   633→  }, []);
   634→
   635→  // Next/Prev
   636→  const handleNext = useCallback(() => {
   637→    if (queue.length === 0) return;
   638→    const nextIdx = (queueIndex + 1) % queue.length;
   639→    setQueueIndex(nextIdx);
   640→    setCurrentTrack(queue[nextIdx]);
   641→  }, [queue, queueIndex]);
   642→
   643→  const handlePrev = useCallback(() => {
   644→    if (queue.length === 0) return;
   645→    const prevIdx = (queueIndex - 1 + queue.length) % queue.length;
   646→    setQueueIndex(prevIdx);
   647→    setCurrentTrack(queue[prevIdx]);
   648→  }, [queue, queueIndex]);
   649→
   650→  // Close
   651→  const handleClosePlayer = useCallback(() => {
   652→    setShowFullScreen(false);
   653→  }, []);
   654→
   655→  const isSearching = searchQuery.trim().length > 0;
   656→  const hasPlayer = currentTrack !== null;
   657→
   658→  return (
   659→    <div>
   660→      {/* ── Search Bar (sticky) ── */}
   661→      <div className="sticky top-0 z-30 bg-black/80 backdrop-blur-xl px-4 py-3 pt-14 md:pt-3">
   662→        <div className="relative">
   663→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-white/30" />
   664→          <Input
   665→            ref={searchInputRef}
   666→            type="text"
   667→            placeholder="Search songs, artists..."
   668→            defaultValue={searchQuery}
   669→            onChange={(e) => handleSearchInput(e.target.value)}
   670→            className="pl-10 pr-10 bg-white/10 border-white/10 text-white placeholder:text-white/30 focus:border-amber-500/50 h-10 rounded-xl text-sm"
   671→          />
   672→          {searchQuery && (
   673→            <button
   674→              onClick={() => {
   675→                handleSearchInput('');
   676→                if (searchInputRef.current) searchInputRef.current.value = '';
   677→              }}
   678→              className="absolute right-3 top-1/2 -translate-y-1/2 text-white/30 hover:text-white transition-colors"
   679→            >
   680→              <X className="w-4 h-4" />
   681→            </button>
   682→          )}
   683→        </div>
   684→      </div>
   685→
   686→      {/* ── Mood / Genre Cards ── */}
   687→      {!isSearching && (
   688→        <div className="px-4 mb-6">
   689→          <SectionHeader title="Browse Moods" icon={<span className="text-lg">🎵</span>} />
   690→          <div className="flex gap-3 overflow-x-auto pb-2 -mx-4 px-4 content-scroll">
   691→            {MOOD_CARDS.map((mood) => (
   692→              <MoodCard
   693→                key={mood.label}
   694→                label={mood.label}
   695→                gradient={mood.gradient}
   696→                emoji={mood.emoji}
   697→                onClick={() => handleMoodClick(mood)}
   698→                isActive={activeMood === mood.label}
   699→              />
   700→            ))}
   701→          </div>
   702→        </div>
   703→      )}
   704→
   705→      {/* ── Genre Pills (when searching) ── */}
   706→      {isSearching && (
   707→        <div className="px-4 mb-4">
   708→          <div className="flex gap-2 overflow-x-auto pb-1 -mx-4 px-4 content-scroll">
   709→            {GENRE_PILLS.map((genre) => (
   710→              <button
   711→                key={genre}
   712→                onClick={() => {
   713→                  handleSearchInput(genre);
   714→                  if (searchInputRef.current) searchInputRef.current.value = genre;
   715→                }}
   716→                className="shrink-0 px-3 py-1 rounded-full text-xs font-medium bg-white/5 text-white/50 border border-white/5 active:bg-white/10 active:text-white transition-colors"
   717→              >
   718→                {genre}
   719→              </button>
   720→            ))}
   721→          </div>
   722→        </div>
   723→      )}
   724→
   725→      {/* ── Content ── */}
   726→      <div className="px-4">
   727→        {isSearching ? (
   728→          <>
   729→            {searchLoading ? (
   730→              <TrackGridSkeleton />
   731→            ) : searchResults.length > 0 ? (
   732→              <>
   733→                <SectionHeader
   734→                  icon={<Search className="w-4 h-4" />}
   735→                  title={searchQuery}
   736→                />
   737→                <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6 gap-3 md:gap-4 mb-6">
   738→                  {searchResults.map((track, i) => (
   739→                    <TrackCard
   740→                      key={track.videoId}
   741→                      track={track}
   742→                      onPlay={(t) => handlePlay(t, searchResults)}
   743→                      isActive={currentTrack?.videoId === track.videoId}
   744→                      index={i}
   745→                    />
   746→                  ))}
   747→                </div>
   748→              </>
   749→            ) : (
   750→              <div className="text-center py-20">
   751→                <Music className="w-10 h-10 text-white/10 mx-auto mb-3" />
   752→                <p className="text-white/30 text-sm">No results found</p>
   753→              </div>
   754→            )}
   755→
   756→            {/* Related */}
   757→            {relatedResults.length > 0 && (
   758→              <>
   759→                <SectionHeader
   760→                  icon={<TrendingUp className="w-4 h-4" />}
   761→                  title="You Might Also Like"
   762→                />
   763→                <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6 gap-3 md:gap-4 mb-6">
   764→                  {relatedResults.map((track, i) => (
   765→                    <TrackCard
   766→                      key={track.videoId}
   767→                      track={track}
   768→                      onPlay={(t) => handlePlay(t, relatedResults)}
   769→                      isActive={currentTrack?.videoId === track.videoId}
   770→                      index={i}
   771→                    />
   772→                  ))}
   773→                </div>
   774→              </>
   775→            )}
   776→          </>
   777→        ) : (
   778→          <>
   779→            {loading ? (
   780→              <>
   781→                {/* Mood skeleton */}
   782→                <div className="flex gap-3 overflow-x-auto pb-2 -mx-4 px-4 mb-6">
   783→                  {Array.from({ length: 4 }).map((_, i) => (
   784→                    <MoodCardSkeleton key={i} />
   785→                  ))}
   786→                </div>
   787→                <TrackGridSkeleton />
   788→              </>
   789→            ) : (
   790→              <>
   791→                <SectionHeader
   792→                  icon={<TrendingUp className="w-4 h-4" />}
   793→                  title="Trending Now"
   794→                />
   795→                <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6 gap-3 md:gap-4">
   796→                  {trending.map((track, i) => (
   797→                    <TrackCard
   798→                      key={track.videoId}
   799→                      track={track}
   800→                      onPlay={(t) => handlePlay(t, trending)}
   801→                      isActive={currentTrack?.videoId === track.videoId}
   802→                      index={i}
   803→                    />
   804→                  ))}
   805→                </div>
   806→              </>
   807→            )}
   808→          </>
   809→        )}
   810→      </div>
   811→
   812→      {/* Footer */}
   813→      <div className="mt-8 border-t border-white/5 pt-4 pb-2 px-4 text-center">
   814→        <p className="text-[10px] text-white/15">
   815→          StreamVault Music — Powered by YouTube
   816→        </p>
   817→      </div>
   818→
   819→      {/* ── Mini Player ── */}
   820→      <AnimatePresence>
   821→        {currentTrack && !showFullScreen && (
   822→          <MiniPlayer
   823→            key={currentTrack.videoId}
   824→            track={currentTrack}
   825→            onExpand={() => setShowFullScreen(true)}
   826→            onNext={handleNext}
   827→            onPrev={handlePrev}
   828→          />
   829→        )}
   830→      </AnimatePresence>
   831→
   832→      {/* ── Full Screen Player ── */}
   833→      <AnimatePresence>
   834→        {currentTrack && showFullScreen && (
   835→          <FullScreenPlayer
   836→            key={currentTrack.videoId}
   837→            track={currentTrack}
   838→            onClose={handleClosePlayer}
   839→            onNext={handleNext}
   840→            onPrev={handlePrev}
   841→          />
   842→        )}
   843→      </AnimatePresence>
   844→    </div>
   845→  );
   846→}
