    44→    44→function formatDuration(iso: string): string {
    45→    45→  const m = iso.match(/PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?/);
    46→    46→  if (!m) return '0:00';
    47→    47→  const h = parseInt(m[1] || '0');
    48→    48→  const min = parseInt(m[2] || '0');
    49→    49→  const s = parseInt(m[3] || '0');
    50→    50→  if (h > 0) return `${h}:${min.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    51→    51→  return `${min}:${s.toString().padStart(2, '0')}`;
    52→    52→}
    53→    53→
    54→    54→
    55→    55→function formatDurationSeconds(iso: string): number {
    56→    56→  const m = iso.match(/PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?/);
    57→    57→  if (!m) return 0;
    58→    58→  const h = parseInt(m[1] || '0');
    59→    59→  const min = parseInt(m[2] || '0');
    60→    60→  const s = parseInt(m[3] || '0');
    61→    61→  return h * 3600 + min * 60 + s;
    62→    62→}
    63→    63→
    64→    64→function formatTime(sec: number): string {
    65→    65→  const m = Math.floor(sec / 60);
    66→    66→  const s = Math.floor(sec % 60);
    67→    67→  return `${m}:${s.toString().padStart(2, '0')}`;
    68→    68→}

   448→   448→
   449→   449→function MiniPlayer({
   450→   450→  track,
   451→   451→  onExpand,
   452→   452→  onNext,
   453→   453→  onPrev,
   454→   454→}: {
   455→   455→  track: YoutubeVideoResult;
   456→   456→  onExpand: () => void;
   457→   457→  onNext: () => void;
   458→   458→  onPrev: () => void;
   459→   459→}) {
   460→   460→  const [progress, setProgress] = useState(0);
   461→   461→  const totalSec = formatDurationSeconds(track.duration);
   462→   462→
   463→   463→  useEffect(() => {
   464→   464→    const interval = setInterval(() => {
   465→   465→      setProgress((p) => {
   466→   466→        if (p >= 100) return 100;
   467→   467→        return p + (100 / Math.max(totalSec, 1)) * 0.5;
   468→   468→      });
   469→   469→    }, 500);
   470→   470→    return () => clearInterval(interval);
   471→   471→  }, [totalSec]);
   472→   472→
   473→   473→  return (
   474→   474→    <motion.div
   475→   475→      initial={{ y: 80, opacity: 0 }}
   476→   476→      animate={{ y: 0, opacity: 1 }}
   477→   477→      exit={{ y: 80, opacity: 0 }}
   478→   478→      transition={{ type: 'spring', stiffness: 300, damping: 30 }}
   479→   479→      className="fixed left-0 right-0 z-40"
   480→   480→      style={{
   481→   481→        bottom: 'env(safe-area-inset-bottom, 0px)',
   482→   482→      }}
   483→   483→    >
   484→   484→      {/* Progress line */}
   485→   485→      <div className="h-[2px] w-full bg-white/5">
   486→   486→        <motion.div
   487→   487→          className="h-full bg-amber-400"
   488→   488→          style={{ width: `${Math.min(progress, 100)}%` }}
   489→   489→        />
   490→   490→      </div>
   491→   491→
   492→   492→      {/* Bar */}
   493→   493→      <button
   494→   494→        onClick={onExpand}
   495→   495→        className="w-full bg-zinc-900/95 backdrop-blur-xl border-t border-white/10 flex items-center gap-3 px-3 py-2.5"
   496→   496→      >
   497→   497→        {/* Thumbnail */}
   498→   498→        <div className="w-10 h-10 rounded-lg overflow-hidden flex-shrink-0 bg-white/10 shadow-lg">
   499→   499→          <img src={track.thumbnail} alt={track.title} className="w-full h-full object-cover" />
   500→   500→        </div>
   501→   501→
   502→   502→        {/* Title */}
   503→   503→        <div className="flex-1 min-w-0 text-left">
   504→   504→          <p className="text-[13px] font-medium text-white truncate leading-tight">{track.title}</p>
   505→   505→          <p className="text-[11px] text-white/40 truncate mt-0.5">{track.channelTitle}</p>
   506→   506→        </div>
   507→   507→
   508→   508→        {/* Controls */}
   509→   509→        <div className="flex items-center gap-0.5 shrink-0">
   510→   510→          <div
   511→   511→            role="button"
   512→   512→            onClick={(e) => { e.stopPropagation(); onPrev(); }}
   513→   513→            className="p-2 text-white/60 active:text-white"
   514→   514→          >
   515→   515→            <SkipBack className="w-4 h-4" />
   516→   516→          </div>
   517→   517→          <div className="w-8 h-8 rounded-full bg-white flex items-center justify-center">
   518→   518→            <Pause className="w-3.5 h-3.5 text-black fill-black" />
   519→   519→          </div>
   520→   520→          <div
   521→   521→            role="button"
   522→   522→            onClick={(e) => { e.stopPropagation(); onNext(); }}
   523→   523→            className="p-2 text-white/60 active:text-white"
   524→   524→          >
   525→   525→            <SkipForward className="w-4 h-4" />
   526→   526→          </div>
   527→   527→        </div>
   528→   528→      </button>
   529→   529→    </motion.div>
   530→   530→  );
   531→   531→}
