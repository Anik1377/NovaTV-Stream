     1→'use client';
     2→
     3→import { useState, useRef, useEffect, useCallback, useMemo } from 'react';
     4→import {
     5→  X,
     6→  Maximize2,
     7→  Minimize2,
     8→  RefreshCw,
     9→  ChevronLeft,
    10→  ShieldCheck,
    11→  Server,
    12→  Check,
    13→  Loader2,
    14→  ArrowRight,
    15→} from 'lucide-react';
    16→import { motion, AnimatePresence, useMotionValue, useTransform, animate } from 'framer-motion';
    17→import { providers, getProvider, getEmbedUrl, type Provider } from '@/lib/providers';
    18→import { getRankedProviderIds, getServerScore, recordServerPick } from '@/lib/server-rankings';
    19→import { useAppStore } from '@/store/app-store';
    20→import { useAuthStore } from '@/store/auth-store';
    21→import { useIsIOS } from '@/hooks/use-ios';
    22→import { useIsMobile } from '@/hooks/use-mobile';
    23→import { recordWatchHistory } from '@/lib/watch-history';
    24→
    25→interface VideoPlayerProps {
    26→  src: string;
    27→  title?: string;
    28→  onClose: () => void;
    29→  mediaType: 'movie' | 'tv';
    30→  tmdbId: number;
    31→  season?: number;
    32→  episode?: number;
    33→}
    34→
    35→/* ------------------------------------------------------------------ */
    36→/*  iOS native fullscreen detection                                    */
    37→/* ------------------------------------------------------------------ */
    38→function useIOSFullscreenDetect(onReturn: () => void) {
    39→  useEffect(() => {
    40→    const handleEnd = () => onReturn();
    41→    document.addEventListener('webkitendfullscreen', handleEnd);
    42→    return () => document.removeEventListener('webkitendfullscreen', handleEnd);
    43→  }, [onReturn]);
    44→}
    45→
    46→/* ------------------------------------------------------------------ */
    47→/*  Score badge helper                                                 */
    48→/* ------------------------------------------------------------------ */
    49→function scoreBadge(score: number) {
    50→  if (score >= 80) return { label: 'Top Pick', cls: 'bg-emerald-500/20 text-emerald-400' };
    51→  if (score >= 40) return { label: 'Popular', cls: 'bg-amber-500/20 text-amber-400' };
    52→  if (score > 0) return { label: 'Used', cls: 'bg-zinc-500/20 text-zinc-400' };
    53→  return null;
    54→}
    55→
    56→/* ------------------------------------------------------------------ */
    57→/*  Main component                                                     */
    58→/* ------------------------------------------------------------------ */
    59→export function VideoPlayer({
    60→  src,
    61→  title,
    62→  onClose,
    63→  mediaType,
    64→  tmdbId,
    65→  season,
    66→  episode,
    67→}: VideoPlayerProps) {
    68→  const [loading, setLoading] = useState(true);
    69→  const [error, setError] = useState(false);
    70→  const [isFullscreen, setIsFullscreen] = useState(false);
    71→  const [mobileSheetOpen, setMobileSheetOpen] = useState(false);
    72→  const [currentSrc, setCurrentSrc] = useState(src);
    73→  const [switchingTo, setSwitchingTo] = useState<string | null>(null);
    74→  const loadTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    75→  const dismissTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    76→
    77→  const containerRef = useRef<HTMLDivElement>(null);
    78→  const iframeRef = useRef<HTMLIFrameElement>(null);
    79→  const { selectedProvider, setSelectedProvider, selectedMovie, selectedTv } = useAppStore();
    80→  const user = useAuthStore((s) => s.user);
    81→  const isIOS = useIsIOS();
    82→  const isMobile = useIsMobile();
    83→
    84→  // Swipe-to-dismiss (mobile only)
    85→  const dragY = useMotionValue(0);
    86→  const iframeScale = useTransform(dragY, [0, 250], [1, 0.92]);
    87→  const iframeRadius = useTransform(dragY, [0, 250], [8, 24]);
    88→
    89→  /* ---------------------------------------------------------------- */
    90→  /*  Ranked provider list                                              */
    91→  /* ---------------------------------------------------------------- */
    92→  const rankedIds = useMemo(() => getRankedProviderIds(), []);
    93→  const rankedSet = useMemo(() => new Set(rankedIds), [rankedIds]);
    94→
    95→  const sortedProviders = useMemo(() => {
    96→    const ranked = rankedIds
    97→      .map((id) => providers.find((p) => p.id === id))
    98→      .filter((p): p is Provider => !!p);
    99→    const unranked = providers.filter((p) => !rankedSet.has(p.id));
   100→    return [...ranked, ...unranked];
   101→  }, [rankedIds, rankedSet]);
   102→
   103→  const activeProvider = getProvider(selectedProvider);
   104→
   105→  /* ---------------------------------------------------------------- */
   106→  /*  Drag end handler                                                  */
   107→  /* ---------------------------------------------------------------- */
   108→  const handleDragEnd = useCallback(
   109→    (_: unknown, info: { offset: { y: number }; velocity: { y: number } }) => {
   110→      const shouldClose = info.offset.y > 120 || info.velocity.y > 500;
   111→      if (shouldClose) {
   112→        animate(dragY, 600, {
   113→          duration: 0.25,
   114→          ease: 'easeOut',
   115→          onComplete: onClose,
   116→        });
   117→      } else {
   118→        animate(dragY, 0, { type: 'spring', stiffness: 400, damping: 30 });
   119→      }
   120→    },
   121→    [dragY, onClose],
   122→  );
   123→
   124→  /* ---------------------------------------------------------------- */
   125→  /*  Record watch history                                              */
   126→  /* ---------------------------------------------------------------- */
   127→  useEffect(() => {
   128→    if (!title) return;
   129→    const item = mediaType === 'tv' ? selectedTv : selectedMovie;
   130→    recordWatchHistory({
   131→      tmdbId,
   132→      title,
   133→      posterPath: item?.poster_path || null,
   134→      backdropPath: item?.backdrop_path || null,
   135→      mediaType,
   136→      season: season ?? null,
   137→      episode: episode ?? null,
   138→    });
   139→    if (user) {
   140→      fetch('/api/profile/history', {
   141→        method: 'POST',
   142→        headers: { 'Content-Type': 'application/json' },
   143→        body: JSON.stringify({
   144→          tmdbId,
   145→          title,
   146→          posterPath: item?.poster_path || null,
   147→          mediaType,
   148→          season: season ?? null,
   149→          episode: episode ?? null,
   150→        }),
   151→      }).catch(() => {});
   152→    }
   153→  }, [tmdbId, mediaType, season, episode, title, user, selectedMovie, selectedTv]);
   154→
   155→  /* ---------------------------------------------------------------- */
   156→  /*  Loading timeout — mobile iframes may never fire onLoad            */
   157→  /* ---------------------------------------------------------------- */
   158→  useEffect(() => {
   159→    // Clear any previous timeouts
   160→    if (loadTimeoutRef.current) clearTimeout(loadTimeoutRef.current);
   161→    if (dismissTimeoutRef.current) clearTimeout(dismissTimeoutRef.current);
   162→
   163→    // On mobile, auto-dismiss loading overlay after 6s so user can interact
   164→    // Desktop gives more time since onLoad usually fires reliably
   165→    const dismissDelay = isMobile ? 6000 : 12000;
   166→
   167→    dismissTimeoutRef.current = setTimeout(() => {
   168→      setLoading(false);
   169→    }, dismissDelay);
   170→
   171→    // On mobile, if still loading after 15s, show error with next server option
   172→    if (isMobile) {
   173→      loadTimeoutRef.current = setTimeout(() => {
   174→        setLoading(false);
   175→        setError(true);
   176→      }, 15000);
   177→    }
   178→
   179→    return () => {
   180→      if (loadTimeoutRef.current) clearTimeout(loadTimeoutRef.current);
   181→      if (dismissTimeoutRef.current) clearTimeout(dismissTimeoutRef.current);
   182→    };
   183→  }, [currentSrc, isMobile]);
   184→
   185→  /* ---------------------------------------------------------------- */
   186→  /*  Body scroll lock                                                  */
   187→  /* ---------------------------------------------------------------- */
   188→  useEffect(() => {
   189→    const scrollY = window.scrollY;
   190→    document.body.style.setProperty('--player-scroll-y', `${scrollY}px`);
   191→    document.body.classList.add('player-open');
   192→    document.body.style.top = `-${scrollY}px`;
   193→    return () => {
   194→      document.body.classList.remove('player-open');
   195→      document.body.style.top = '';
   196→      window.scrollTo(0, scrollY);
   197→    };
   198→  }, []);
   199→
   200→  /* ---------------------------------------------------------------- */
   201→  /*  iOS native fullscreen "Done" button                               */
   202→  /* ---------------------------------------------------------------- */
   203→  useIOSFullscreenDetect(onClose);
   204→
   205→  /* ---------------------------------------------------------------- */
   206→  /*  Auto-try next server on error                                     */
   207→  /* ---------------------------------------------------------------- */
   208→  const tryNextServer = () => {
   209→    const currentIdx = sortedProviders.findIndex(p => p.id === selectedProvider);
   210→    const next = sortedProviders[(currentIdx + 1) % sortedProviders.length];
   211→    if (next) {
   212→      switchProvider(next.id);
   213→    }
   214→  };
   215→
   216→  /* ---------------------------------------------------------------- */
   217→  /*  Switch provider                                                   */
   218→  /* ---------------------------------------------------------------- */
   219→  const switchProvider = useCallback(
   220→    (providerId: string) => {
   221→      setSwitchingTo(providerId);
   222→      recordServerPick(providerId);
   223→      setSelectedProvider(providerId);
   224→      const newSrc = getEmbedUrl(providerId, mediaType, tmdbId, season, episode);
   225→      setCurrentSrc(newSrc);
   226→      setLoading(true);
   227→      setError(false);
   228→      setMobileSheetOpen(false);
   229→
   230→      const item = mediaType === 'tv' ? selectedTv : selectedMovie;
   231→      recordWatchHistory({
   232→        tmdbId,
   233→        title: title || 'Unknown',
   234→        posterPath: item?.poster_path || null,
   235→        backdropPath: item?.backdrop_path || null,
   236→        mediaType,
   237→        season: season ?? null,
   238→        episode: episode ?? null,
   239→      });
   240→
   241→      setTimeout(() => setSwitchingTo(null), 3000);
   242→    },
   243→    [mediaType, tmdbId, season, episode, setSelectedProvider, selectedTv, selectedMovie, title],
   244→  );
   245→
   246→  /* ---------------------------------------------------------------- */
   247→  /*  Fullscreen toggle                                                 */
   248→  /* ---------------------------------------------------------------- */
   249→  const toggleFullscreen = async () => {
   250→    if (!containerRef.current) return;
   251→    try {
   252→      if (!document.fullscreenElement) {
   253→        await containerRef.current.requestFullscreen();
   254→        setIsFullscreen(true);
   255→      } else {
   256→        await document.exitFullscreen();
   257→        setIsFullscreen(false);
   258→      }
   259→    } catch {
   260→      // Fullscreen API unavailable (e.g. iOS)
   261→    }
   262→  };
   263→
   264→  useEffect(() => {
   265→    const handler = () => setIsFullscreen(!!document.fullscreenElement);
   266→    document.addEventListener('fullscreenchange', handler);
   267→    return () => document.removeEventListener('fullscreenchange', handler);
   268→  }, []);
   269→
   270→  /* ---------------------------------------------------------------- */
   271→  /*  Keyboard shortcuts                                                */
   272→  /* ---------------------------------------------------------------- */
   273→  useEffect(() => {
   274→    const handleKey = (e: KeyboardEvent) => {
   275→      if (e.key === 'Escape' && !mobileSheetOpen) onClose();
   276→    };
   277→    window.addEventListener('keydown', handleKey);
   278→    return () => window.removeEventListener('keydown', handleKey);
   279→  }, [onClose, mobileSheetOpen]);
   280→
   281→  /* ---------------------------------------------------------------- */
   282→  /*  Active server score / badge                                       */
   283→  /* ---------------------------------------------------------------- */
   284→  const activeScore = useMemo(() => getServerScore(selectedProvider), [selectedProvider]);
   285→  const activeBadge = useMemo(() => scoreBadge(activeScore), [activeScore]);
   286→  const isTopRanked = rankedIds[0] === selectedProvider;
   287→
   288→  /* ---------------------------------------------------------------- */
   289→  /*  Mobile bottom sheet                                               */
   290→  /* ---------------------------------------------------------------- */
   291→  const mobileSheet = (
   292→    <AnimatePresence>
   293→      {mobileSheetOpen && (
   294→        <>
   295→          {/* Backdrop */}
   296→          <motion.div
   297→            initial={{ opacity: 0 }}
   298→            animate={{ opacity: 1 }}
   299→            exit={{ opacity: 0 }}
   300→            className="absolute inset-0 bg-black/60 z-20"
   301→            onClick={() => setMobileSheetOpen(false)}
   302→          />
   303→          {/* Sheet */}
   304→          <motion.div
   305→            initial={{ y: '100%' }}
   306→            animate={{ y: 0 }}
   307→            exit={{ y: '100%' }}
   308→            transition={{ type: 'spring', stiffness: 300, damping: 30 }}
   309→            className="absolute bottom-0 left-0 right-0 max-h-[70vh] bg-[#111] rounded-t-2xl z-30 flex flex-col"
   310→            style={{ paddingBottom: isIOS ? 'env(safe-area-inset-bottom, 16px)' : 16 }}
   311→          >
   312→            {/* Drag handle */}
   313→            <div className="flex justify-center pt-3 pb-2">
   314→              <div className="w-10 h-1 rounded-full bg-white/20" />
   315→            </div>
   316→
   317→            {/* Header */}
   318→            <div className="flex items-center justify-between px-5 pb-3">
   319→              <div className="flex items-center gap-2">
   320→                <Server className="w-4 h-4 text-white/50" />
   321→                <h4 className="text-white/90 text-sm font-semibold">Servers</h4>
   322→              </div>
   323→              <button
   324→                onClick={() => setMobileSheetOpen(false)}
   325→                className="w-7 h-7 rounded-lg flex items-center justify-center text-white/40 hover:text-white/80 hover:bg-white/[0.06] transition-colors"
   326→                aria-label="Close server sheet"
   327→              >
   328→                <X className="w-3.5 h-3.5" />
   329→              </button>
   330→            </div>
   331→
   332→            {/* Server list */}
   333→            <div className="flex-1 overflow-y-auto content-scroll px-3 pb-2">
   334→              {sortedProviders.map((p) => {
   335→                const isActive = p.id === selectedProvider;
   336→                const score = getServerScore(p.id);
   337→                const badge = scoreBadge(score);
   338→                const isTop = rankedIds[0] === p.id;
   339→                const isSwitching = switchingTo === p.id;
   340→
   341→                return (
   342→                  <button
   343→                    key={p.id}
   344→                    onClick={() => switchProvider(p.id)}
   345→                    disabled={isSwitching}
   346→                    className={
   347→                      'w-full flex items-center gap-3 px-3 py-3 rounded-xl text-left transition-colors ' +
   348→                      (isActive ? 'bg-white/[0.06]' : 'hover:bg-white/[0.03]')
   349→                    }
   350→                  >
   351→                    <div
   352→                      className="w-9 h-9 rounded-lg flex items-center justify-center text-sm font-bold shrink-0"
   353→                      style={{ backgroundColor: p.color + '18', color: p.color }}
   354→                    >
   355→                      {p.icon}
   356→                    </div>
   357→                    <div className="flex-1 min-w-0">
   358→                      <div className="flex items-center gap-1.5">
   359→                        <span className="text-white/90 text-sm font-medium truncate">{p.name}</span>
   360→                        {isTop && <ShieldCheck className="w-3.5 h-3.5 text-emerald-400 shrink-0" />}
   361→                      </div>
   362→                      <p className="text-white/40 text-[11px] truncate mt-0.5">{p.description}</p>
   363→                    </div>
   364→                    <div className="flex flex-col items-end gap-1 shrink-0">
   365→                      {badge && (
   366→                        <span className={'px-1.5 py-0.5 rounded text-[9px] font-semibold ' + badge.cls}>
   367→                          {badge.label}
   368→                        </span>
   369→                      )}
   370→                      {isSwitching ? (
   371→                        <Loader2 className="w-4 h-4 animate-spin text-white/40" />
   372→                      ) : isActive ? (
   373→                        <Check className="w-4 h-4" style={{ color: p.color }} />
   374→                      ) : null}
   375→                    </div>
   376→                  </button>
   377→                );
   378→              })}
   379→            </div>
   380→          </motion.div>
   381→        </>
   382→      )}
   383→    </AnimatePresence>
   384→  );
   385→
   386→  /* ================================================================== */
   387→  /*  RENDER                                                            */
   388→  /* ================================================================== */
   389→  return (
   390→    <AnimatePresence>
   391→      <motion.div
   392→        initial={{ opacity: 0, y: 40 }}
   393→        animate={{ opacity: 1, y: 0 }}
   394→        exit={{ opacity: 0, y: 60 }}
   395→        transition={{ type: 'spring', stiffness: 300, damping: 30, mass: 0.8 }}
   396→        style={{ y: dragY }}
   397→        drag={isMobile ? 'y' : false}
   398→        dragConstraints={{ top: 0, bottom: 0 }}
   399→        dragElastic={{ top: 0.1, bottom: 0.4 }}
   400→        onDragEnd={handleDragEnd}
   401→        className="fixed inset-0 z-[100] bg-black/95 backdrop-blur-md flex flex-col ios-player-container"
   402→      >
   403→        {/* iOS: translucent status bar spacer */}
   404→        <div
   405→          className="shrink-0 w-full"
   406→          style={{ height: isIOS ? 'env(safe-area-inset-top, 0px)' : 0 }}
   407→        />
   408→
   409→        {/* ==================== TOP BAR ==================== */}
   410→        <div
   411→          className="flex items-center gap-3 px-3 md:px-5 h-12 shrink-0 z-10 justify-between"
   412→          style={{
   413→            paddingTop: isIOS
   414→              ? 'max(2px, env(safe-area-inset-top, 0px) - 10px)'
   415→              : 0,
   416→          }}
   417→        >
   418→          {/* Left: iOS Done button or empty spacer */}
   419→          {isIOS ? (
   420→            <button
   421→              onClick={onClose}
   422→              className="flex items-center gap-1 px-3 py-1.5 rounded-full text-white/80 hover:text-white text-sm font-semibold transition-colors active:scale-95"
   423→              style={{
   424→                background: 'rgba(255,255,255,0.12)',
   425→                backdropFilter: 'blur(20px)',
   426→                WebkitBackdropFilter: 'blur(20px)',
   427→              }}
   428→              aria-label="Done"
   429→            >
   430→              <ChevronLeft className="w-4 h-4" strokeWidth={2.5} />
   431→              <span>Done</span>
   432→            </button>
   433→          ) : (
   434→            <div className="w-16" />
   435→          )}
   436→
   437→          {/* Center: title + active server pill */}
   438→          <div className="flex items-center gap-2 min-w-0 flex-1 justify-center">
   439→            <h3 className="text-white/70 font-medium text-sm truncate">
   440→              {title || 'Now Playing'}
   441→            </h3>
   442→            <span
   443→              className="shrink-0 inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-[10px] font-semibold"
   444→              style={{
   445→                backgroundColor: activeProvider.color + '20',
   446→                color: activeProvider.color,
   447→              }}
   448→            >
   449→              {isTopRanked && <ShieldCheck className="w-3 h-3" />}
   450→              {activeProvider.name}
   451→            </span>
   452→          </div>
   453→
   454→          {/* Right: fullscreen + close (desktop) */}
   455→          <div className="flex items-center gap-1 shrink-0">
   456→            {!isIOS && (
   457→              <button
   458→                onClick={toggleFullscreen}
   459→                className="w-9 h-9 rounded-full flex items-center justify-center text-white/60 hover:text-white hover:bg-white/10 transition-colors"
   460→                aria-label={isFullscreen ? 'Exit fullscreen' : 'Enter fullscreen'}
   461→              >
   462→                {isFullscreen ? <Minimize2 className="w-4 h-4" /> : <Maximize2 className="w-4 h-4" />}
   463→              </button>
   464→            )}
   465→            {!isIOS && (
   466→              <button
   467→                onClick={onClose}
   468→                className="w-9 h-9 rounded-full flex items-center justify-center text-white/60 hover:text-white hover:bg-white/10 transition-colors"
   469→                aria-label="Close player"
   470→              >
   471→                <X className="w-5 h-5" />
   472→              </button>
   473→            )}
   474→          </div>
   475→        </div>
   476→
   477→        {/* iOS swipe hint */}
   478→        {isIOS && (
   479→          <motion.div
   480→            initial={{ opacity: 1, y: 0 }}
   481→            animate={{ opacity: 0, y: -10 }}
   482→            transition={{ delay: 2.5, duration: 1 }}
   483→            className="flex justify-center pb-1"
   484→          >
   485→            <div className="flex items-center gap-1.5 px-3 py-1 rounded-full bg-white/[0.08]">
   486→              <div className="w-1 h-1 rounded-full bg-white/30" />
   487→              <span className="text-white/50 text-[10px] font-medium">Swipe down to close</span>
   488→              <div className="w-1 h-1 rounded-full bg-white/30" />
   489→            </div>
   490→          </motion.div>
   491→        )}
   492→
   493→        {/* ==================== VIDEO AREA ==================== */}
   494→        <div
   495→          className="flex-1 flex items-center justify-center p-3 md:p-8 relative overflow-hidden"
   496→        >
   497→          <motion.div
   498→            ref={containerRef}
   499→            style={{
   500→              scale: isMobile ? iframeScale : 1,
   501→              borderRadius: isMobile ? iframeRadius : 8,
   502→            }}
   503→            className={
   504→              'relative bg-black overflow-hidden shadow-2xl shadow-black/80 ' +
   505→              (!isMobile ? 'w-full max-w-6xl aspect-video' : 'w-full h-full')
   506→            }
   507→          >
   508→            {/* Loading state: animated pulsing ring — pointer-events-none so user can tap iframe */}
   509→            <AnimatePresence>
   510→              {loading && !error && (
   511→                <motion.div
   512→                  key="loading"
   513→                  initial={{ opacity: 0 }}
   514→                  animate={{ opacity: 1 }}
   515→                  exit={{ opacity: 0 }}
   516→                  className="absolute inset-0 flex flex-col items-center justify-center z-10 bg-black/80 gap-4 pointer-events-none"
   517→                >
   518→                  <div className="relative w-16 h-16 flex items-center justify-center">
   519→                    <div
   520→                      className="absolute inset-0 rounded-full animate-ping opacity-30"
   521→                      style={{ backgroundColor: activeProvider.color }}
   522→                    />
   523→                    <div
   524→                      className="absolute inset-1 rounded-full animate-pulse opacity-60"
   525→                      style={{ backgroundColor: activeProvider.color }}
   526→                    />
   527→                    <div
   528→                      className="relative w-10 h-10 rounded-full flex items-center justify-center"
   529→                      style={{ backgroundColor: activeProvider.color + '25' }}
   530→                    >
   531→                      <Loader2
   532→                        className="w-5 h-5 animate-spin"
   533→                        style={{ color: activeProvider.color }}
   534→                      />
   535→                    </div>
   536→                  </div>
   537→                  <p className="text-white/50 text-xs font-medium">
   538→                    Connecting to {activeProvider.name}...
   539→                  </p>
   540→                  {isMobile && (
   541→                    <p className="text-white/30 text-[10px]">
   542→                      Tap the player if video loads
   543→                    </p>
   544→                  )}
   545→                </motion.div>
   546→              )}
   547→            </AnimatePresence>
   548→
   549→            {/* Error state */}
   550→            {error && (
   551→              <div className="absolute inset-0 flex flex-col items-center justify-center gap-4 z-10 bg-black">
   552→                <div className="w-14 h-14 rounded-2xl bg-white/[0.06] flex items-center justify-center">
   553→                  <RefreshCw className="w-6 h-6 text-white/30" />
   554→                </div>
   555→                <div className="text-center">
   556→                  <p className="text-white/50 text-sm font-medium">Unable to load from <span style={{ color: activeProvider.color }}>{activeProvider.name}</span></p>
   557→                  <p className="text-white/25 text-xs mt-1">This server may not have this title</p>
   558→                </div>
   559→                <div className="flex items-center gap-3">
   560→                  <button
   561→                    onClick={() => {
   562→                      setError(false);
