    17→    17→} from '@/lib/avatars';

    23→    23→

    39→    39→
    40→    40→export function ProfilePage() {
    41→    41→  const { user, loading: authLoading, updateProfile, logout } = useAuthStore();
    42→    42→  const goHome = useAppStore(s => s.goHome);
    43→    43→  const watchlist = useAppStore(s => s.watchlist);
    44→    44→  const selectMovie = useAppStore(s => s.selectMovie);
    45→    45→  const selectTv = useAppStore(s => s.selectTv);
    46→    46→  const selectPerson = useAppStore(s => s.selectPerson);
    47→    47→
    48→    48→  const [tab, setTab] = useState<Tab>('profile');
    49→    49→  const [authModalOpen, setAuthModalOpen] = useState(false);
    50→    50→
    51→    51→  // Profile editing
    52→    52→  const [editing, setEditing] = useState(false);
    53→    53→  const [editStep, setEditStep] = useState<EditStep>('main');
    54→    54→  const [editName, setEditName] = useState('');
    55→    55→  const [editBio, setEditBio] = useState('');
    56→    56→  const [selectedAvatar, setSelectedAvatar] = useState('hero');
    57→    57→  const [selectedColor, setSelectedColor] = useState('#e11d48');
    58→    58→  const [selectedGenres, setSelectedGenres] = useState<string[]>([]);
    59→    59→  const [saving, setSaving] = useState(false);
    60→    60→  const [saveError, setSaveError] = useState('');
    61→    61→
    62→    62→  // Browse history
    63→    63→  const [history, setHistory] = useState<BrowseHistoryItem[]>([]);
    64→    64→  const [historyLoading, setHistoryLoading] = useState(false);
    65→    65→  const [historyFilter, setHistoryFilter] = useState<HistoryFilter>('all');
    66→    66→  const [historyPage, setHistoryPage] = useState(1);
    67→    67→  const [historyTotal, setHistoryTotal] = useState(0);
    68→    68→  const [historyLoadMore, setHistoryLoadMore] = useState(false);
    69→    69→  const [deletingId, setDeletingId] = useState<string | null>(null);
    70→    70→
    71→    71→  // Profile stats from API
    72→    72→  const [profileStats, setProfileStats] = useState<{ watchHistoryCount: number } | null>(null);
    73→    73→
    74→    74→  // Computed accent color for UI
    75→    75→  const accentColor = useMemo(() => {

    77→    77→    const def = getAvatarDef(user?.avatar);

    80→    80→
    81→    81→  const fetchProfile = useCallback(async () => {
    82→    82→    if (!user) return;
    83→    83→    try {
    84→    84→      const res = await fetch('/api/profile');
    85→    85→      const data = await res.json();
    86→    86→      if (res.ok) setProfileStats(data.stats);
    87→    87→    } catch { /* ignore */ }
    88→    88→  }, [user]);
    89→    89→
    90→    90→  // Watch history from localStorage
    91→    91→  const [watchItems, setWatchItems] = useState<WatchHistoryEntry[]>([]);
    92→    92→
    93→    93→  const fetchHistory = useCallback(async (filter: HistoryFilter, page: number, append: boolean) => {
    94→    94→    // Always load watch history from localStorage
    95→    95→    let wh = getWatchHistory();
    96→    96→    if (filter !== 'all') wh = wh.filter((i) => i.mediaType === filter);
    97→    97→    setWatchItems(wh);
    98→    98→
    99→    99→    // Non-authenticated users: use localStorage for browse history
   100→   100→    if (!user) {
   101→   101→      setHistoryLoading(true);
   102→   102→      let local = getLocalBrowseHistory();
   103→   103→      if (filter !== 'all') local = local.filter((i) => i.mediaType === filter);
   104→   104→      setHistory(local);
   105→   105→      setHistoryTotal(local.length + wh.length);
   106→   106→      setHistoryLoading(false);
   107→   107→      return;
   108→   108→    }
   109→   109→    // Authenticated: use server API for browse history
   110→   110→    if (append) { setHistoryLoadMore(true); } else { setHistoryLoading(true); }
   111→   111→    try {
   112→   112→      const params = new URLSearchParams({ limit: '50', offset: String((page - 1) * 50) });
   113→   113→      if (filter !== 'all') params.set('type', filter);
   114→   114→      const res = await fetch(`/api/history?${params}`);
   115→   115→      const data = await res.json();
   116→   116→      if (res.ok) {
   117→   117→        setHistory(prev => append ? [...prev, ...data.items] : data.items);
   118→   118→        setHistoryTotal(data.total + wh.length);
   119→   119→        if (!append) setHistoryPage(1);
   120→   120→      }
   121→   121→    } catch { /* ignore */ }
   122→   122→    setHistoryLoading(false);
   123→   123→    setHistoryLoadMore(false);
   124→   124→  }, [user]);
   125→   125→
   126→   126→  useEffect(() => {
   127→   127→    if (tab === 'profile') fetchProfile();
   128→   128→    if (tab === 'history') fetchHistory(historyFilter, 1, false);
   129→   129→  }, [tab, historyFilter, user, fetchHistory, fetchProfile]);
   130→   130→
   131→   131→  // Refresh watch history when a new watch event is recorded
   132→   132→  useEffect(() => {
   133→   133→    const refresh = () => {
   134→   134→      if (tab === 'history') {
   135→   135→        let wh = getWatchHistory();
   136→   136→        if (historyFilter !== 'all') wh = wh.filter((i) => i.mediaType === historyFilter);
   137→   137→        setWatchItems(wh);
   138→   138→      }
   139→   139→    };
   140→   140→    window.addEventListener('watch-history-updated', refresh);
   141→   141→    return () => window.removeEventListener('watch-history-updated', refresh);
   142→   142→  }, [tab, historyFilter]);
   143→   143→
   144→   144→  useEffect(() => {
   145→   145→    window.scrollTo({ top: 0, behavior: 'smooth' });
   146→   146→  }, []);
   147→   147→
   148→   148→  const startEdit = () => {
   149→   149→    setEditName(user?.name || '');
   150→   150→    setEditBio(user?.bio || '');
   151→   151→    setSelectedAvatar(user?.avatar || 'hero');
   152→   152→    setSelectedColor(user?.accentColor || getAvatarDef(user?.avatar).color);
   153→   153→    setSelectedGenres(user?.favoriteGenres || []);
   154→   154→    setEditStep('main');
   155→   155→    setEditing(true);
   156→   156→    setSaveError('');
   157→   157→  };
   158→   158→
   159→   159→  const saveProfile = async () => {
   160→   160→    setSaving(true);
   161→   161→    setSaveError('');
   162→   162→    const result = await updateProfile({
   163→   163→      name: editName,
   164→   164→      bio: editBio,
   165→   165→      avatar: selectedAvatar,
   166→   166→      accentColor: selectedColor,
   167→   167→      favoriteGenres: selectedGenres,
   168→   168→    });
   169→   169→    setSaving(false);
   170→   170→    if (result.error) {
   171→   171→      setSaveError(result.error);
   172→   172→    } else {
   173→   173→      setEditing(false);
   174→   174→    }
   175→   175→  };
   176→   176→
   177→   177→  const toggleGenre = (genre: string) => {
   178→   178→    setSelectedGenres(prev =>
   179→   179→      prev.includes(genre) ? prev.filter(g => g !== genre) : [...prev, genre]
   180→   180→    );
   181→   181→  };
   182→   182→
   183→   183→  const clearHistory = async () => {
   184→   184→    if (!confirm('Clear all history? This cannot be undone.')) return;
   185→   185→    if (user) {
   186→   186→      await fetch('/api/history', { method: 'DELETE' });
   187→   187→    } else {
   188→   188→      clearLocalHistory();
   189→   189→    }
   190→   190→    clearWatchHistory();
   191→   191→    setHistory([]);
   192→   192→    setWatchItems([]);
   193→   193→    setHistoryTotal(0);
   194→   194→  };
   195→   195→
   196→   196→  const handleDeleteHistoryItem = async (id: string, e: React.MouseEvent) => {
   197→   197→    e.stopPropagation();
   198→   198→    setDeletingId(id);
   199→   199→    if (user) {
   200→   200→      await fetch(`/api/history?id=${id}`, { method: 'DELETE' });
   201→   201→    } else {
   202→   202→      deleteLocalHistoryItem(id);
   203→   203→    }
   204→   204→    setHistory(prev => prev.filter(item => item.id !== id));
   205→   205→    setHistoryTotal(prev => prev - 1);
   206→   206→    setDeletingId(null);
   207→   207→  };
   208→   208→
   209→   209→  const handleDeleteWatchItem = (tmdbId: number, mediaType: string, e: React.MouseEvent) => {
   210→   210→    e.stopPropagation();
   211→   211→    removeWatchHistory(tmdbId, mediaType);
   212→   212→    setWatchItems(prev => prev.filter(item => !(item.tmdbId === tmdbId && item.mediaType === mediaType)));
   213→   213→    setHistoryTotal(prev => prev - 1);
   214→   214→  };
   215→   215→
   216→   216→  const handleHistoryClick = (item: BrowseHistoryItem) => {
   217→   217→    if (item.mediaType === 'person') {
   218→   218→      selectPerson({ id: item.tmdbId, name: item.title, profilePath: item.posterPath });
   219→   219→      return;
   220→   220→    }
   221→   221→    const media: Movie = {
   222→   222→      id: item.tmdbId, title: item.title, name: item.title,
   223→   223→      poster_path: item.posterPath, media_type: item.mediaType,
   224→   224→      vote_average: 0, genre_ids: [], overview: '', popularity: 0,
   225→   225→      release_date: '', first_air_date: '', backdrop_path: null, original_language: '',
   226→   226→    };
   227→   227→    if (item.mediaType === 'tv') selectTv(media); else selectMovie(media);
   228→   228→  };
   229→   229→
   230→   230→  const handleWatchItemClick = (item: WatchHistoryEntry) => {
   231→   231→    const media: Movie = {
   232→   232→      id: item.tmdbId, title: item.title, name: item.title,
   233→   233→      poster_path: item.posterPath, media_type: item.mediaType,
   234→   234→      vote_average: 0, genre_ids: [], overview: '', popularity: 0,
   235→   235→      release_date: '', first_air_date: '', backdrop_path: item.backdropPath, original_language: '',
   236→   236→    };
   237→   237→    if (item.mediaType === 'tv') selectTv(media); else selectMovie(media);
   238→   238→  };
   239→   239→
   240→   240→  const handleHistoryFilterChange = (f: HistoryFilter) => {
   241→   241→    if (f === historyFilter) return;
   242→   242→    setHistoryFilter(f);
   243→   243→  };
   244→   244→
   245→   245→  const loadMoreHistory = () => {
   246→   246→    const nextPage = historyPage + 1;
   247→   247→    setHistoryPage(nextPage);
   248→   248→    fetchHistory(historyFilter, nextPage, true);
   249→   249→  };
   250→   250→
   251→   251→  const handleWatchlistClick = (tmdbId: number, mediaType: string) => {
   252→   252→    const item: Movie = {
   253→   253→      id: tmdbId, title: '', name: '', poster_path: null,
   254→   254→      media_type: mediaType as 'movie' | 'tv', vote_average: 0, genre_ids: [],
   255→   255→      overview: '', popularity: 0, release_date: '', first_air_date: '',
   256→   256→      backdrop_path: null, original_language: '',
   257→   257→    };
   258→   258→    if (mediaType === 'tv') selectTv(item); else selectMovie(item);
   259→   259→  };
   260→   260→
   261→   261→  // Not logged in — but still show the page with limited tabs
   262→   262→  const isLoggedIn = !!user && !authLoading;
   263→   263→
   264→   264→  const memberSince = user ? new Date(user.createdAt || Date.now()).toLocaleDateString('en-US', { month: 'long', year: 'numeric' }) : '';
   265→   265→  const avatarDef = getAvatarDef(user?.avatar);
   266→   266→
   267→   267→  // Default to history tab when not logged in
   268→   268→  useEffect(() => {
   269→   269→    if (!isLoggedIn && tab === 'profile') setTab('history');
   270→   270→  }, [isLoggedIn]);
   271→   271→
   272→   272→  const tabsList: { key: Tab; label: string; icon: React.ReactNode }[] = [
   273→   273→    ...(isLoggedIn ? [{ key: 'profile' as Tab, label: 'Profile', icon: <User className="w-4 h-4" /> }] : []),
   274→   274→    ...(isLoggedIn ? [{ key: 'watchlist' as Tab, label: `Watchlist (${watchlist.length})`, icon: <Bookmark className="w-4 h-4" /> }] : []),
   275→   275→    { key: 'history', label: `History${historyTotal > 0 ? ` (${historyTotal})` : ''}`, icon: <Clock className="w-4 h-4" /> },
   276→   276→  ];
   277→   277→
   278→   278→  return (
   279→   279→    <div className="min-h-screen pb-10">
   280→   280→      {/* Back button */}
   281→   281→      <div className="pt-20 px-4 md:px-8">
   282→   282→        <button onClick={goHome} className="flex items-center gap-2 text-white/50 hover:text-white mb-6 transition-colors group">
   283→   283→          <div className="w-8 h-8 rounded-full bg-white/[0.06] flex items-center justify-center group-hover:bg-white/10 transition-colors">
   284→   284→            <ArrowLeft className="w-4 h-4" />
   285→   285→          </div>
   286→   286→          <span className="text-sm font-medium">Back</span>
   287→   287→        </button>
   288→   288→
   289→   289→        {/* ── Profile Header (logged in only) ── */}
   290→   290→        {isLoggedIn && (
   291→   291→        <div className="flex flex-col sm:flex-row items-center sm:items-start gap-6 mb-8">
   292→   292→          {/* Avatar */}
   293→   293→          <motion.div className="relative" initial={{ scale: 0.8, opacity: 0 }} animate={{ scale: 1, opacity: 1 }} transition={{ type: 'spring', stiffness: 300, damping: 20 }}>
   294→   294→            <div className="w-28 h-28 rounded-2xl overflow-hidden shadow-2xl" style={{ boxShadow: `0 12px 40px ${accentColor}44` }}>
   295→   295→              <ProfileAvatar slug={user?.avatar} size={112} />
   296→   296→            </div>
   297→   297→            {!editing && (
   298→   298→              <button onClick={startEdit} className="absolute -bottom-1 -right-1 w-9 h-9 rounded-xl bg-white/10 backdrop-blur-sm border border-white/20 flex items-center justify-center hover:bg-white/20 transition-colors" style={{ boxShadow: `0 4px 12px ${accentColor}22` }}>
   299→   299→                <Pencil className="w-3.5 h-3.5 text-white" />
   300→   300→              </button>
   301→   301→            )}
   302→   302→            {/* Online indicator */}
   303→   303→            <div className="absolute -bottom-0.5 -left-0.5 w-4 h-4 rounded-full border-2 border-zinc-950" style={{ backgroundColor: '#22c55e' }} />
   304→   304→          </motion.div>
   305→   305→
   306→   306→          {/* Info */}
   307→   307→          <div className="text-center sm:text-left flex-1">
   308→   308→            <h1 className="text-2xl md:text-3xl font-bold text-white">{user?.name || 'StreamVault User'}</h1>
   309→   309→            <p className="text-white/40 text-sm mt-1">{user?.email}</p>
   310→   310→            {user?.bio && !editing && <p className="text-white/60 text-sm mt-2 max-w-md">{user.bio}</p>}
   311→   311→            <div className="flex flex-wrap items-center justify-center sm:justify-start gap-3 mt-3 text-xs text-white/50">
   312→   312→              <span className="flex items-center gap-1.5"><Calendar className="w-3.5 h-3.5" />Joined {memberSince}</span>
   313→   313→              <span className="flex items-center gap-1.5"><Bookmark className="w-3.5 h-3.5" />{watchlist.length} saved</span>
   314→   314→              {profileStats && <span className="flex items-center gap-1.5"><Clock className="w-3.5 h-3.5" />{profileStats.watchHistoryCount} watched</span>}
   315→   315→            </div>
   316→   316→            {/* Favorite genres badges */}
   317→   317→            {user?.favoriteGenres && user.favoriteGenres.length > 0 && !editing && (
   318→   318→              <div className="flex flex-wrap gap-1.5 mt-3 justify-center sm:justify-start">
   319→   319→                {user.favoriteGenres.slice(0, 5).map(g => (
   320→   320→                  <span key={g} className="px-2.5 py-1 rounded-lg text-[10px] font-semibold uppercase tracking-wider text-white/80" style={{ backgroundColor: `${accentColor}22`, border: `1px solid ${accentColor}33` }}>{g}</span>
   321→   321→                ))}
   322→   322→                {user.favoriteGenres.length > 5 && (
   323→   323→                  <span className="px-2.5 py-1 rounded-lg text-[10px] font-semibold text-white/40 bg-white/[0.06]">+{user.favoriteGenres.length - 5}</span>
   324→   324→                )}
   325→   325→              </div>
   326→   326→            )}
   327→   327→          </div>
   328→   328→
   329→   329→          {/* Actions */}
   330→   330→          <div className="sm:ml-auto flex gap-2">
   331→   331→            <button onClick={startEdit} className="flex items-center gap-2 px-4 py-2.5 rounded-xl text-white/80 text-sm font-medium transition-all hover:brightness-110" style={{ backgroundColor: `${accentColor}22`, border: `1px solid ${accentColor}33` }}>
   332→   332→              <Pencil className="w-4 h-4" /><span className="hidden sm:inline">Edit Profile</span>
   333→   333→            </button>
   334→   334→            <button onClick={async () => { await logout(); goHome(); }} className="flex items-center gap-2 px-4 py-2 rounded-xl bg-white/[0.06] hover:bg-white/10 text-white/60 hover:text-white text-sm transition-all border border-white/[0.06]">
   335→   335→              <LogOut className="w-4 h-4" /><span className="hidden sm:inline">Sign Out</span>
   336→   336→            </button>
   337→   337→          </div>
   338→   338→        </div>
   339→   339→        )}
   340→   340→        {!isLoggedIn && (
   341→   341→          <div className="mb-8">
   342→   342→            <h1 className="text-2xl md:text-3xl font-bold text-white">Browsing History</h1>
   343→   343→            <p className="text-white/40 text-sm mt-1">Your recently viewed content is saved locally. <button onClick={() => setAuthModalOpen(true)} className="text-white/60 hover:text-white underline underline-offset-2 transition-colors">Sign in</button> to sync across devices.</p>
   344→   344→          </div>
   345→   345→        )}
   346→   346→
   347→   347→        {/* Tabs */}
   348→   348→        <div className="flex gap-1 mb-6 border-b border-white/[0.06]">
   349→   349→          {tabsList.map((t) => (
   350→   350→            <button key={t.key} onClick={() => setTab(t.key)} className={`flex items-center gap-2 px-4 py-3 text-sm font-medium transition-all relative border-b-2 -mb-px ${tab === t.key ? 'text-white' : 'text-white/40 border-transparent hover:text-white/70'}`} style={tab === t.key ? { borderColor: accentColor } : undefined}>
   351→   351→              {t.icon}
   352→   352→              <span className="hidden sm:inline">{t.label}</span>
   353→   353→              <span className="sm:hidden">{t.key === 'watchlist' ? watchlist.length : ''}</span>
   354→   354→            </button>
   355→   355→          ))}
   356→   356→        </div>
   357→   357→
   358→   358→        {/* Tab Content */}
   359→   359→        <AnimatePresence mode="wait">
   360→   360→          {tab === 'profile' && (
   361→   361→            <motion.div key="profile-tab" initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -8 }} className="max-w-2xl">
   362→   362→              {editing ? (
   363→   363→                <EditPanel
   364→   364→                  editStep={editStep} setEditStep={setEditStep}
   365→   365→                  editName={editName} setEditName={setEditName}
   366→   366→                  editBio={editBio} setEditBio={setEditBio}
   367→   367→                  selectedAvatar={selectedAvatar} setSelectedAvatar={setSelectedAvatar}
   368→   368→                  selectedColor={selectedColor} setSelectedColor={setSelectedColor}
   369→   369→                  selectedGenres={selectedGenres} toggleGenre={toggleGenre}
   370→   370→                  accentColor={accentColor}
   371→   371→                  saving={saving} saveError={saveError}
   372→   372→                  onSave={saveProfile} onCancel={() => setEditing(false)}
   373→   373→                />
   374→   374→              ) : (
   375→   375→                <div className="space-y-4">
   376→   376→                  <div className="bg-white/[0.03] rounded-2xl border border-white/[0.06] p-5 space-y-4">
   377→   377→                    <h3 className="text-sm font-semibold text-white/70 flex items-center gap-2"><Shield className="w-4 h-4 text-white/50" />Account Info</h3>
   378→   378→                    <div className="space-y-3">
   379→   379→                      <InfoRow label="Name" value={user?.name || '\u2014'} />
   380→   380→                      <InfoRow label="Email" value={user?.email} />
   381→   381→                      <InfoRow label="Avatar" value={avatarDef.name} />
