     1→'use client';
     2→
     3→import { useState, useRef, useEffect, useCallback } from 'react';
     4→import {
     5→  X, Loader2, ArrowLeft, Play, Tv, ChevronDown, ChevronUp,
     6→  Zap, AlertTriangle, RefreshCw, MonitorPlay, LayoutGrid,
     7→} from 'lucide-react';
     8→import { motion, AnimatePresence } from 'framer-motion';
     9→import { providers, getProvider, getEmbedUrl } from '@/lib/providers';
    10→import { useAppStore } from '@/store/app-store';
    11→import { useAuthStore } from '@/store/auth-store';
    12→import { useIsMobile } from '@/hooks/use-mobile';
    13→import { recordWatchHistory } from '@/lib/watch-history';
    14→import { getImageUrl } from '@/lib/tmdb';
    15→import type { Episode } from '@/lib/types';
    16→
    17→interface VideoPlayerProps {
    18→  src: string;
    19→  title?: string;
    20→  onClose: () => void;
    21→  mediaType: 'movie' | 'tv';
    22→  tmdbId: number;
    23→  season?: number;
    24→  episode?: number;
    25→}
    26→
    27→function useIOSFullscreenDetect(onReturn: () => void) {
    28→  useEffect(() => {
    29→    const h = () => onReturn();
    30→    document.addEventListener('webkitendfullscreen', h);
    31→    return () => document.removeEventListener('webkitendfullscreen', h);
    32→  }, [onReturn]);
    33→}
    34→
    35→/* Animated equalizer bars for "now playing" indicator */
    36→function PlayingBars({ color }: { color: string }) {
    37→  return (
    38→    <div className="flex items-end gap-[2px] h-3">
    39→      {[0, 1, 2].map((i) => (
    40→        <motion.div
    41→          key={i}
    42→          className="w-[3px] rounded-full"
    43→          style={{ backgroundColor: color }}
    44→          animate={{ height: ['40%', '100%', '60%', '100%', '40%'] }}
    45→          transition={{
    46→            duration: 1.2,
    47→            repeat: Infinity,
    48→            delay: i * 0.2,
    49→            ease: 'easeInOut',
    50→          }}
    51→        />
    52→      ))}
    53→    </div>
    54→  );
    55→}
    56→
    57→export function VideoPlayer({ src, title, onClose, mediaType, tmdbId, season, episode }: VideoPlayerProps) {
    58→  const [loading, setLoading] = useState(true);
    59→  const [error, setError] = useState(false);
    60→  const [eps, setEps] = useState<Episode[]>([]);
    61→  const [epsLoading, setEpsLoading] = useState(false);
    62→  const [showEps, setShowEps] = useState(mediaType === 'tv');
    63→  const [curSrc, setCurSrc] = useState(src);
    64→  const [curSeason, setCurSeason] = useState(season || 1);
    65→  const [showServerMenu, setShowServerMenu] = useState(false);
    66→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    67→  const iframeRef = useRef<HTMLIFrameElement>(null);
    68→  const serverMenuRef = useRef<HTMLDivElement>(null);
    69→  const { selectedProvider, setSelectedProvider, selectedMovie, selectedTv } = useAppStore();
    70→  const user = useAuthStore(s => s.user);
    71→  const isMobile = useIsMobile();
    72→  const provider = getProvider(selectedProvider);
    73→
    74→  useEffect(() => {
    75→    if (timerRef.current) clearTimeout(timerRef.current);
    76→    timerRef.current = setTimeout(() => setLoading(false), isMobile ? 5000 : 10000);
    77→    return () => { if (timerRef.current) clearTimeout(timerRef.current); };
    78→  }, [curSrc, isMobile]);
    79→
    80→  const fetchEps = useCallback(async (id: number, s: number) => {
    81→    setEpsLoading(true);
    82→    try {
    83→      const r = await fetch(`/api/tmdb/tv/${id}/season/${s}`);
    84→      const d = await r.json();
    85→      setEps(d.episodes || []);
    86→    } catch { setEps([]); }
    87→    finally { setEpsLoading(false); }
    88→  }, []);
    89→
    90→  useEffect(() => {
    91→    if (mediaType === 'tv') fetchEps(tmdbId, curSeason);
    92→  }, [mediaType, tmdbId, curSeason, fetchEps]);
    93→
    94→  useEffect(() => {
    95→    if (!title) return;
    96→    const item = mediaType === 'tv' ? selectedTv : selectedMovie;
    97→    recordWatchHistory({ tmdbId, title, posterPath: item?.poster_path || null, backdropPath: item?.backdrop_path || null, mediaType, season: season ?? null, episode: episode ?? null });
    98→    if (user) { fetch('/api/profile/history', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ tmdbId, title, posterPath: item?.poster_path || null, mediaType, season: season ?? null, episode: episode ?? null }) }).catch(() => {}); }
    99→  }, [tmdbId, mediaType, season, episode, title, user, selectedMovie, selectedTv]);
   100→
   101→  useEffect(() => {
   102→    const y = window.scrollY;
   103→    document.body.style.setProperty('--player-scroll-y', `${y}px`);
   104→    document.body.classList.add('player-open');
   105→    document.body.style.top = `-${y}px`;
   106→    return () => { document.body.classList.remove('player-open'); document.body.style.top = ''; window.scrollTo(0, y); };
   107→  }, []);
   108→
   109→  useIOSFullscreenDetect(onClose);
   110→
   111→  useEffect(() => {
   112→    const fn = (e: KeyboardEvent) => { if (e.key === 'Escape') onClose(); };
   113→    window.addEventListener('keydown', fn);
   114→    return () => window.removeEventListener('keydown', fn);
   115→  }, [onClose]);
   116→
   117→  /* Close server menu on outside click */
   118→  useEffect(() => {
   119→    if (!showServerMenu) return;
   120→    const handler = (e: MouseEvent) => {
   121→      if (serverMenuRef.current && !serverMenuRef.current.contains(e.target as Node)) {
   122→        setShowServerMenu(false);
   123→      }
   124→    };
   125→    document.addEventListener('mousedown', handler);
   126→    return () => document.removeEventListener('mousedown', handler);
   127→  }, [showServerMenu]);
   128→
   129→  const switchServer = (id: string) => {
   130→    setSelectedProvider(id);
   131→    setCurSrc(getEmbedUrl(id, mediaType, tmdbId, season, episode));
   132→    setLoading(true); setError(false);
   133→    setShowServerMenu(false);
   134→  };
   135→
   136→  const playEp = (ep: Episode) => {
   137→    setCurSrc(getEmbedUrl(selectedProvider, 'tv', tmdbId, curSeason, ep.episode_number));
   138→    setLoading(true); setError(false);
   139→    useAppStore.getState().setSelectedEpisode(ep);
   140→  };
   141→
   142→  const doRetry = () => {
   143→    setError(false); setLoading(true);
   144→    const sep = curSrc.includes('?') ? '&' : '?';
   145→    setCurSrc(curSrc + sep + '_r=' + Date.now());
   146→  };
   147→
   148→  const curEp = eps.find(e => e.episode_number === episode);
   149→  const epLabel = mediaType === 'tv' && season && episode ? `S${String(season).padStart(2, '0')}E${String(episode).padStart(2, '0')}` : '';
   150→
   151→  /* ── Tier labels for server groups ── */
   152→  const tierLabels = ['Recommended', 'Alternatives', 'Fallbacks'] as const;
   153→  const tierProviders = [
   154→    providers.slice(0, 3),
   155→    providers.slice(3, 8),
   156→    providers.slice(8),
   157→  ];
   158→
   159→  /* ── Episode panel content (shared between mobile & desktop) ── */
   160→  const epsContent = (
   161→    <div className="flex flex-col h-full">
   162→      {/* Season tabs */}
   163→      <div className="flex items-center gap-1 px-3 py-2.5 border-b border-white/[0.06] overflow-x-auto no-scrollbar shrink-0">
   164→        <LayoutGrid className="w-3.5 h-3.5 text-white/20 shrink-0 mr-1" />
   165→        {Array.from({ length: 20 }, (_, i) => i + 1).map((s) => (
   166→          <button
   167→            key={s}
   168→            onClick={() => { setCurSeason(s); fetchEps(tmdbId, s); }}
   169→            className={`shrink-0 px-3 py-1.5 rounded-lg text-xs font-semibold tracking-wide transition-all duration-200 min-h-[34px] ${
   170→              curSeason === s
   171→                ? 'bg-white text-black shadow-sm'
   172→                : 'text-white/30 hover:text-white/60 hover:bg-white/[0.06] active:text-white/80 active:bg-white/10'
   173→            }`}
   174→          >S{s}</button>
   175→        ))}
   176→      </div>
   177→
   178→      {/* Episode list */}
   179→      <div className="flex-1 overflow-y-auto content-scroll">
   180→        {epsLoading ? (
   181→          <div className="p-3 space-y-2">
   182→            {Array.from({ length: 5 }).map((_, i) => (
   183→              <div key={i} className="h-[72px] bg-white/[0.03] rounded-2xl animate-pulse" />
   184→            ))}
   185→          </div>
   186→        ) : eps.length === 0 ? (
   187→          <div className="flex flex-col items-center justify-center py-16 text-white/20 text-xs gap-2">
   188→            <Tv className="w-6 h-6" />
   189→            <span>No episodes found</span>
   190→          </div>
   191→        ) : (
   192→          <div className="p-2 space-y-0.5">
   193→            {eps.map((ep, idx) => {
   194→              const cur = ep.episode_number === episode;
   195→              return (
   196→                <button
   197→                  key={ep.id}
   198→                  onClick={() => playEp(ep)}
   199→                  className={`w-full flex items-center gap-3 p-2 rounded-2xl text-left transition-all duration-200 group ${
   200→                    cur
   201→                      ? 'bg-white/[0.08] ring-1 ring-white/[0.06]'
   202→                      : 'hover:bg-white/[0.04] active:bg-white/[0.06]'
   203→                  }`}
   204→                >
   205→                  {/* Thumbnail */}
   206→                  <div className="relative w-[88px] aspect-video rounded-xl overflow-hidden bg-white/[0.04] shrink-0">
   207→                    {ep.still_path ? (
   208→                      <img
   209→                        src={getImageUrl(ep.still_path, 'w300')}
   210→                        alt={ep.name}
   211→                        className="w-full h-full object-cover"
   212→                        loading="lazy"
   213→                      />
   214→                    ) : (
   215→                      <div className="w-full h-full flex items-center justify-center bg-gradient-to-br from-white/[0.03] to-white/[0.06]">
   216→                        <Play className="w-4 h-4 text-white/10" />
   217→                      </div>
   218→                    )}
   219→
   220→                    {/* Currently playing overlay */}
   221→                    {cur && (
   222→                      <div className="absolute inset-0 bg-black/60 flex items-center justify-center backdrop-blur-[2px]">
   223→                        <PlayingBars color={provider.color} />
   224→                      </div>
   225→                    )}
   226→
   227→                    {/* Hover play overlay */}
   228→                    {!cur && (
   229→                      <div className="absolute inset-0 flex items-center justify-center opacity-0 group-hover:opacity-100 transition-opacity duration-200 bg-black/40">
   230→                        <div className="w-7 h-7 rounded-full bg-white/20 backdrop-blur-sm flex items-center justify-center">
   231→                          <Play className="w-3 h-3 text-white fill-white ml-0.5" />
   232→                        </div>
   233→                      </div>
   234→                    )}
   235→
   236→                    {/* Runtime badge */}
   237→                    {ep.runtime && (
   238→                      <div className="absolute bottom-1 right-1 bg-black/70 backdrop-blur-sm px-1.5 py-0.5 rounded-md text-[9px] text-white/70 font-medium">
   239→                        {ep.runtime}m
   240→                      </div>
   241→                    )}
   242→                  </div>
   243→
   244→                  {/* Info */}
   245→                  <div className="flex-1 min-w-0 py-0.5">
   246→                    <div className="flex items-center gap-1.5">
   247→                      <span className={`text-[11px] font-mono font-semibold tabular-nums ${cur ? 'text-white/50' : 'text-white/25'}`}>
   248→                        E{String(ep.episode_number).padStart(2, '0')}
   249→                      </span>
   250→                      {ep.vote_average > 0 && (
   251→                        <span className="text-[10px] text-amber-400/70 font-medium">
   252→                          {ep.vote_average.toFixed(1)}
   253→                        </span>
   254→                      )}
   255→                    </div>
   256→                    <p className={`text-xs font-medium truncate mt-0.5 ${cur ? 'text-white' : 'text-white/50'}`}>
   257→                      {ep.name}
   258→                    </p>
   259→                    {ep.overview && (
   260→                      <p className="text-[10px] text-white/20 line-clamp-1 mt-0.5 leading-relaxed">
   261→                        {ep.overview}
   262→                      </p>
   263→                    )}
   264→                  </div>
   265→                </button>
   266→              );
   267→            })}
   268→          </div>
   269→        )}
   270→      </div>
   271→    </div>
   272→  );
   273→
   274→  /* ── Server selector dropdown ── */
   275→  const serverMenu = (
   276→    <AnimatePresence>
   277→      {showServerMenu && (
   278→        <motion.div
   279→          ref={serverMenuRef}
   280→          initial={{ opacity: 0, y: 8, scale: 0.97 }}
   281→          animate={{ opacity: 1, y: 0, scale: 1 }}
   282→          exit={{ opacity: 0, y: 8, scale: 0.97 }}
   283→          transition={{ duration: 0.15, ease: [0.23, 1, 0.32, 1] }}
   284→          className="absolute bottom-full left-0 mb-2 w-[280px] bg-[#141414]/95 backdrop-blur-xl border border-white/[0.08] rounded-2xl shadow-2xl shadow-black/60 overflow-hidden z-50"
   285→        >
   286→          <div className="px-4 py-3 border-b border-white/[0.06]">
   287→            <p className="text-white/40 text-[11px] font-semibold uppercase tracking-wider">Select Server</p>
   288→          </div>
   289→          <div className="max-h-[340px] overflow-y-auto content-scroll py-1.5">
   290→            {tierProviders.map((tier, tIdx) => (
   291→              <div key={tIdx}>
   292→                <div className="px-4 pt-2 pb-1">
   293→                  <p className="text-white/20 text-[10px] font-semibold uppercase tracking-widest">{tierLabels[tIdx]}</p>
   294→                </div>
   295→                {tier.map((p) => {
   296→                  const active = p.id === selectedProvider;
   297→                  return (
   298→                    <button
   299→                      key={p.id}
   300→                      onClick={() => switchServer(p.id)}
   301→                      className={`w-full flex items-center gap-3 px-4 py-2.5 text-left transition-all duration-150 ${
   302→                        active
   303→                          ? 'bg-white/[0.08]'
   304→                          : 'hover:bg-white/[0.04]'
   305→                      }`}
   306→                    >
   307→                      <div
   308→                        className="w-7 h-7 rounded-lg flex items-center justify-center shrink-0 text-xs"
   309→                        style={{
   310→                          backgroundColor: active ? p.color + '25' : 'white/[0.04]',
   311→                        }}
   312→                      >
   313→                        <span style={{ color: active ? p.color : 'white/30' }}>{p.icon}</span>
   314→                      </div>
   315→                      <div className="flex-1 min-w-0">
   316→                        <div className="flex items-center gap-2">
   317→                          <span className={`text-sm font-medium ${active ? 'text-white' : 'text-white/60'}`}>
   318→                            {p.name}
   319→                          </span>
   320→                          {p.primary && (
   321→                            <span className="px-1.5 py-0.5 rounded-md bg-amber-400/10 text-amber-400 text-[9px] font-bold uppercase tracking-wider">
   322→                              Best
   323→                            </span>
   324→                          )}
   325→                        </div>
   326→                        <p className="text-white/25 text-[11px] truncate">{p.description}</p>
   327→                      </div>
   328→                      {active && (
   329→                        <div className="w-2 h-2 rounded-full shrink-0" style={{ backgroundColor: p.color }} />
   330→                      )}
   331→                    </button>
   332→                  );
   333→                })}
   334→              </div>
   335→            ))}
   336→          </div>
   337→        </motion.div>
   338→      )}
   339→    </AnimatePresence>
   340→  );
   341→
   342→  return (
   343→    <AnimatePresence>
   344→      <motion.div
   345→        initial={{ opacity: 0 }}
   346→        animate={{ opacity: 1 }}
   347→        exit={{ opacity: 0 }}
   348→        transition={{ duration: 0.2 }}
   349→        className="fixed inset-0 z-[100] bg-[#050505] flex flex-col"
   350→      >
   351→        {/* ── Header with glassmorphism ── */}
   352→        <div
   353→          className="flex items-center justify-between px-2 sm:px-4 shrink-0 z-10 backdrop-blur-xl bg-gradient-to-b from-black/60 to-transparent"
   354→          style={{ paddingTop: 'env(safe-area-inset-top, 0px)' }}
   355→        >
   356→          <button
   357→            onClick={onClose}
   358→            className="flex items-center justify-center w-10 h-10 rounded-xl text-white/60 hover:text-white active:text-white active:bg-white/10 transition-all duration-200"
   359→            aria-label="Go back"
   360→          >
   361→            <ArrowLeft className="w-5 h-5" />
   362→          </button>
   363→
   364→          {/* Center title with episode badge */}
   365→          <div className="flex items-center gap-2.5 min-w-0 max-w-[60%]">
   366→            {epLabel && (
   367→              <span
   368→                className="shrink-0 px-2 py-0.5 rounded-md text-[10px] font-bold tracking-wider uppercase"
   369→                style={{ backgroundColor: provider.color + '20', color: provider.color }}
   370→              >
   371→                {epLabel}
   372→              </span>
   373→            )}
   374→            <h3 className="text-white/80 text-sm font-semibold truncate">
   375→              {title || 'Now Playing'}
   376→            </h3>
   377→          </div>
   378→
   379→          <button
   380→            onClick={onClose}
   381→            className="flex items-center justify-center w-10 h-10 rounded-xl text-white/30 hover:text-white/70 active:text-white active:bg-white/10 transition-all duration-200"
   382→            aria-label="Close"
   383→          >
   384→            <X className="w-[18px] h-[18px]" />
   385→          </button>
   386→        </div>
   387→
   388→        {/* ── Main content area ── */}
   389→        <div className="flex-1 flex min-h-0 overflow-hidden">
   390→          <div className="flex-1 flex flex-col min-w-0 min-h-0">
   391→            {/* Video iframe area */}
   392→            <div className="relative flex-1 min-h-0 bg-black">
   393→              {/* Loading overlay */}
   394→              <AnimatePresence>
   395→                {loading && !error && (
   396→                  <motion.div
   397→                    initial={{ opacity: 0 }}
   398→                    animate={{ opacity: 1 }}
   399→                    exit={{ opacity: 0 }}
   400→                    transition={{ duration: 0.3 }}
   401→                    className="absolute inset-0 flex items-center justify-center z-[1] pointer-events-none bg-black/40"
   402→                  >
   403→                    <div className="flex flex-col items-center gap-4">
   404→                      <div className="relative">
   405→                        <div
   406→                          className="w-16 h-16 rounded-2xl flex items-center justify-center"
   407→                          style={{ backgroundColor: provider.color + '15' }}
   408→                        >
   409→                          <Loader2
   410→                            className="w-7 h-7 animate-spin"
   411→                            style={{ color: provider.color }}
   412→                          />
   413→                        </div>
   414→                        <motion.div
   415→                          className="absolute inset-0 rounded-2xl"
   416→                          style={{ border: `2px solid ${provider.color}30` }}
   417→                          animate={{ opacity: [0.3, 0.8, 0.3], scale: [1, 1.08, 1] }}
   418→                          transition={{ duration: 2, repeat: Infinity, ease: 'easeInOut' }}
   419→                        />
   420→                      </div>
   421→                      <div className="text-center">
   422→                        <p className="text-white/50 text-sm font-medium">Connecting to {provider.name}</p>
   423→                        <p className="text-white/25 text-xs mt-1">Please wait...</p>
   424→                      </div>
   425→                    </div>
   426→                  </motion.div>
   427→                )}
   428→              </AnimatePresence>
   429→
   430→              {/* Error overlay */}
   431→              <AnimatePresence>
   432→                {error && (
   433→                  <motion.div
   434→                    initial={{ opacity: 0 }}
   435→                    animate={{ opacity: 1 }}
   436→                    exit={{ opacity: 0 }}
   437→                    transition={{ duration: 0.2 }}
   438→                    className="absolute inset-0 flex flex-col items-center justify-center gap-4 z-[2] bg-[#050505]"
   439→                  >
   440→                    <div
   441→                      className="w-16 h-16 rounded-2xl flex items-center justify-center"
   442→                      style={{ backgroundColor: 'rgba(239, 68, 68, 0.1)' }}
   443→                    >
   444→                      <AlertTriangle className="w-7 h-7 text-red-400" />
   445→                    </div>
   446→                    <div className="text-center">
   447→                      <p className="text-white/60 text-sm font-medium">Failed to load stream</p>
   448→                      <p className="text-white/30 text-xs mt-1">{provider.name} couldn't serve this content</p>
   449→                    </div>
   450→                    <div className="flex items-center gap-2">
   451→                      <button
   452→                        onClick={doRetry}
   453→                        className="flex items-center gap-2 px-5 py-2.5 rounded-xl bg-white/[0.08] hover:bg-white/[0.12] active:bg-white/[0.16] text-white text-sm font-medium transition-all duration-200"
   454→                      >
   455→                        <RefreshCw className="w-3.5 h-3.5" />
   456→                        Retry
   457→                      </button>
   458→                      <button
   459→                        onClick={() => setShowServerMenu(true)}
   460→                        className="flex items-center gap-2 px-5 py-2.5 rounded-xl text-white/60 hover:text-white hover:bg-white/[0.08] text-sm font-medium transition-all duration-200"
   461→                      >
   462→                        <MonitorPlay className="w-3.5 h-3.5" />
   463→                        Switch Server
   464→                      </button>
   465→                    </div>
   466→                  </motion.div>
   467→                )}
   468→              </AnimatePresence>
   469→
   470→              <iframe
   471→                ref={iframeRef}
   472→                key={curSrc}
   473→                src={curSrc}
   474→                className="absolute inset-0 w-full h-full"
   475→                allowFullScreen
   476→                allow="autoplay; fullscreen; encrypted-media; picture-in-picture"
   477→                playsInline
   478→                onLoad={() => { setLoading(false); if (timerRef.current) clearTimeout(timerRef.current); }}
   479→                onError={() => { setLoading(false); setError(true); }}
   480→                title={title || 'Video Player'}
   481→              />
   482→            </div>
   483→
   484→            {/* ── Bottom bar ── */}
   485→            <div
   486→              className="shrink-0 bg-[#0a0a0a]/90 backdrop-blur-xl border-t border-white/[0.05]"
   487→              style={{ paddingBottom: 'env(safe-area-inset-bottom, 0px)' }}
   488→            >
   489→              <div className="flex items-center justify-between px-3 pt-3 pb-2">
   490→                {/* Left: Active server indicator + status */}
   491→                <div className="flex items-center gap-2.5 min-w-0 flex-1">
   492→                  {/* Server selector trigger */}
   493→                  <div className="relative">
   494→                    <button
   495→                      onClick={() => setShowServerMenu(!showServerMenu)}
   496→                      className={`flex items-center gap-2 px-3 py-2 rounded-xl text-xs font-semibold transition-all duration-200 border ${
   497→                        showServerMenu
   498→                          ? 'border-white/15 bg-white/[0.08] text-white'
