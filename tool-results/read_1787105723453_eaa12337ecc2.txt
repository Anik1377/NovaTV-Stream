     1→'use client';
     2→
     3→import { useState, useEffect, useCallback, useMemo } from 'react';
     4→import {
     5→  ArrowLeft, User, Bookmark, Clock, LogOut, Pencil, Check, X, Trash2,
     6→  Loader2, Film, Tv, Users, Calendar, Shield, Palette, Heart, Camera,
     7→  ChevronDown, ShieldOff,
     8→} from 'lucide-react';
     9→import { motion, AnimatePresence } from 'framer-motion';
    10→import { useAuthStore } from '@/store/auth-store';
    11→import { useAppStore } from '@/store/app-store';
    12→import { Input } from '@/components/ui/input';
    13→import { AuthModal } from '@/components/auth/AuthModal';
    14→import {
    15→  ProfileAvatar, AVATAR_DEFS, ACCENT_COLORS, GENRE_OPTIONS,
    16→  getAvatarDef, type AvatarDef,
    17→} from '@/lib/avatars';
    18→import { getImageUrl } from '@/lib/tmdb';
    19→import { getLocalBrowseHistory, deleteLocalHistoryItem, clearLocalHistory } from '@/lib/useRecordHistory';
    20→import type { BrowseHistoryEntry } from '@/lib/useRecordHistory';
    21→import type { Movie } from '@/lib/types';
    22→
    23→interface BrowseHistoryItem {
    24→  id: string;
    25→  tmdbId: number;
    26→  title: string;
    27→  posterPath: string | null;
    28→  mediaType: 'movie' | 'tv' | 'person';
    29→  subtitle: string | null;
    30→  visitedAt: string;
    31→}
    32→
    33→type HistoryFilter = 'all' | 'movie' | 'tv' | 'person';
    34→
    35→type Tab = 'profile' | 'watchlist' | 'history';
    36→
    37→type EditStep = 'main' | 'avatar' | 'color' | 'genres';
    38→
    39→export function ProfilePage() {
    40→  const { user, loading: authLoading, updateProfile, logout } = useAuthStore();
    41→  const goHome = useAppStore(s => s.goHome);
    42→  const watchlist = useAppStore(s => s.watchlist);
    43→  const selectMovie = useAppStore(s => s.selectMovie);
    44→  const selectTv = useAppStore(s => s.selectTv);
    45→  const selectPerson = useAppStore(s => s.selectPerson);
    46→
    47→  const [tab, setTab] = useState<Tab>('profile');
    48→  const [authModalOpen, setAuthModalOpen] = useState(false);
    49→
    50→  // Profile editing
    51→  const [editing, setEditing] = useState(false);
    52→  const [editStep, setEditStep] = useState<EditStep>('main');
    53→  const [editName, setEditName] = useState('');
    54→  const [editBio, setEditBio] = useState('');
    55→  const [selectedAvatar, setSelectedAvatar] = useState('hero');
    56→  const [selectedColor, setSelectedColor] = useState('#e11d48');
    57→  const [selectedGenres, setSelectedGenres] = useState<string[]>([]);
    58→  const [saving, setSaving] = useState(false);
    59→  const [saveError, setSaveError] = useState('');
    60→
    61→  // Browse history
    62→  const [history, setHistory] = useState<BrowseHistoryItem[]>([]);
    63→  const [historyLoading, setHistoryLoading] = useState(false);
    64→  const [historyFilter, setHistoryFilter] = useState<HistoryFilter>('all');
    65→  const [historyPage, setHistoryPage] = useState(1);
    66→  const [historyTotal, setHistoryTotal] = useState(0);
    67→  const [historyLoadMore, setHistoryLoadMore] = useState(false);
    68→  const [deletingId, setDeletingId] = useState<string | null>(null);
    69→
    70→  // Profile stats from API
    71→  const [profileStats, setProfileStats] = useState<{ watchHistoryCount: number } | null>(null);
    72→
    73→  // Computed accent color for UI
    74→  const accentColor = useMemo(() => {
    75→    if (user?.accentColor) return user.accentColor;
    76→    const def = getAvatarDef(user?.avatar);
    77→    return def.color;
    78→  }, [user?.avatar, user?.accentColor]);
    79→
    80→  const fetchProfile = useCallback(async () => {
    81→    if (!user) return;
    82→    try {
    83→      const res = await fetch('/api/profile');
    84→      const data = await res.json();
    85→      if (res.ok) setProfileStats(data.stats);
    86→    } catch { /* ignore */ }
    87→  }, [user]);
    88→
    89→  const fetchHistory = useCallback(async (filter: HistoryFilter, page: number, append: boolean) => {
    90→    // Non-authenticated users: use localStorage
    91→    if (!user) {
    92→      setHistoryLoading(true);
    93→      let local = getLocalBrowseHistory();
    94→      if (filter !== 'all') local = local.filter((i) => i.mediaType === filter);
    95→      setHistory(local);
    96→      setHistoryTotal(local.length);
    97→      setHistoryLoading(false);
    98→      return;
    99→    }
   100→    // Authenticated: use server API
   101→    if (append) { setHistoryLoadMore(true); } else { setHistoryLoading(true); }
   102→    try {
   103→      const params = new URLSearchParams({ limit: '50', offset: String((page - 1) * 50) });
   104→      if (filter !== 'all') params.set('type', filter);
   105→      const res = await fetch(`/api/history?${params}`);
   106→      const data = await res.json();
   107→      if (res.ok) {
   108→        setHistory(prev => append ? [...prev, ...data.items] : data.items);
   109→        setHistoryTotal(data.total);
   110→        if (!append) setHistoryPage(1);
   111→      }
   112→    } catch { /* ignore */ }
   113→    setHistoryLoading(false);
   114→    setHistoryLoadMore(false);
   115→  }, [user]);
   116→
   117→  useEffect(() => {
   118→    if (tab === 'profile') fetchProfile();
   119→    if (tab === 'history') fetchHistory(historyFilter, 1, false);
   120→  }, [tab, historyFilter, user, fetchHistory, fetchProfile]);
   121→
   122→  useEffect(() => {
   123→    window.scrollTo({ top: 0, behavior: 'smooth' });
   124→  }, []);
   125→
   126→  const startEdit = () => {
   127→    setEditName(user?.name || '');
   128→    setEditBio(user?.bio || '');
   129→    setSelectedAvatar(user?.avatar || 'hero');
   130→    setSelectedColor(user?.accentColor || getAvatarDef(user?.avatar).color);
   131→    setSelectedGenres(user?.favoriteGenres || []);
   132→    setEditStep('main');
   133→    setEditing(true);
   134→    setSaveError('');
   135→  };
   136→
   137→  const saveProfile = async () => {
   138→    setSaving(true);
   139→    setSaveError('');
   140→    const result = await updateProfile({
   141→      name: editName,
   142→      bio: editBio,
   143→      avatar: selectedAvatar,
   144→      accentColor: selectedColor,
   145→      favoriteGenres: selectedGenres,
   146→    });
   147→    setSaving(false);
   148→    if (result.error) {
   149→      setSaveError(result.error);
   150→    } else {
   151→      setEditing(false);
   152→    }
   153→  };
   154→
   155→  const toggleGenre = (genre: string) => {
   156→    setSelectedGenres(prev =>
   157→      prev.includes(genre) ? prev.filter(g => g !== genre) : [...prev, genre]
   158→    );
   159→  };
   160→
   161→  const clearHistory = async () => {
   162→    if (!confirm('Clear all browsing history? This cannot be undone.')) return;
   163→    if (user) {
   164→      await fetch('/api/history', { method: 'DELETE' });
   165→    } else {
   166→      clearLocalHistory();
   167→    }
   168→    setHistory([]);
   169→    setHistoryTotal(0);
   170→  };
   171→
   172→  const handleDeleteHistoryItem = async (id: string, e: React.MouseEvent) => {
   173→    e.stopPropagation();
   174→    setDeletingId(id);
   175→    if (user) {
   176→      await fetch(`/api/history?id=${id}`, { method: 'DELETE' });
   177→    } else {
   178→      deleteLocalHistoryItem(id);
   179→    }
   180→    setHistory(prev => prev.filter(item => item.id !== id));
   181→    setHistoryTotal(prev => prev - 1);
   182→    setDeletingId(null);
   183→  };
   184→
   185→  const handleHistoryClick = (item: BrowseHistoryItem) => {
   186→    if (item.mediaType === 'person') {
   187→      selectPerson({ id: item.tmdbId, name: item.title, profilePath: item.posterPath });
   188→      return;
   189→    }
   190→    const media: Movie = {
   191→      id: item.tmdbId, title: item.title, name: item.title,
   192→      poster_path: item.posterPath, media_type: item.mediaType,
   193→      vote_average: 0, genre_ids: [], overview: '', popularity: 0,
   194→      release_date: '', first_air_date: '', backdrop_path: null, original_language: '',
   195→    };
   196→    if (item.mediaType === 'tv') selectTv(media); else selectMovie(media);
   197→  };
   198→
   199→  const handleHistoryFilterChange = (f: HistoryFilter) => {
   200→    if (f === historyFilter) return;
   201→    setHistoryFilter(f);
   202→  };
   203→
   204→  const loadMoreHistory = () => {
   205→    const nextPage = historyPage + 1;
   206→    setHistoryPage(nextPage);
   207→    fetchHistory(historyFilter, nextPage, true);
   208→  };
   209→
   210→  const handleWatchlistClick = (tmdbId: number, mediaType: string) => {
   211→    const item: Movie = {
   212→      id: tmdbId, title: '', name: '', poster_path: null,
   213→      media_type: mediaType as 'movie' | 'tv', vote_average: 0, genre_ids: [],
   214→      overview: '', popularity: 0, release_date: '', first_air_date: '',
   215→      backdrop_path: null, original_language: '',
   216→    };
   217→    if (mediaType === 'tv') selectTv(item); else selectMovie(item);
   218→  };
   219→
   220→  // Not logged in — but still show the page with limited tabs
   221→  const isLoggedIn = !!user && !authLoading;
   222→
   223→  const memberSince = user ? new Date(user.createdAt || Date.now()).toLocaleDateString('en-US', { month: 'long', year: 'numeric' }) : '';
   224→  const avatarDef = getAvatarDef(user?.avatar);
   225→
   226→  // Default to history tab when not logged in
   227→  useEffect(() => {
   228→    if (!isLoggedIn && tab === 'profile') setTab('history');
   229→  }, [isLoggedIn]);
   230→
   231→  const tabsList: { key: Tab; label: string; icon: React.ReactNode }[] = [
   232→    ...(isLoggedIn ? [{ key: 'profile' as Tab, label: 'Profile', icon: <User className="w-4 h-4" /> }] : []),
   233→    ...(isLoggedIn ? [{ key: 'watchlist' as Tab, label: `Watchlist (${watchlist.length})`, icon: <Bookmark className="w-4 h-4" /> }] : []),
   234→    { key: 'history', label: `History${historyTotal > 0 ? ` (${historyTotal})` : ''}`, icon: <Clock className="w-4 h-4" /> },
   235→  ];
   236→
   237→  return (
   238→    <div className="min-h-screen pb-10">
   239→      {/* Back button */}
   240→      <div className="pt-20 px-4 md:px-8">
   241→        <button onClick={goHome} className="flex items-center gap-2 text-white/50 hover:text-white mb-6 transition-colors group">
   242→          <div className="w-8 h-8 rounded-full bg-white/[0.06] flex items-center justify-center group-hover:bg-white/10 transition-colors">
   243→            <ArrowLeft className="w-4 h-4" />
   244→          </div>
   245→          <span className="text-sm font-medium">Back</span>
   246→        </button>
   247→
   248→        {/* ── Profile Header (logged in only) ── */}
   249→        {isLoggedIn && (
   250→        <div className="flex flex-col sm:flex-row items-center sm:items-start gap-6 mb-8">
   251→          {/* Avatar */}
   252→          <motion.div className="relative" initial={{ scale: 0.8, opacity: 0 }} animate={{ scale: 1, opacity: 1 }} transition={{ type: 'spring', stiffness: 300, damping: 20 }}>
   253→            <div className="w-28 h-28 rounded-2xl overflow-hidden shadow-2xl" style={{ boxShadow: `0 12px 40px ${accentColor}44` }}>
   254→              <ProfileAvatar slug={user?.avatar} size={112} />
   255→            </div>
   256→            {!editing && (
   257→              <button onClick={startEdit} className="absolute -bottom-1 -right-1 w-9 h-9 rounded-xl bg-white/10 backdrop-blur-sm border border-white/20 flex items-center justify-center hover:bg-white/20 transition-colors" style={{ boxShadow: `0 4px 12px ${accentColor}22` }}>
   258→                <Pencil className="w-3.5 h-3.5 text-white" />
   259→              </button>
   260→            )}
   261→            {/* Online indicator */}
   262→            <div className="absolute -bottom-0.5 -left-0.5 w-4 h-4 rounded-full border-2 border-zinc-950" style={{ backgroundColor: '#22c55e' }} />
   263→          </motion.div>
   264→
   265→          {/* Info */}
   266→          <div className="text-center sm:text-left flex-1">
   267→            <h1 className="text-2xl md:text-3xl font-bold text-white">{user?.name || 'StreamVault User'}</h1>
   268→            <p className="text-white/40 text-sm mt-1">{user?.email}</p>
   269→            {user?.bio && !editing && <p className="text-white/60 text-sm mt-2 max-w-md">{user.bio}</p>}
   270→            <div className="flex flex-wrap items-center justify-center sm:justify-start gap-3 mt-3 text-xs text-white/50">
   271→              <span className="flex items-center gap-1.5"><Calendar className="w-3.5 h-3.5" />Joined {memberSince}</span>
   272→              <span className="flex items-center gap-1.5"><Bookmark className="w-3.5 h-3.5" />{watchlist.length} saved</span>
   273→              {profileStats && <span className="flex items-center gap-1.5"><Clock className="w-3.5 h-3.5" />{profileStats.watchHistoryCount} watched</span>}
   274→            </div>
   275→            {/* Favorite genres badges */}
   276→            {user?.favoriteGenres && user.favoriteGenres.length > 0 && !editing && (
   277→              <div className="flex flex-wrap gap-1.5 mt-3 justify-center sm:justify-start">
   278→                {user.favoriteGenres.slice(0, 5).map(g => (
   279→                  <span key={g} className="px-2.5 py-1 rounded-lg text-[10px] font-semibold uppercase tracking-wider text-white/80" style={{ backgroundColor: `${accentColor}22`, border: `1px solid ${accentColor}33` }}>{g}</span>
   280→                ))}
   281→                {user.favoriteGenres.length > 5 && (
   282→                  <span className="px-2.5 py-1 rounded-lg text-[10px] font-semibold text-white/40 bg-white/[0.06]">+{user.favoriteGenres.length - 5}</span>
   283→                )}
   284→              </div>
   285→            )}
   286→          </div>
   287→
   288→          {/* Actions */}
   289→          <div className="sm:ml-auto flex gap-2">
   290→            <button onClick={startEdit} className="flex items-center gap-2 px-4 py-2.5 rounded-xl text-white/80 text-sm font-medium transition-all hover:brightness-110" style={{ backgroundColor: `${accentColor}22`, border: `1px solid ${accentColor}33` }}>
   291→              <Pencil className="w-4 h-4" /><span className="hidden sm:inline">Edit Profile</span>
   292→            </button>
   293→            <button onClick={async () => { await logout(); goHome(); }} className="flex items-center gap-2 px-4 py-2 rounded-xl bg-white/[0.06] hover:bg-white/10 text-white/60 hover:text-white text-sm transition-all border border-white/[0.06]">
   294→              <LogOut className="w-4 h-4" /><span className="hidden sm:inline">Sign Out</span>
   295→            </button>
   296→          </div>
   297→        </div>
   298→        )}
   299→        {!isLoggedIn && (
   300→          <div className="mb-8">
   301→            <h1 className="text-2xl md:text-3xl font-bold text-white">Browsing History</h1>
   302→            <p className="text-white/40 text-sm mt-1">Your recently viewed content is saved locally. <button onClick={() => setAuthModalOpen(true)} className="text-white/60 hover:text-white underline underline-offset-2 transition-colors">Sign in</button> to sync across devices.</p>
   303→          </div>
   304→        )}
   305→
   306→        {/* Tabs */}
   307→        <div className="flex gap-1 mb-6 border-b border-white/[0.06]">
   308→          {tabsList.map((t) => (
   309→            <button key={t.key} onClick={() => setTab(t.key)} className={`flex items-center gap-2 px-4 py-3 text-sm font-medium transition-all relative border-b-2 -mb-px ${tab === t.key ? 'text-white' : 'text-white/40 border-transparent hover:text-white/70'}`} style={tab === t.key ? { borderColor: accentColor } : undefined}>
   310→              {t.icon}
   311→              <span className="hidden sm:inline">{t.label}</span>
   312→              <span className="sm:hidden">{t.key === 'watchlist' ? watchlist.length : ''}</span>
   313→            </button>
   314→          ))}
   315→        </div>
   316→
   317→        {/* Tab Content */}
   318→        <AnimatePresence mode="wait">
   319→          {tab === 'profile' && (
   320→            <motion.div key="profile-tab" initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -8 }} className="max-w-2xl">
   321→              {editing ? (
   322→                <EditPanel
   323→                  editStep={editStep} setEditStep={setEditStep}
   324→                  editName={editName} setEditName={setEditName}
   325→                  editBio={editBio} setEditBio={setEditBio}
   326→                  selectedAvatar={selectedAvatar} setSelectedAvatar={setSelectedAvatar}
   327→                  selectedColor={selectedColor} setSelectedColor={setSelectedColor}
   328→                  selectedGenres={selectedGenres} toggleGenre={toggleGenre}
   329→                  accentColor={accentColor}
   330→                  saving={saving} saveError={saveError}
   331→                  onSave={saveProfile} onCancel={() => setEditing(false)}
   332→                />
   333→              ) : (
   334→                <div className="space-y-4">
   335→                  <div className="bg-white/[0.03] rounded-2xl border border-white/[0.06] p-5 space-y-4">
   336→                    <h3 className="text-sm font-semibold text-white/70 flex items-center gap-2"><Shield className="w-4 h-4 text-white/50" />Account Info</h3>
   337→                    <div className="space-y-3">
   338→                      <InfoRow label="Name" value={user?.name || '\u2014'} />
   339→                      <InfoRow label="Email" value={user?.email} />
   340→                      <InfoRow label="Avatar" value={avatarDef.name} />
   341→                      <InfoRow label="Bio" value={user?.bio || '\u2014'} truncate />
   342→                      <InfoRow label="Theme" value={user?.accentColor ? `Custom (${ACCENT_COLORS.find(c => c.value === user.accentColor)?.name || 'Custom'})` : `Avatar-matched (${avatarDef.name})`} />
   343→                      <InfoRow label="Favorite Genres" value={user?.favoriteGenres?.length ? user.favoriteGenres.join(', ') : '\u2014'} />
   344→                      <InfoRow label="Member Since" value={memberSince} />
   345→                    </div>
   346→                  </div>
   347→
   348→                  {/* Content Settings */}
   349→                  <div className="bg-white/[0.03] rounded-2xl border border-white/[0.06] p-5 space-y-4">
   350→                    <h3 className="text-sm font-semibold text-white/70 flex items-center gap-2"><ShieldOff className="w-4 h-4 text-white/50" />Content Settings</h3>
   351→                    <div className="flex items-center justify-between">
   352→                      <div>
   353→                        <p className="text-sm text-white/80 font-medium">Adult Content</p>
   354→                        <p className="text-xs text-white/35 mt-0.5">Show 18+ rated titles in a dedicated section</p>
   355→                      </div>
   356→                      <button
   357→                        onClick={async () => {
   358→                          const newVal = !user?.adultEnabled;
   359→                          await updateProfile({ adultEnabled: newVal });
   360→                        }}
   361→                        className={`relative w-11 h-6 rounded-full transition-colors duration-200 ${user?.adultEnabled ? 'bg-red-600' : 'bg-white/15'}`}
   362→                      >
   363→                        <div className={`absolute top-0.5 left-0.5 w-5 h-5 rounded-full bg-white shadow-sm transition-transform duration-200 ${user?.adultEnabled ? 'translate-x-5' : 'translate-x-0'}`} />
   364→                      </button>
   365→                    </div>
   366→                  </div>
   367→                </div>
   368→              )}
   369→            </motion.div>
   370→          )}
   371→
   372→          {tab === 'watchlist' && (
   373→            <motion.div key="watchlist-tab" initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -8 }}>
   374→              {watchlist.length === 0 ? (
   375→                <EmptyState icon={<Bookmark className="w-9 h-9 text-white/10" />} title="No items in watchlist" description="Browse movies and shows, and tap the heart icon to save them here." actionLabel="Browse Content" onAction={goHome} />
   376→              ) : (
   377→                <div className="grid grid-cols-3 sm:grid-cols-4 md:grid-cols-5 lg:grid-cols-6 gap-3 md:gap-4">
   378→                  {watchlist.map((id) => (<WatchlistCard key={id} tmdbId={id} onClick={handleWatchlistClick} />))}
   379→                </div>
   380→              )}
   381→            </motion.div>
   382→          )}
   383→
   384→          {tab === 'history' && (
   385→            <motion.div key="history-tab" initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -8 }}>
   386→              {/* Filter tabs */}
   387→              <div className="flex gap-2 mb-5 flex-wrap">
   388→                {([
   389→                  { key: 'all' as const, label: 'All', icon: <Clock className="w-3.5 h-3.5" /> },
   390→                  { key: 'movie' as const, label: 'Movies', icon: <Film className="w-3.5 h-3.5" /> },
   391→                  { key: 'tv' as const, label: 'TV Shows', icon: <Tv className="w-3.5 h-3.5" /> },
   392→                  { key: 'person' as const, label: 'People', icon: <Users className="w-3.5 h-3.5" /> },
   393→                ]).map(f => {
   394→                  const active = historyFilter === f.key;
   395→                  return (
   396→                    <button
   397→                      key={f.key}
   398→                      onClick={() => handleHistoryFilterChange(f.key)}
   399→                      className={`flex items-center gap-1.5 px-4 py-1.5 rounded-full text-xs font-medium transition-all ${
   400→                        active
   401→                          ? 'text-white border'
   402→                          : 'bg-white/[0.06] text-white/50 hover:text-white/80 border border-transparent hover:border-white/10'
   403→                      }`}
   404→                      style={active ? { backgroundColor: `${accentColor}22`, borderColor: `${accentColor}44`, color: accentColor } : undefined}
   405→                    >
   406→                      {f.icon}
   407→                      {f.label}
   408→                    </button>
   409→                  );
   410→                })}
   411→              </div>
   412→
   413→              {historyLoading ? (
   414→                <div className="flex items-center justify-center py-20"><Loader2 className="w-6 h-6 text-white/50 animate-spin" /></div>
   415→              ) : history.length === 0 ? (
   416→                <EmptyState icon={<Clock className="w-9 h-9 text-white/10" />} title="No browsing history" description="Movies, shows, and people you view will appear here." />
   417→              ) : (
   418→                <>
   419→                  <div className="flex items-center justify-between mb-4">
   420→                    <p className="text-sm text-white/40">{history.length} of {historyTotal} items</p>
   421→                    <button onClick={clearHistory} className="flex items-center gap-1.5 text-sm text-white/50 hover:text-red-400 transition-colors"><Trash2 className="w-3.5 h-3.5" />Clear All</button>
   422→                  </div>
   423→                  <div className="space-y-2">
   424→                    <AnimatePresence mode="popLayout">
   425→                      {history.map((item) => {
   426→                        const isPerson = item.mediaType === 'person';
   427→                        const imgSize = isPerson ? 'w185' : 'w92';
   428→                        const aspectClass = isPerson ? 'aspect-[3/4]' : 'aspect-[2/3]';
   429→                        const imgUrl = item.posterPath ? getImageUrl(item.posterPath, imgSize) : null;
