    16→    16→  getAvatarDef, type AvatarDef,
    17→    17→} from '@/lib/avatars';

    20→    20→import type { BrowseHistoryEntry } from '@/lib/useRecordHistory';

    22→    22→import type { Movie } from '@/lib/types';
    23→    23→

    39→    39→
    40→    40→export function ProfilePage() {
    41→    41→  const { user, loading: authLoading, updateProfile, logout } = useAuthStore();
    42→    42→  const goHome = useAppStore(s => s.goHome);
    43→    43→  const watchlist = useAppStore(s => s.watchlist);
    44→    44→  const selectMovie = useAppStore(s => s.selectMovie);
    45→    45→  const selectTv = useAppStore(s => s.selectTv);
    46→    46→  const selectPerson = useAppStore(s => s.selectPerson);
    47→    47→
    48→    48→  const [tab, setTab] = useState<Tab>('profile');
    49→    49→  const [authModalOpen, setAuthModalOpen] = useState(false);
    50→    50→
    51→    51→  // Profile editing
    52→    52→  const [editing, setEditing] = useState(false);
    53→    53→  const [editStep, setEditStep] = useState<EditStep>('main');
    54→    54→  const [editName, setEditName] = useState('');
    55→    55→  const [editBio, setEditBio] = useState('');
    56→    56→  const [selectedAvatar, setSelectedAvatar] = useState('hero');
    57→    57→  const [selectedColor, setSelectedColor] = useState('#e11d48');
    58→    58→  const [selectedGenres, setSelectedGenres] = useState<string[]>([]);
    59→    59→  const [saving, setSaving] = useState(false);
    60→    60→  const [saveError, setSaveError] = useState('');
    61→    61→
    62→    62→  // Browse history
    63→    63→  const [history, setHistory] = useState<BrowseHistoryItem[]>([]);
    64→    64→  const [historyLoading, setHistoryLoading] = useState(false);
    65→    65→  const [historyFilter, setHistoryFilter] = useState<HistoryFilter>('all');
    66→    66→  const [historyPage, setHistoryPage] = useState(1);
    67→    67→  const [historyTotal, setHistoryTotal] = useState(0);
    68→    68→  const [historyLoadMore, setHistoryLoadMore] = useState(false);
    69→    69→  const [deletingId, setDeletingId] = useState<string | null>(null);
    70→    70→
    71→    71→  // Profile stats from API
    72→    72→  const [profileStats, setProfileStats] = useState<{ watchHistoryCount: number } | null>(null);
    73→    73→
    74→    74→  // Computed accent color for UI
    75→    75→  const accentColor = useMemo(() => {
    76→    76→    if (user?.accentColor) return user.accentColor;
    77→    77→    const def = getAvatarDef(user?.avatar);

    80→    80→
    81→    81→  const fetchProfile = useCallback(async () => {
    82→    82→    if (!user) return;
    83→    83→    try {
    84→    84→      const res = await fetch('/api/profile');
    85→    85→      const data = await res.json();
    86→    86→      if (res.ok) setProfileStats(data.stats);
    87→    87→    } catch { /* ignore */ }
    88→    88→  }, [user]);
    89→    89→
    90→    90→  // Watch history from localStorage
    91→    91→  const [watchItems, setWatchItems] = useState<WatchHistoryEntry[]>([]);
    92→    92→
    93→    93→  const fetchHistory = useCallback(async (filter: HistoryFilter, page: number, append: boolean) => {
    94→    94→    // Always load watch history from localStorage
    95→    95→    let wh = getWatchHistory();
    96→    96→    if (filter !== 'all') wh = wh.filter((i) => i.mediaType === filter);
    97→    97→    setWatchItems(wh);
    98→    98→
    99→    99→    // Non-authenticated users: use localStorage for browse history
   100→   100→    if (!user) {
   101→   101→      setHistoryLoading(true);
   102→   102→      let local = getLocalBrowseHistory();
   103→   103→      if (filter !== 'all') local = local.filter((i) => i.mediaType === filter);
   104→   104→      setHistory(local);
   105→   105→      setHistoryTotal(local.length + wh.length);
   106→   106→      setHistoryLoading(false);
   107→   107→      return;
   108→   108→    }
   109→   109→    // Authenticated: use server API for browse history
   110→   110→    if (append) { setHistoryLoadMore(true); } else { setHistoryLoading(true); }
   111→   111→    try {
   112→   112→      const params = new URLSearchParams({ limit: '50', offset: String((page - 1) * 50) });
   113→   113→      if (filter !== 'all') params.set('type', filter);
   114→   114→      const res = await fetch(`/api/history?${params}`);
   115→   115→      const data = await res.json();
   116→   116→      if (res.ok) {
   117→   117→        setHistory(prev => append ? [...prev, ...data.items] : data.items);
   118→   118→        setHistoryTotal(data.total + wh.length);
   119→   119→        if (!append) setHistoryPage(1);
   120→   120→      }
   121→   121→    } catch { /* ignore */ }
   122→   122→    setHistoryLoading(false);
   123→   123→    setHistoryLoadMore(false);
   124→   124→  }, [user]);
   125→   125→
   126→   126→  useEffect(() => {
   127→   127→    if (tab === 'profile') fetchProfile();
   128→   128→    if (tab === 'history') fetchHistory(historyFilter, 1, false);
   129→   129→  }, [tab, historyFilter, user, fetchHistory, fetchProfile]);
   130→   130→
   131→   131→  useEffect(() => {
   132→   132→    window.scrollTo({ top: 0, behavior: 'smooth' });
   133→   133→  }, []);
   134→   134→
   135→   135→  const startEdit = () => {
   136→   136→    setEditName(user?.name || '');
   137→   137→    setEditBio(user?.bio || '');
   138→   138→    setSelectedAvatar(user?.avatar || 'hero');
   139→   139→    setSelectedColor(user?.accentColor || getAvatarDef(user?.avatar).color);
   140→   140→    setSelectedGenres(user?.favoriteGenres || []);
   141→   141→    setEditStep('main');
   142→   142→    setEditing(true);
   143→   143→    setSaveError('');
   144→   144→  };
   145→   145→
   146→   146→  const saveProfile = async () => {
   147→   147→    setSaving(true);
   148→   148→    setSaveError('');
   149→   149→    const result = await updateProfile({
   150→   150→      name: editName,
   151→   151→      bio: editBio,
   152→   152→      avatar: selectedAvatar,
   153→   153→      accentColor: selectedColor,
   154→   154→      favoriteGenres: selectedGenres,
   155→   155→    });
   156→   156→    setSaving(false);
   157→   157→    if (result.error) {
   158→   158→      setSaveError(result.error);
   159→   159→    } else {
   160→   160→      setEditing(false);
   161→   161→    }
   162→   162→  };
   163→   163→
   164→   164→  const toggleGenre = (genre: string) => {
   165→   165→    setSelectedGenres(prev =>
   166→   166→      prev.includes(genre) ? prev.filter(g => g !== genre) : [...prev, genre]
   167→   167→    );
   168→   168→  };
   169→   169→
   170→   170→  const clearHistory = async () => {
   171→   171→    if (!confirm('Clear all history? This cannot be undone.')) return;
   172→   172→    if (user) {
   173→   173→      await fetch('/api/history', { method: 'DELETE' });
   174→   174→    } else {
   175→   175→      clearLocalHistory();
   176→   176→    }
   177→   177→    clearWatchHistory();
   178→   178→    setHistory([]);
   179→   179→    setWatchItems([]);
   180→   180→    setHistoryTotal(0);
   181→   181→  };
   182→   182→
   183→   183→  const handleDeleteHistoryItem = async (id: string, e: React.MouseEvent) => {
   184→   184→    e.stopPropagation();
   185→   185→    setDeletingId(id);
   186→   186→    if (user) {
   187→   187→      await fetch(`/api/history?id=${id}`, { method: 'DELETE' });
   188→   188→    } else {
   189→   189→      deleteLocalHistoryItem(id);
   190→   190→    }
   191→   191→    setHistory(prev => prev.filter(item => item.id !== id));
   192→   192→    setHistoryTotal(prev => prev - 1);
   193→   193→    setDeletingId(null);
   194→   194→  };
   195→   195→
   196→   196→  const handleDeleteWatchItem = (tmdbId: number, mediaType: string, e: React.MouseEvent) => {
   197→   197→    e.stopPropagation();
   198→   198→    removeWatchHistory(tmdbId, mediaType);
   199→   199→    setWatchItems(prev => prev.filter(item => !(item.tmdbId === tmdbId && item.mediaType === mediaType)));
   200→   200→    setHistoryTotal(prev => prev - 1);
   201→   201→  };
   202→   202→
   203→   203→  const handleHistoryClick = (item: BrowseHistoryItem) => {
   204→   204→    if (item.mediaType === 'person') {
   205→   205→      selectPerson({ id: item.tmdbId, name: item.title, profilePath: item.posterPath });
   206→   206→      return;
   207→   207→    }
   208→   208→    const media: Movie = {
   209→   209→      id: item.tmdbId, title: item.title, name: item.title,
   210→   210→      poster_path: item.posterPath, media_type: item.mediaType,
   211→   211→      vote_average: 0, genre_ids: [], overview: '', popularity: 0,
   212→   212→      release_date: '', first_air_date: '', backdrop_path: null, original_language: '',
   213→   213→    };
   214→   214→    if (item.mediaType === 'tv') selectTv(media); else selectMovie(media);
   215→   215→  };
   216→   216→
   217→   217→  const handleWatchItemClick = (item: WatchHistoryEntry) => {
   218→   218→    const media: Movie = {
   219→   219→      id: item.tmdbId, title: item.title, name: item.title,
   220→   220→      poster_path: item.posterPath, media_type: item.mediaType,
   221→   221→      vote_average: 0, genre_ids: [], overview: '', popularity: 0,
   222→   222→      release_date: '', first_air_date: '', backdrop_path: item.backdropPath, original_language: '',
   223→   223→    };
   224→   224→    if (item.mediaType === 'tv') selectTv(media); else selectMovie(media);
   225→   225→  };
   226→   226→
   227→   227→  const handleHistoryFilterChange = (f: HistoryFilter) => {
   228→   228→    if (f === historyFilter) return;
   229→   229→    setHistoryFilter(f);
   230→   230→  };
   231→   231→
   232→   232→  const loadMoreHistory = () => {
   233→   233→    const nextPage = historyPage + 1;
   234→   234→    setHistoryPage(nextPage);
   235→   235→    fetchHistory(historyFilter, nextPage, true);
   236→   236→  };
   237→   237→
   238→   238→  const handleWatchlistClick = (tmdbId: number, mediaType: string) => {
   239→   239→    const item: Movie = {
   240→   240→      id: tmdbId, title: '', name: '', poster_path: null,
   241→   241→      media_type: mediaType as 'movie' | 'tv', vote_average: 0, genre_ids: [],
   242→   242→      overview: '', popularity: 0, release_date: '', first_air_date: '',
   243→   243→      backdrop_path: null, original_language: '',
   244→   244→    };
   245→   245→    if (mediaType === 'tv') selectTv(item); else selectMovie(item);
   246→   246→  };
   247→   247→
   248→   248→  // Not logged in — but still show the page with limited tabs
   249→   249→  const isLoggedIn = !!user && !authLoading;
   250→   250→
   251→   251→  const memberSince = user ? new Date(user.createdAt || Date.now()).toLocaleDateString('en-US', { month: 'long', year: 'numeric' }) : '';
   252→   252→  const avatarDef = getAvatarDef(user?.avatar);
   253→   253→
   254→   254→  // Default to history tab when not logged in
   255→   255→  useEffect(() => {
   256→   256→    if (!isLoggedIn && tab === 'profile') setTab('history');
   257→   257→  }, [isLoggedIn]);
   258→   258→
   259→   259→  const tabsList: { key: Tab; label: string; icon: React.ReactNode }[] = [
   260→   260→    ...(isLoggedIn ? [{ key: 'profile' as Tab, label: 'Profile', icon: <User className="w-4 h-4" /> }] : []),
   261→   261→    ...(isLoggedIn ? [{ key: 'watchlist' as Tab, label: `Watchlist (${watchlist.length})`, icon: <Bookmark className="w-4 h-4" /> }] : []),
   262→   262→    { key: 'history', label: `History${historyTotal > 0 ? ` (${historyTotal})` : ''}`, icon: <Clock className="w-4 h-4" /> },
   263→   263→  ];
   264→   264→
   265→   265→  return (
   266→   266→    <div className="min-h-screen pb-10">
   267→   267→      {/* Back button */}
   268→   268→      <div className="pt-20 px-4 md:px-8">
   269→   269→        <button onClick={goHome} className="flex items-center gap-2 text-white/50 hover:text-white mb-6 transition-colors group">
   270→   270→          <div className="w-8 h-8 rounded-full bg-white/[0.06] flex items-center justify-center group-hover:bg-white/10 transition-colors">
   271→   271→            <ArrowLeft className="w-4 h-4" />
   272→   272→          </div>
   273→   273→          <span className="text-sm font-medium">Back</span>
   274→   274→        </button>
   275→   275→
   276→   276→        {/* ── Profile Header (logged in only) ── */}
   277→   277→        {isLoggedIn && (
   278→   278→        <div className="flex flex-col sm:flex-row items-center sm:items-start gap-6 mb-8">
   279→   279→          {/* Avatar */}
   280→   280→          <motion.div className="relative" initial={{ scale: 0.8, opacity: 0 }} animate={{ scale: 1, opacity: 1 }} transition={{ type: 'spring', stiffness: 300, damping: 20 }}>
   281→   281→            <div className="w-28 h-28 rounded-2xl overflow-hidden shadow-2xl" style={{ boxShadow: `0 12px 40px ${accentColor}44` }}>
   282→   282→              <ProfileAvatar slug={user?.avatar} size={112} />
   283→   283→            </div>
   284→   284→            {!editing && (
   285→   285→              <button onClick={startEdit} className="absolute -bottom-1 -right-1 w-9 h-9 rounded-xl bg-white/10 backdrop-blur-sm border border-white/20 flex items-center justify-center hover:bg-white/20 transition-colors" style={{ boxShadow: `0 4px 12px ${accentColor}22` }}>
   286→   286→                <Pencil className="w-3.5 h-3.5 text-white" />
   287→   287→              </button>
   288→   288→            )}
   289→   289→            {/* Online indicator */}
   290→   290→            <div className="absolute -bottom-0.5 -left-0.5 w-4 h-4 rounded-full border-2 border-zinc-950" style={{ backgroundColor: '#22c55e' }} />
   291→   291→          </motion.div>
   292→   292→
   293→   293→          {/* Info */}
   294→   294→          <div className="text-center sm:text-left flex-1">
   295→   295→            <h1 className="text-2xl md:text-3xl font-bold text-white">{user?.name || 'StreamVault User'}</h1>
   296→   296→            <p className="text-white/40 text-sm mt-1">{user?.email}</p>
   297→   297→            {user?.bio && !editing && <p className="text-white/60 text-sm mt-2 max-w-md">{user.bio}</p>}
   298→   298→            <div className="flex flex-wrap items-center justify-center sm:justify-start gap-3 mt-3 text-xs text-white/50">
   299→   299→              <span className="flex items-center gap-1.5"><Calendar className="w-3.5 h-3.5" />Joined {memberSince}</span>
   300→   300→              <span className="flex items-center gap-1.5"><Bookmark className="w-3.5 h-3.5" />{watchlist.length} saved</span>
   301→   301→              {profileStats && <span className="flex items-center gap-1.5"><Clock className="w-3.5 h-3.5" />{profileStats.watchHistoryCount} watched</span>}
   302→   302→            </div>
   303→   303→            {/* Favorite genres badges */}
   304→   304→            {user?.favoriteGenres && user.favoriteGenres.length > 0 && !editing && (
   305→   305→              <div className="flex flex-wrap gap-1.5 mt-3 justify-center sm:justify-start">
   306→   306→                {user.favoriteGenres.slice(0, 5).map(g => (
   307→   307→                  <span key={g} className="px-2.5 py-1 rounded-lg text-[10px] font-semibold uppercase tracking-wider text-white/80" style={{ backgroundColor: `${accentColor}22`, border: `1px solid ${accentColor}33` }}>{g}</span>
   308→   308→                ))}
   309→   309→                {user.favoriteGenres.length > 5 && (
   310→   310→                  <span className="px-2.5 py-1 rounded-lg text-[10px] font-semibold text-white/40 bg-white/[0.06]">+{user.favoriteGenres.length - 5}</span>
   311→   311→                )}
   312→   312→              </div>
   313→   313→            )}
   314→   314→          </div>
   315→   315→
   316→   316→          {/* Actions */}
   317→   317→          <div className="sm:ml-auto flex gap-2">
   318→   318→            <button onClick={startEdit} className="flex items-center gap-2 px-4 py-2.5 rounded-xl text-white/80 text-sm font-medium transition-all hover:brightness-110" style={{ backgroundColor: `${accentColor}22`, border: `1px solid ${accentColor}33` }}>
   319→   319→              <Pencil className="w-4 h-4" /><span className="hidden sm:inline">Edit Profile</span>
   320→   320→            </button>
   321→   321→            <button onClick={async () => { await logout(); goHome(); }} className="flex items-center gap-2 px-4 py-2 rounded-xl bg-white/[0.06] hover:bg-white/10 text-white/60 hover:text-white text-sm transition-all border border-white/[0.06]">
   322→   322→              <LogOut className="w-4 h-4" /><span className="hidden sm:inline">Sign Out</span>
   323→   323→            </button>
   324→   324→          </div>
   325→   325→        </div>
   326→   326→        )}
   327→   327→        {!isLoggedIn && (
   328→   328→          <div className="mb-8">
   329→   329→            <h1 className="text-2xl md:text-3xl font-bold text-white">Browsing History</h1>
   330→   330→            <p className="text-white/40 text-sm mt-1">Your recently viewed content is saved locally. <button onClick={() => setAuthModalOpen(true)} className="text-white/60 hover:text-white underline underline-offset-2 transition-colors">Sign in</button> to sync across devices.</p>
   331→   331→          </div>
   332→   332→        )}
   333→   333→
   334→   334→        {/* Tabs */}
   335→   335→        <div className="flex gap-1 mb-6 border-b border-white/[0.06]">
   336→   336→          {tabsList.map((t) => (
   337→   337→            <button key={t.key} onClick={() => setTab(t.key)} className={`flex items-center gap-2 px-4 py-3 text-sm font-medium transition-all relative border-b-2 -mb-px ${tab === t.key ? 'text-white' : 'text-white/40 border-transparent hover:text-white/70'}`} style={tab === t.key ? { borderColor: accentColor } : undefined}>
   338→   338→              {t.icon}
   339→   339→              <span className="hidden sm:inline">{t.label}</span>
   340→   340→              <span className="sm:hidden">{t.key === 'watchlist' ? watchlist.length : ''}</span>
   341→   341→            </button>
   342→   342→          ))}
   343→   343→        </div>
   344→   344→
   345→   345→        {/* Tab Content */}
   346→   346→        <AnimatePresence mode="wait">
   347→   347→          {tab === 'profile' && (
   348→   348→            <motion.div key="profile-tab" initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -8 }} className="max-w-2xl">
   349→   349→              {editing ? (
   350→   350→                <EditPanel
   351→   351→                  editStep={editStep} setEditStep={setEditStep}
   352→   352→                  editName={editName} setEditName={setEditName}
   353→   353→                  editBio={editBio} setEditBio={setEditBio}
   354→   354→                  selectedAvatar={selectedAvatar} setSelectedAvatar={setSelectedAvatar}
   355→   355→                  selectedColor={selectedColor} setSelectedColor={setSelectedColor}
   356→   356→                  selectedGenres={selectedGenres} toggleGenre={toggleGenre}
   357→   357→                  accentColor={accentColor}
   358→   358→                  saving={saving} saveError={saveError}
   359→   359→                  onSave={saveProfile} onCancel={() => setEditing(false)}
   360→   360→                />
   361→   361→              ) : (
   362→   362→                <div className="space-y-4">
   363→   363→                  <div className="bg-white/[0.03] rounded-2xl border border-white/[0.06] p-5 space-y-4">
   364→   364→                    <h3 className="text-sm font-semibold text-white/70 flex items-center gap-2"><Shield className="w-4 h-4 text-white/50" />Account Info</h3>
   365→   365→                    <div className="space-y-3">
   366→   366→                      <InfoRow label="Name" value={user?.name || '\u2014'} />
   367→   367→                      <InfoRow label="Email" value={user?.email} />
   368→   368→                      <InfoRow label="Avatar" value={avatarDef.name} />
   369→   369→                      <InfoRow label="Bio" value={user?.bio || '\u2014'} truncate />
   370→   370→                      <InfoRow label="Theme" value={user?.accentColor ? `Custom (${ACCENT_COLORS.find(c => c.value === user.accentColor)?.name || 'Custom'})` : `Avatar-matched (${avatarDef.name})`} />
   371→   371→                      <InfoRow label="Favorite Genres" value={user?.favoriteGenres?.length ? user.favoriteGenres.join(', ') : '\u2014'} />
   372→   372→                      <InfoRow label="Member Since" value={memberSince} />
   373→   373→                    </div>
   374→   374→                  </div>
   375→   375→
   376→   376→                  {/* Content Settings */}
   377→   377→                  <div className="bg-white/[0.03] rounded-2xl border border-white/[0.06] p-5 space-y-4">
