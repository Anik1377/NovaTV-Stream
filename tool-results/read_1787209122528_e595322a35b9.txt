    35→    35→/* ------------------------------------------------------------------ */
    36→    36→/*  iOS native fullscreen detection                                    */
    37→    37→/* ------------------------------------------------------------------ */
    38→    38→function useIOSFullscreenDetect(onReturn: () => void) {
    39→    39→  useEffect(() => {
    40→    40→    const handleEnd = () => onReturn();
    41→    41→    document.addEventListener('webkitendfullscreen', handleEnd);
    42→    42→    return () => document.removeEventListener('webkitendfullscreen', handleEnd);
    43→    43→  }, [onReturn]);
    44→    44→}
    45→    45→
    46→    46→/* ------------------------------------------------------------------ */
    47→    47→/*  Score badge helper                                                 */
    48→    48→/* ------------------------------------------------------------------ */
    49→    49→function scoreBadge(score: number) {
    50→    50→  if (score >= 80) return { label: 'Top Pick', cls: 'bg-emerald-500/20 text-emerald-400' };
    51→    51→  if (score >= 40) return { label: 'Popular', cls: 'bg-amber-500/20 text-amber-400' };
    52→    52→  if (score > 0) return { label: 'Used', cls: 'bg-zinc-500/20 text-zinc-400' };
    53→    53→  return null;
    54→    54→}
