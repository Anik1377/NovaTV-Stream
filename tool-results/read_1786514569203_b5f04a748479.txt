    53→function formatDuration(iso: string): string {
    54→  const m = iso.match(/PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?/);
    55→  if (!m) return '0:00';
    56→  const h = parseInt(m[1] || '0');
    57→  const min = parseInt(m[2] || '0');
    58→  const s = parseInt(m[3] || '0');
    59→  if (h > 0) return `${h}:${min.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    60→  return `${min}:${s.toString().padStart(2, '0')}`;
    61→}
    62→
    63→function formatDurationSeconds(iso: string): number {
    64→  const m = iso.match(/PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?/);
    65→  if (!m) return 0;
    66→  const h = parseInt(m[1] || '0');
    67→  const min = parseInt(m[2] || '0');
    68→  const s = parseInt(m[3] || '0');
    69→  return h * 3600 + min * 60 + s;
    70→}
    71→
    72→function formatTime(sec: number): string {
    73→  const m = Math.floor(sec / 60);
    74→  const s = Math.floor(sec % 60);
    75→  return `${m}:${s.toString().padStart(2, '0')}`;
    76→}
    77→
    78→function formatViews(n: number): string {
    79→  if (n >= 1_000_000_000) return `${(n / 1_000_000_000).toFixed(1)}B`;
    80→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
    81→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`;
    82→  return String(n);
    83→}

   100→function useYouTubePlayer(videoId: string | null) {
   101→  const playerRef = useRef<any>(null);
   102→  const containerRef = useRef<HTMLDivElement>(null);
   103→  const [state, setState] = useState<PlayerState>({ isPlaying: false, currentTime: 0, duration: 0 });
   104→  const onEndRef = useRef<() => void>(() => {});
   105→
   106→  // Load the IFrame API script once globally
   107→  useEffect(() => {
   108→    if (window.YT?.Player) return;
   109→    const existing = document.querySelector('script[src*="youtube.com/iframe_api"]');
   110→    if (!existing) {
   111→      const s = document.createElement('script');
   112→      s.src = 'https://www.youtube.com/iframe_api';
   113→      document.head.appendChild(s);
   114→    }
   115→    window.onYouTubeIframeAPIReady = () => {};
   116→  }, []);
   117→
   118→  // Create / destroy player when videoId changes
   119→  useEffect(() => {
   120→    if (!videoId || !containerRef.current) {
   121→      if (playerRef.current) {
   122→        try { playerRef.current.destroy(); } catch { /* ignore */ }
   123→        playerRef.current = null;
   124→      }
   125→      return;
   126→    }
   127→
   128→    let destroyed = false;
   129→
   130→    const tryCreate = (): boolean => {
   131→      if (destroyed || !containerRef.current || !window.YT?.Player) return false;
   132→
   133→      // Destroy previous player
   134→      if (playerRef.current) {
   135→        try { playerRef.current.destroy(); } catch { /* ignore */ }
   136→        playerRef.current = null;
   137→      }
   138→
   139→      playerRef.current = new window.YT.Player(containerRef.current, {
   140→        videoId,
   141→        height: '1',
   142→        width: '1',
   143→        playerVars: {
   144→          autoplay: 1,
   145→          controls: 0,
   146→          disablekb: 1,
   147→          fs: 0,
   148→          modestbranding: 1,
   149→          rel: 0,
   150→          showinfo: 0,
   151→        },
   152→        events: {
   153→          onReady: (e: any) => {
   154→            if (!destroyed) {
   155→              e.target.playVideo();
   156→              setState((s) => ({ ...s, duration: e.target.getDuration() || 0 }));
   157→            }
   158→          },
   159→          onStateChange: (e: any) => {
   160→            if (destroyed) return;
   161→            if (e.data === 1) { // PLAYING
   162→              setState((s) => ({ ...s, isPlaying: true, duration: e.target.getDuration() || s.duration }));
   163→            } else if (e.data === 2) { // PAUSED
   164→              setState((s) => ({ ...s, isPlaying: false }));
   165→            } else if (e.data === 0) { // ENDED
   166→              setState((s) => ({ ...s, isPlaying: false }));
   167→              onEndRef.current();
   168→            }
   169→          },
   170→        },
   171→      });
   172→      return true;
   173→    };
   174→
   175→    // Try immediately or poll for API readiness
   176→    if (!tryCreate()) {
   177→      const iv = setInterval(() => {
   178→        if (tryCreate()) clearInterval(iv);
   179→      }, 150);
   180→      return () => clearInterval(iv);
   181→    }
   182→
   183→    // Progress polling
   184→    const progressIv = setInterval(() => {
   185→      if (playerRef.current?.getCurrentTime && !destroyed) {
   186→        setState((s) => ({ ...s, currentTime: playerRef.current.getCurrentTime() }));
   187→      }
   188→    }, 500);
   189→
   190→    return () => {
   191→      destroyed = true;
   192→      clearInterval(progressIv);
   193→      if (playerRef.current) {
   194→        try { playerRef.current.destroy(); } catch { /* ignore */ }
   195→        playerRef.current = null;
   196→      }
   197→    };
   198→  }, [videoId]);
   199→
   200→  const togglePlay = useCallback(() => {
   201→    if (!playerRef.current) return;
   202→    setState((s) => {
   203→      if (s.isPlaying) playerRef.current.pauseVideo();
   204→      else playerRef.current.playVideo();
   205→      return { ...s, isPlaying: !s.isPlaying };
   206→    });
   207→  }, []);
   208→
   209→  const seekTo = useCallback((seconds: number) => {
   210→    playerRef.current?.seekTo(seconds, true);
   211→  }, []);
   212→
   213→  const setOnEnd = useCallback((cb: () => void) => {
   214→    onEndRef.current = cb;
   215→  }, []);
   216→
   217→  return { containerRef, ...state, togglePlay, seekTo, setOnEnd };
   218→}

   531→            className="w-full max-w-[300px] md:max-w-[360px] aspect-square rounded-2xl overflow-hidden shadow-2xl shadow-black/60"
   532→          >
   533→            {!imgError ? (
