   297→function FullScreenPlayer({
   298→  track,
   299→  onClose,
   300→  onNext,
   301→  onPrev,
   302→  isPlaying,
   303→  onTogglePlay,
   304→  currentTime,
   305→  duration,
   306→  onSeek,
   307→}: {
   308→  track: SaavnTrack;
   309→  onClose: () => void;
   310→  onNext: () => void;
   311→  onPrev: () => void;
   312→  isPlaying: boolean;
   313→  onTogglePlay: () => void;
   314→  currentTime: number;
   315→  duration: number;
   316→  onSeek: (pct: number) => void;
   317→}) {
   318→  const [imgError, setImgError] = useState(false);
   319→  const [liked, setLiked] = useState(false);
   320→
   321→  const progress = duration > 0 ? (currentTime / duration) * 100 : 0;
   322→
   323→  return (
   324→    <motion.div
   325→      initial={{ y: '100%' }}
   326→      animate={{ y: 0 }}
   327→      exit={{ y: '100%' }}
   328→      transition={{ type: 'spring', stiffness: 300, damping: 30 }}
   329→      className="fixed inset-0 z-[60] flex flex-col bg-black"
   330→    >
   331→      <div className="absolute inset-0 z-0">
   332→        {!imgError ? (
   333→          <img src={track.thumbnail} alt="" className="w-full h-full object-cover scale-150 blur-3xl opacity-30" onError={() => setImgError(true)} />
   334→        ) : (
   335→          <div className="w-full h-full bg-gradient-to-b from-zinc-900 to-black" />
   336→        )}
   337→        <div className="absolute inset-0 bg-black/50" />
   338→      </div>
   339→
   340→      <div className="relative z-10 flex flex-col h-full">
   341→        <div className="flex items-center justify-between px-5 pt-[env(safe-area-inset-top,12px)] pb-2">
   342→          <button onClick={onClose} className="w-9 h-9 flex items-center justify-center text-white/60 hover:text-white transition-colors">
   343→            <ChevronDown className="w-6 h-6" />
   344→          </button>
   345→          <div className="text-center">
   346→            <p className="text-[11px] font-medium text-white/30 uppercase tracking-wider">Playing from</p>
   347→            <p className="text-xs text-white/50 mt-0.5">JioSaavn</p>
   348→          </div>
   349→          <button onClick={() => setLiked(!liked)} className="w-9 h-9 flex items-center justify-center text-white/60 hover:text-white transition-colors">
   350→            <Heart className={`w-5 h-5 transition-colors ${liked ? 'text-[#FA2D48] fill-[#FA2D48]' : ''}`} />
   351→          </button>
   352→        </div>
   353→
   354→        <div className="flex-1 flex items-center justify-center px-8 py-4">
   355→          <motion.div
   356→            initial={{ scale: 0.8, opacity: 0 }}
   357→            animate={{ scale: 1, opacity: 1 }}
   358→            transition={{ delay: 0.1, type: 'spring', stiffness: 200 }}
   359→            className="w-full max-w-[300px] md:max-w-[360px] aspect-square rounded-2xl overflow-hidden shadow-2xl shadow-black/60"
   360→          >
   361→            {!imgError ? (
   362→              <img src={track.thumbnail} alt={track.title} className="w-full h-full object-cover" onError={() => setImgError(true)} />
   363→            ) : (
   364→              <div className="w-full h-full bg-gradient-to-br from-zinc-800 to-zinc-900 flex items-center justify-center">
   365→                <Music className="w-16 h-16 text-white/20" />
   366→              </div>
   367→            )}
   368→          </motion.div>
   369→        </div>
   370→
   371→        <div className="px-8 pb-[env(safe-area-inset-bottom,16px)]">
   372→          <div className="flex items-start justify-between gap-3 mb-5">
   373→            <div className="min-w-0 flex-1">
   374→              <h2 className="text-xl font-bold text-white truncate">{track.title}</h2>

   376→            </div>
   377→          </div>
   378→
   379→          <div className="mb-5">
   380→            <div
   381→              className="relative w-full h-1 bg-white/10 rounded-full cursor-pointer group"
   382→              onClick={(e) => {
   383→                const rect = e.currentTarget.getBoundingClientRect();
   384→                const pct = ((e.clientX - rect.left) / rect.width) * 100;
   385→                onSeek(Math.max(0, Math.min(100, pct)));
   386→              }}
   387→            >
   388→              <div className="absolute left-0 top-0 h-full bg-white/80 rounded-full group-hover:bg-white transition-colors" style={{ width: `${Math.min(progress, 100)}%` }} />
   389→            </div>

   391→              <span className="text-[11px] text-white/30 tabular-nums">{formatTime(currentTime)}</span>
   392→              <span className="text-[11px] text-white/30 tabular-nums">{formatTime(duration)}</span>
   393→            </div>
   394→          </div>
   395→
   396→          <div className="flex items-center justify-between px-2 mb-5">
   397→            <button className="p-2 text-white/40 hover:text-white/70 transition-colors"><Shuffle className="w-5 h-5" /></button>
   398→            <button onClick={onPrev} className="p-2 text-white hover:text-white transition-colors"><SkipBack className="w-6 h-6 fill-white" /></button>

   401→                <Pause className="w-7 h-7 text-black fill-black" />

   403→                <Play className="w-7 h-7 text-black fill-black ml-1" />

   406→            <button onClick={onNext} className="p-2 text-white hover:text-white transition-colors"><SkipForward className="w-6 h-6 fill-white" /></button>

   408→          </div>
   409→        </div>
   410→      </div>
   411→    </motion.div>
   412→  );
   413→}

   497→        </button>
   498→      )}
   499→    </div>
   500→  );
   501→}

   505→export function MusicPage() {
   506→  const [trending, setTrending] = useState<SaavnTrack[]>([]);
   507→  const [searchResults, setSearchResults] = useState<SaavnTrack[]>([]);
   508→  const [loading, setLoading] = useState(true);
   509→  const [searchLoading, setSearchLoading] = useState(false);
   510→  const [searchQuery, setSearchQuery] = useState('');
   511→  const [activeMood, setActiveMood] = useState('');
   512→  const [currentTrack, setCurrentTrack] = useState<SaavnTrack | null>(null);
   513→  const [showFullScreen, setShowFullScreen] = useState(false);
   514→  const [queue, setQueue] = useState<SaavnTrack[]>([]);
   515→  const [queueIndex, setQueueIndex] = useState(0);
   516→  const [isPlaying, setIsPlaying] = useState(false);
   517→  const [currentTime, setCurrentTime] = useState(0);
   518→  const [audioDuration, setAudioDuration] = useState(0);
   519→
   520→  const searchTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   521→  const searchInputRef = useRef<HTMLInputElement>(null);
   522→  const audioRef = useRef<HTMLAudioElement | null>(null);
   523→
   524→  // Initialize audio element
   525→  useEffect(() => {
   526→    const audio = new Audio();
   527→    audio.preload = 'auto';
   528→
   529→    audio.addEventListener('timeupdate', () => {
   530→      setCurrentTime(audio.currentTime);
   531→    });
   532→    audio.addEventListener('loadedmetadata', () => {
   533→      setAudioDuration(audio.duration);
   534→    });
   535→    audio.addEventListener('ended', () => {
   536→      // Auto next
   537→      setQueueIndex((prev) => {
   538→        const next = (prev + 1) % queue.length;
   539→        if (queue[next]) {
   540→          setCurrentTrack(queue[next]);
   541→          playSong(queue[next]);
   542→        }
   543→        return next;
   544→      });
   545→    });
   546→    audio.addEventListener('play', () => setIsPlaying(true));
   547→    audio.addEventListener('pause', () => setIsPlaying(false));
   548→
   549→    audioRef.current = audio;
   550→    return () => {
   551→      audio.pause();
   552→      audio.src = '';
   553→    };
   554→  }, []);
   555→
   556→  // Update ended handler when queue changes
   557→  useEffect(() => {
   558→    const audio = audioRef.current;
   559→    if (!audio) return;
   560→
   561→    const handleEnded = () => {
   562→      setQueueIndex((prev) => {
   563→        const next = (prev + 1) % queue.length;
   564→        if (queue[next]) {
   565→          setCurrentTrack(queue[next]);
   566→          playSong(queue[next]);
   567→        }
   568→        return next;
   569→      });
   570→    };
   571→    audio.addEventListener('ended', handleEnded);
   572→    return () => audio.removeEventListener('ended', handleEnded);
   573→  }, [queue]);
   574→
   575→  const playSong = useCallback((track: SaavnTrack) => {
   576→    const audio = audioRef.current;
   577→    if (!audio || !track.audioUrl) return;
   578→    audio.src = track.audioUrl;
   579→    audio.play().catch(() => {});
   580→  }, []);
   581→
   582→  // Fetch trending on mount
   583→  useEffect(() => {
   584→    async function fetchTrending() {
   585→      try {
   586→        const res = await fetch('/api/saavn/trending?limit=25');
   587→        const data: ApiResponse = await res.json();
   588→        setTrending(data.results || []);
   589→      } catch (err) {
   590→        console.error('Failed to fetch trending:', err);
   591→      } finally {
   592→        setLoading(false);
   593→      }
   594→    }
   595→    fetchTrending();
   596→  }, []);
   597→
   598→  // Search with debounce
   599→  const handleSearchInput = useCallback((value: string) => {
   600→    setSearchQuery(value);
   601→    setActiveMood('');
   602→    if (searchTimerRef.current) clearTimeout(searchTimerRef.current);
   603→    if (!value.trim()) {
   604→      setSearchResults([]);
   605→      return;
   606→    }
   607→    searchTimerRef.current = setTimeout(async () => {
   608→      setSearchLoading(true);
   609→      try {
   610→        const res = await fetch(`/api/saavn/search?query=${encodeURIComponent(value)}&limit=20`);
   611→        const data: ApiResponse = await res.json();
   612→        setSearchResults(data.results || []);
   613→      } catch {
   614→        setSearchResults([]);
   615→      } finally {
   616→        setSearchLoading(false);
   617→      }
   618→    }, 500);
   619→  }, []);
   620→
   621→  const handleMoodClick = useCallback((mood: typeof MOOD_CATEGORIES[number]) => {
   622→    const isCurrentlyActive = activeMood === mood.label;
   623→    setActiveMood(isCurrentlyActive ? '' : mood.label);
   624→    if (!isCurrentlyActive) {
   625→      handleSearchInput(mood.query);
   626→      if (searchInputRef.current) searchInputRef.current.value = mood.query;
   627→    } else {
   628→      setSearchQuery('');
   629→      setSearchResults([]);
   630→      if (searchInputRef.current) searchInputRef.current.value = '';
   631→    }
   632→  }, [activeMood, handleSearchInput]);
   633→
   634→  const handlePlay = useCallback((track: SaavnTrack, trackList?: SaavnTrack[]) => {
   635→    setCurrentTrack(track);
   636→    setShowFullScreen(true);
   637→    playSong(track);
   638→    if (trackList) {
   639→      const idx = trackList.findIndex((t) => t.id === track.id);
   640→      setQueue(trackList);
   641→      setQueueIndex(idx >= 0 ? idx : 0);
   642→    }
   643→  }, [playSong]);
   644→
   645→  const handleNext = useCallback(() => {
   646→    if (queue.length === 0) return;
   647→    const nextIdx = (queueIndex + 1) % queue.length;
   648→    setQueueIndex(nextIdx);
   649→    setCurrentTrack(queue[nextIdx]);
   650→    playSong(queue[nextIdx]);
   651→  }, [queue, queueIndex, playSong]);
   652→
   653→  const handlePrev = useCallback(() => {
   654→    if (queue.length === 0) return;
   655→    const prevIdx = (queueIndex - 1 + queue.length) % queue.length;
   656→    setQueueIndex(prevIdx);
   657→    setCurrentTrack(queue[prevIdx]);
   658→    playSong(queue[prevIdx]);
   659→  }, [queue, queueIndex, playSong]);
   660→
   661→  const handleTogglePlay = useCallback(() => {
   662→    const audio = audioRef.current;
   663→    if (!audio) return;
   664→    if (audio.paused) {
   665→      audio.play().catch(() => {});
   666→    } else {
   667→      audio.pause();
   668→    }
   669→  }, []);
   670→
   671→  const handleSeek = useCallback((pct: number) => {
   672→    const audio = audioRef.current;
   673→    if (!audio || !audio.duration) return;
   674→    audio.currentTime = (pct / 100) * audio.duration;
   675→  }, []);
   676→
   677→  const handleClosePlayer = useCallback(() => {
   678→    setShowFullScreen(false);
   679→  }, []);
   680→
   681→  const isSearching = searchQuery.trim().length > 0;
   682→  const hasPlayer = currentTrack !== null;
   683→
   684→  const rankedTracks = isSearching ? searchResults.slice(0, 6) : trending.slice(0, 6);
   685→  const listTracks = isSearching ? searchResults.slice(6) : trending.slice(6);
   686→
   687→  return (
   688→    <div className="relative min-h-screen">
   689→      <div
   690→        className="fixed inset-0 pointer-events-none z-0"
   691→        style={{
   692→          background: 'radial-gradient(ellipse at 70% -10%, rgba(138,42,42,0.35) 0%, rgba(45,21,21,0.15) 30%, transparent 65%)',
   693→        }}
   694→      />
   695→
   696→      <div className="sticky top-0 z-30 bg-black/80 backdrop-blur-xl">
   697→        <div className="px-4 pt-12 md:pt-3 pb-3">
   698→          <div className="relative max-w-md">
   699→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-white/30" />
   700→            <input
   701→              ref={searchInputRef}
   702→              type="text"
   703→              placeholder="Search songs, artists..."
   704→              defaultValue={searchQuery}
   705→              onChange={(e) => handleSearchInput(e.target.value)}
   706→              className="w-full pl-10 pr-10 h-9 rounded-lg bg-[#1C1C1C] border-0 text-white placeholder:text-white/30 text-[15px] font-normal focus:outline-none focus:ring-1 focus:ring-white/20 transition-shadow"
   707→            />
   708→            {searchQuery && (
   709→              <button
   710→                onClick={() => {
   711→                  handleSearchInput('');
   712→                  if (searchInputRef.current) searchInputRef.current.value = '';
   713→                }}
   714→                className="absolute right-3 top-1/2 -translate-y-1/2 text-white/30 hover:text-white/60 transition-colors"
   715→              >
   716→                <X className="w-4 h-4" />
   717→              </button>
   718→            )}
   719→          </div>
   720→        </div>
   721→      </div>
   722→
   723→      <div className="relative z-10">
   724→        {!isSearching && (
   725→          <div className="px-4 mb-8 mt-2">
   726→            <SectionHeader title="Browse" />
   727→            <div className="flex gap-3 overflow-x-auto pb-2 -mx-4 px-4 content-scroll">
   728→              {loading
   729→                ? Array.from({ length: 5 }).map((_, i) => <CategorySkeleton key={i} />)
   730→                : MOOD_CATEGORIES.map((mood) => (
   731→                    <CategoryCard
   732→                      key={mood.label}
   733→                      label={mood.label}
   734→                      icon={mood.icon}
   735→                      color={mood.color}
   736→                      onClick={() => handleMoodClick(mood)}
   737→                      isActive={activeMood === mood.label}
   738→                    />
   739→                  ))
   740→              }
   741→            </div>
   742→          </div>
   743→        )}
   744→
   745→        {isSearching && (
   746→          <div className="px-4 mb-6 mt-2">
   747→            <div className="flex gap-2 overflow-x-auto pb-1 -mx-4 px-4 content-scroll">
   748→              {GENRE_PILLS.map((genre) => (
   749→                <button
   750→                  key={genre}
   751→                  onClick={() => {
   752→                    handleSearchInput(genre);
   753→                    if (searchInputRef.current) searchInputRef.current.value = genre;
   754→                  }}
   755→                  className="shrink-0 px-3.5 py-1.5 rounded-full text-[13px] font-medium bg-white/[0.06] text-white/50 border border-white/[0.06] hover:bg-white/10 hover:text-white/80 active:bg-white/15 active:text-white transition-colors"
   756→                >
   757→                  {genre}
   758→                </button>

   766→            <p className="text-[22px] md:text-[24px] font-normal text-[#CCCCCC] leading-relaxed">
   767→              Discover what&apos;s trending in <span className="text-white font-bold">music right now</span>.
   768→            </p>
   769→          </div>

   772→        {isSearching ? (

   790→                  />

   792→              </div>
   793→            </div>
   794→          ) : null
   795→        ) : loading ? (
   796→          <div className="px-4 mb-8">
   797→            <SectionHeader title="Trending Now" />
   798→            <div className="flex gap-3 overflow-x-auto pb-2 -mx-4 px-4 content-scroll">
   799→              {Array.from({ length: 4 }).map((_, i) => <RankCardSkeleton key={i} />)}
   800→            </div>
   801→          </div>

   809→                  track={track}
   810→                  rank={i + 1}
   811→                  onPlay={() => handlePlay(track, trending)}

   840→                <div className="text-center py-20">
   841→                  <Music className="w-10 h-10 text-white/10 mx-auto mb-3" />
   842→                  <p className="text-white/25 text-sm">No results found</p>

   849→                <div className="rounded-xl overflow-hidden -mx-4">
   850→                  {Array.from({ length: 6 }).map((_, i) => <TrackRowSkeleton key={i} />)}
   851→                </div>
   852→              ) : listTracks.length > 0 ? (

   902→            onTogglePlay={handleTogglePlay}
   903→            currentTime={currentTime}
   904→            duration={audioDuration}
