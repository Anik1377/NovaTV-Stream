     1→     1→'use client';
     2→     2→
     3→     3→import { useState, useRef, useEffect, useCallback, useMemo } from 'react';
     4→     4→import {
     5→     5→  X,
     6→     6→  Maximize2,
     7→     7→  Minimize2,
     8→     8→  RefreshCw,
     9→     9→  ChevronLeft,
    10→    10→  ShieldCheck,
    11→    11→  Server,
    12→    12→  Check,
    13→    13→  Loader2,
    14→    14→  ArrowRight,
    15→    15→} from 'lucide-react';
    16→    16→import { motion, AnimatePresence, useMotionValue, useTransform, animate } from 'framer-motion';
    17→    17→import { providers, getProvider, getEmbedUrl, type Provider } from '@/lib/providers';
    18→    18→import { getRankedProviderIds, getServerScore, recordServerPick } from '@/lib/server-rankings';
    19→    19→import { useAppStore } from '@/store/app-store';
    20→    20→import { useAuthStore } from '@/store/auth-store';
    21→    21→import { useIsIOS } from '@/hooks/use-ios';
    22→    22→import { useIsMobile } from '@/hooks/use-mobile';
    23→    23→import { recordWatchHistory } from '@/lib/watch-history';
    24→    24→
    25→    25→interface VideoPlayerProps {
    26→    26→  src: string;
    27→    27→  title?: string;
    28→    28→  onClose: () => void;
    29→    29→  mediaType: 'movie' | 'tv';
    30→    30→  tmdbId: number;
    31→    31→  season?: number;
    32→    32→  episode?: number;
    33→    33→}
    34→    34→
    35→    35→/* ------------------------------------------------------------------ */
    36→    36→/*  iOS native fullscreen detection                                    */
    37→    37→/* ------------------------------------------------------------------ */
    38→    38→function useIOSFullscreenDetect(onReturn: () => void) {
    39→    39→  useEffect(() => {
    40→    40→    const handleEnd = () => onReturn();
    41→    41→    document.addEventListener('webkitendfullscreen', handleEnd);
    42→    42→    return () => document.removeEventListener('webkitendfullscreen', handleEnd);
    43→    43→  }, [onReturn]);
    44→    44→}
    45→    45→
    46→    46→/* ------------------------------------------------------------------ */
    47→    47→/*  Score badge helper                                                 */
    48→    48→/* ------------------------------------------------------------------ */
    49→    49→function scoreBadge(score: number) {
    50→    50→  if (score >= 80) return { label: 'Top Pick', cls: 'bg-emerald-500/20 text-emerald-400' };
    51→    51→  if (score >= 40) return { label: 'Popular', cls: 'bg-amber-500/20 text-amber-400' };
    52→    52→  if (score > 0) return { label: 'Used', cls: 'bg-zinc-500/20 text-zinc-400' };
    53→    53→  return null;
    54→    54→}

    56→    56→/* ------------------------------------------------------------------ */
    57→    57→/*  Main component                                                     */
    58→    58→/* ------------------------------------------------------------------ */
    59→    59→export function VideoPlayer({
    60→    60→  src,
    61→    61→  title,
    62→    62→  onClose,
    63→    63→  mediaType,
    64→    64→  tmdbId,
    65→    65→  season,
    66→    66→  episode,
    67→    67→}: VideoPlayerProps) {
    68→    68→  const [loading, setLoading] = useState(true);
    69→    69→  const [error, setError] = useState(false);
    70→    70→  const [isFullscreen, setIsFullscreen] = useState(false);
    71→    71→  const [mobileSheetOpen, setMobileSheetOpen] = useState(false);
    72→    72→  const [currentSrc, setCurrentSrc] = useState(src);
    73→    73→  const [switchingTo, setSwitchingTo] = useState<string | null>(null);
    74→    74→
    75→    75→  const containerRef = useRef<HTMLDivElement>(null);
    76→    76→  const iframeRef = useRef<HTMLIFrameElement>(null);
    77→    77→  const { selectedProvider, setSelectedProvider, selectedMovie, selectedTv } = useAppStore();
    78→    78→  const user = useAuthStore((s) => s.user);
    79→    79→  const isIOS = useIsIOS();
    80→    80→  const isMobile = useIsMobile();
    81→    81→
    82→    82→  // Swipe-to-dismiss (mobile only)
    83→    83→  const dragY = useMotionValue(0);
    84→    84→  const iframeScale = useTransform(dragY, [0, 250], [1, 0.92]);
    85→    85→  const iframeRadius = useTransform(dragY, [0, 250], [8, 24]);
    86→    86→
    87→    87→  /* ---------------------------------------------------------------- */
    88→    88→  /*  Ranked provider list                                              */
    89→    89→  /* ---------------------------------------------------------------- */
    90→    90→  const rankedIds = useMemo(() => getRankedProviderIds(), []);
    91→    91→  const rankedSet = useMemo(() => new Set(rankedIds), [rankedIds]);
    92→    92→
    93→    93→  const sortedProviders = useMemo(() => {
    94→    94→    const ranked = rankedIds
    95→    95→      .map((id) => providers.find((p) => p.id === id))
    96→    96→      .filter((p): p is Provider => !!p);
    97→    97→    const unranked = providers.filter((p) => !rankedSet.has(p.id));
    98→    98→    return [...ranked, ...unranked];
    99→    99→  }, [rankedIds, rankedSet]);
   100→   100→
   101→   101→  const activeProvider = getProvider(selectedProvider);
   102→   102→
   103→   103→  /* ---------------------------------------------------------------- */
   104→   104→  /*  Drag end handler                                                  */
   105→   105→  /* ---------------------------------------------------------------- */
   106→   106→  const handleDragEnd = useCallback(
   107→   107→    (_: unknown, info: { offset: { y: number }; velocity: { y: number } }) => {
   108→   108→      const shouldClose = info.offset.y > 120 || info.velocity.y > 500;
   109→   109→      if (shouldClose) {
   110→   110→        animate(dragY, 600, {
   111→   111→          duration: 0.25,
   112→   112→          ease: 'easeOut',
   113→   113→          onComplete: onClose,
   114→   114→        });
   115→   115→      } else {
   116→   116→        animate(dragY, 0, { type: 'spring', stiffness: 400, damping: 30 });
   117→   117→      }
   118→   118→    },
   119→   119→    [dragY, onClose],
   120→   120→  );
   121→   121→
   122→   122→  /* ---------------------------------------------------------------- */
   123→   123→  /*  Record watch history                                              */
   124→   124→  /* ---------------------------------------------------------------- */
   125→   125→  useEffect(() => {
   126→   126→    if (!title) return;
   127→   127→    const item = mediaType === 'tv' ? selectedTv : selectedMovie;
   128→   128→    recordWatchHistory({
   129→   129→      tmdbId,
   130→   130→      title,
   131→   131→      posterPath: item?.poster_path || null,
   132→   132→      backdropPath: item?.backdrop_path || null,
   133→   133→      mediaType,
   134→   134→      season: season ?? null,
   135→   135→      episode: episode ?? null,
   136→   136→    });
   137→   137→    if (user) {
   138→   138→      fetch('/api/profile/history', {
   139→   139→        method: 'POST',
   140→   140→        headers: { 'Content-Type': 'application/json' },
   141→   141→        body: JSON.stringify({
   142→   142→          tmdbId,
   143→   143→          title,
   144→   144→          posterPath: item?.poster_path || null,
   145→   145→          mediaType,
   146→   146→          season: season ?? null,
   147→   147→          episode: episode ?? null,
   148→   148→        }),
   149→   149→      }).catch(() => {});
   150→   150→    }
   151→   151→  }, [tmdbId, mediaType, season, episode, title, user, selectedMovie, selectedTv]);
   152→   152→
   153→   153→  /* ---------------------------------------------------------------- */
   154→   154→  /*  Body scroll lock                                                  */
   155→   155→  /* ---------------------------------------------------------------- */
   156→   156→  useEffect(() => {
   157→   157→    const scrollY = window.scrollY;
   158→   158→    document.body.style.setProperty('--player-scroll-y', `${scrollY}px`);
   159→   159→    document.body.classList.add('player-open');
   160→   160→    document.body.style.top = `-${scrollY}px`;
   161→   161→    return () => {
   162→   162→      document.body.classList.remove('player-open');
   163→   163→      document.body.style.top = '';
   164→   164→      window.scrollTo(0, scrollY);
   165→   165→    };
   166→   166→  }, []);
   167→   167→
   168→   168→  /* ---------------------------------------------------------------- */
   169→   169→  /*  iOS native fullscreen "Done" button                               */
   170→   170→  /* ---------------------------------------------------------------- */
   171→   171→  useIOSFullscreenDetect(onClose);
   172→   172→
   173→   173→  /* ---------------------------------------------------------------- */
   174→   174→  /*  Auto-try next server on error                                     */
   175→   175→  /* ---------------------------------------------------------------- */
   176→   176→  const tryNextServer = () => {
   177→   177→    const currentIdx = sortedProviders.findIndex(p => p.id === selectedProvider);
   178→   178→    const next = sortedProviders[(currentIdx + 1) % sortedProviders.length];
   179→   179→    if (next) {
   180→   180→      switchProvider(next.id);
   181→   181→    }
   182→   182→  };
   183→   183→
   184→   184→  /* ---------------------------------------------------------------- */
   185→   185→  /*  Switch provider                                                   */
   186→   186→  /* ---------------------------------------------------------------- */
   187→   187→  const switchProvider = useCallback(
   188→   188→    (providerId: string) => {
   189→   189→      setSwitchingTo(providerId);
   190→   190→      recordServerPick(providerId);
   191→   191→      setSelectedProvider(providerId);
   192→   192→      const newSrc = getEmbedUrl(providerId, mediaType, tmdbId, season, episode);
   193→   193→      setCurrentSrc(newSrc);
   194→   194→      setLoading(true);
   195→   195→      setError(false);
   196→   196→      setMobileSheetOpen(false);
   197→   197→
   198→   198→      const item = mediaType === 'tv' ? selectedTv : selectedMovie;
   199→   199→      recordWatchHistory({
   200→   200→        tmdbId,
   201→   201→        title: title || 'Unknown',
   202→   202→        posterPath: item?.poster_path || null,
   203→   203→        backdropPath: item?.backdrop_path || null,
   204→   204→        mediaType,
   205→   205→        season: season ?? null,
   206→   206→        episode: episode ?? null,
   207→   207→      });
   208→   208→
   209→   209→      setTimeout(() => setSwitchingTo(null), 3000);
   210→   210→    },
   211→   211→    [mediaType, tmdbId, season, episode, setSelectedProvider, selectedTv, selectedMovie, title],
   212→   212→  );
   213→   213→
   214→   214→  /* ---------------------------------------------------------------- */
   215→   215→  /*  Fullscreen toggle                                                 */
   216→   216→  /* ---------------------------------------------------------------- */
   217→   217→  const toggleFullscreen = async () => {
   218→   218→    if (!containerRef.current) return;
   219→   219→    try {
   220→   220→      if (!document.fullscreenElement) {
   221→   221→        await containerRef.current.requestFullscreen();
   222→   222→        setIsFullscreen(true);
   223→   223→      } else {
   224→   224→        await document.exitFullscreen();
   225→   225→        setIsFullscreen(false);
   226→   226→      }
   227→   227→    } catch {
   228→   228→      // Fullscreen API unavailable (e.g. iOS)
   229→   229→    }
   230→   230→  };
   231→   231→
   232→   232→  useEffect(() => {
   233→   233→    const handler = () => setIsFullscreen(!!document.fullscreenElement);
   234→   234→    document.addEventListener('fullscreenchange', handler);
   235→   235→    return () => document.removeEventListener('fullscreenchange', handler);
   236→   236→  }, []);
   237→   237→
   238→   238→  /* ---------------------------------------------------------------- */
   239→   239→  /*  Keyboard shortcuts                                                */
   240→   240→  /* ---------------------------------------------------------------- */
   241→   241→  useEffect(() => {
   242→   242→    const handleKey = (e: KeyboardEvent) => {
   243→   243→      if (e.key === 'Escape' && !mobileSheetOpen) onClose();
   244→   244→    };
   245→   245→    window.addEventListener('keydown', handleKey);
   246→   246→    return () => window.removeEventListener('keydown', handleKey);
   247→   247→  }, [onClose, mobileSheetOpen]);
   248→   248→
   249→   249→  /* ---------------------------------------------------------------- */
   250→   250→  /*  Active server score / badge                                       */
   251→   251→  /* ---------------------------------------------------------------- */
   252→   252→  const activeScore = useMemo(() => getServerScore(selectedProvider), [selectedProvider]);
   253→   253→  const activeBadge = useMemo(() => scoreBadge(activeScore), [activeScore]);
   254→   254→  const isTopRanked = rankedIds[0] === selectedProvider;
   255→   255→
   256→   256→  /* ---------------------------------------------------------------- */
   257→   257→  /*  Mobile bottom sheet                                               */
   258→   258→  /* ---------------------------------------------------------------- */
   259→   259→  const mobileSheet = (
   260→   260→    <AnimatePresence>
   261→   261→      {mobileSheetOpen && (
   262→   262→        <>
   263→   263→          {/* Backdrop */}
   264→   264→          <motion.div
   265→   265→            initial={{ opacity: 0 }}
   266→   266→            animate={{ opacity: 1 }}
   267→   267→            exit={{ opacity: 0 }}
   268→   268→            className="absolute inset-0 bg-black/60 z-20"
   269→   269→            onClick={() => setMobileSheetOpen(false)}
   270→   270→          />
   271→   271→          {/* Sheet */}
   272→   272→          <motion.div
   273→   273→            initial={{ y: '100%' }}
   274→   274→            animate={{ y: 0 }}
   275→   275→            exit={{ y: '100%' }}
   276→   276→            transition={{ type: 'spring', stiffness: 300, damping: 30 }}
   277→   277→            className="absolute bottom-0 left-0 right-0 max-h-[70vh] bg-[#111] rounded-t-2xl z-30 flex flex-col"
   278→   278→            style={{ paddingBottom: isIOS ? 'env(safe-area-inset-bottom, 16px)' : 16 }}
   279→   279→          >
   280→   280→            {/* Drag handle */}
   281→   281→            <div className="flex justify-center pt-3 pb-2">
   282→   282→              <div className="w-10 h-1 rounded-full bg-white/20" />
   283→   283→            </div>
   284→   284→
   285→   285→            {/* Header */}
   286→   286→            <div className="flex items-center justify-between px-5 pb-3">
   287→   287→              <div className="flex items-center gap-2">
   288→   288→                <Server className="w-4 h-4 text-white/50" />
   289→   289→                <h4 className="text-white/90 text-sm font-semibold">Servers</h4>
   290→   290→              </div>
   291→   291→              <button
   292→   292→                onClick={() => setMobileSheetOpen(false)}
   293→   293→                className="w-7 h-7 rounded-lg flex items-center justify-center text-white/40 hover:text-white/80 hover:bg-white/[0.06] transition-colors"
   294→   294→                aria-label="Close server sheet"
   295→   295→              >
   296→   296→                <X className="w-3.5 h-3.5" />
   297→   297→              </button>
   298→   298→            </div>
   299→   299→
   300→   300→            {/* Server list */}
   301→   301→            <div className="flex-1 overflow-y-auto content-scroll px-3 pb-2">
   302→   302→              {sortedProviders.map((p) => {
   303→   303→                const isActive = p.id === selectedProvider;
   304→   304→                const score = getServerScore(p.id);
   305→   305→                const badge = scoreBadge(score);
   306→   306→                const isTop = rankedIds[0] === p.id;
   307→   307→                const isSwitching = switchingTo === p.id;
   308→   308→
   309→   309→                return (
   310→   310→                  <button
   311→   311→                    key={p.id}
   312→   312→                    onClick={() => switchProvider(p.id)}
   313→   313→                    disabled={isSwitching}
   314→   314→                    className={
   315→   315→                      'w-full flex items-center gap-3 px-3 py-3 rounded-xl text-left transition-colors ' +
   316→   316→                      (isActive ? 'bg-white/[0.06]' : 'hover:bg-white/[0.03]')
   317→   317→                    }
   318→   318→                  >
   319→   319→                    <div
   320→   320→                      className="w-9 h-9 rounded-lg flex items-center justify-center text-sm font-bold shrink-0"
   321→   321→                      style={{ backgroundColor: p.color + '18', color: p.color }}
   322→   322→                    >
   323→   323→                      {p.icon}
   324→   324→                    </div>
   325→   325→                    <div className="flex-1 min-w-0">
   326→   326→                      <div className="flex items-center gap-1.5">
   327→   327→                        <span className="text-white/90 text-sm font-medium truncate">{p.name}</span>
   328→   328→                        {isTop && <ShieldCheck className="w-3.5 h-3.5 text-emerald-400 shrink-0" />}
   329→   329→                      </div>
   330→   330→                      <p className="text-white/40 text-[11px] truncate mt-0.5">{p.description}</p>
   331→   331→                    </div>
   332→   332→                    <div className="flex flex-col items-end gap-1 shrink-0">
   333→   333→                      {badge && (
   334→   334→                        <span className={'px-1.5 py-0.5 rounded text-[9px] font-semibold ' + badge.cls}>
   335→   335→                          {badge.label}
   336→   336→                        </span>
   337→   337→                      )}
   338→   338→                      {isSwitching ? (
   339→   339→                        <Loader2 className="w-4 h-4 animate-spin text-white/40" />
   340→   340→                      ) : isActive ? (
   341→   341→                        <Check className="w-4 h-4" style={{ color: p.color }} />
   342→   342→                      ) : null}
   343→   343→                    </div>
   344→   344→                  </button>
   345→   345→                );
   346→   346→              })}
   347→   347→            </div>
   348→   348→          </motion.div>
   349→   349→        </>
   350→   350→      )}
   351→   351→    </AnimatePresence>
   352→   352→  );
   353→   353→
   354→   354→  /* ================================================================== */
   355→   355→  /*  RENDER                                                            */
   356→   356→  /* ================================================================== */
   357→   357→  return (
   358→   358→    <AnimatePresence>
   359→   359→      <motion.div
   360→   360→        initial={{ opacity: 0, y: 40 }}
   361→   361→        animate={{ opacity: 1, y: 0 }}
   362→   362→        exit={{ opacity: 0, y: 60 }}
   363→   363→        transition={{ type: 'spring', stiffness: 300, damping: 30, mass: 0.8 }}
   364→   364→        style={{ y: dragY }}
   365→   365→        drag={isMobile ? 'y' : false}
   366→   366→        dragConstraints={{ top: 0, bottom: 0 }}
   367→   367→        dragElastic={{ top: 0.1, bottom: 0.4 }}
   368→   368→        onDragEnd={handleDragEnd}
   369→   369→        className="fixed inset-0 z-[100] bg-black/95 backdrop-blur-md flex flex-col ios-player-container"
   370→   370→      >
   371→   371→        {/* iOS: translucent status bar spacer */}
   372→   372→        <div
   373→   373→          className="shrink-0 w-full"
   374→   374→          style={{ height: isIOS ? 'env(safe-area-inset-top, 0px)' : 0 }}
   375→   375→        />
   376→   376→
   377→   377→        {/* ==================== TOP BAR ==================== */}
   378→   378→        <div
   379→   379→          className="flex items-center gap-3 px-3 md:px-5 h-12 shrink-0 z-10 justify-between"
   380→   380→          style={{
   381→   381→            paddingTop: isIOS
   382→   382→              ? 'max(2px, env(safe-area-inset-top, 0px) - 10px)'
   383→   383→              : 0,
   384→   384→          }}
   385→   385→        >
   386→   386→          {/* Left: iOS Done button or empty spacer */}
   387→   387→          {isIOS ? (
   388→   388→            <button
   389→   389→              onClick={onClose}
   390→   390→              className="flex items-center gap-1 px-3 py-1.5 rounded-full text-white/80 hover:text-white text-sm font-semibold transition-colors active:scale-95"
   391→   391→              style={{
   392→   392→                background: 'rgba(255,255,255,0.12)',
   393→   393→                backdropFilter: 'blur(20px)',
   394→   394→                WebkitBackdropFilter: 'blur(20px)',
   395→   395→              }}
   396→   396→              aria-label="Done"
   397→   397→            >
   398→   398→              <ChevronLeft className="w-4 h-4" strokeWidth={2.5} />
   399→   399→              <span>Done</span>
   400→   400→            </button>
   401→   401→          ) : (
   402→   402→            <div className="w-16" />
   403→   403→          )}
   404→   404→
   405→   405→          {/* Center: title + active server pill */}
   406→   406→          <div className="flex items-center gap-2 min-w-0 flex-1 justify-center">
   407→   407→            <h3 className="text-white/70 font-medium text-sm truncate">
   408→   408→              {title || 'Now Playing'}
   409→   409→            </h3>
   410→   410→            <span
   411→   411→              className="shrink-0 inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-[10px] font-semibold"
   412→   412→              style={{
   413→   413→                backgroundColor: activeProvider.color + '20',
   414→   414→                color: activeProvider.color,
   415→   415→              }}
   416→   416→            >
   417→   417→              {isTopRanked && <ShieldCheck className="w-3 h-3" />}
   418→   418→              {activeProvider.name}
   419→   419→            </span>
   420→   420→          </div>
   421→   421→
   422→   422→          {/* Right: fullscreen + close (desktop) */}
   423→   423→          <div className="flex items-center gap-1 shrink-0">
   424→   424→            {!isIOS && (
   425→   425→              <button
   426→   426→                onClick={toggleFullscreen}
   427→   427→                className="w-9 h-9 rounded-full flex items-center justify-center text-white/60 hover:text-white hover:bg-white/10 transition-colors"
   428→   428→                aria-label={isFullscreen ? 'Exit fullscreen' : 'Enter fullscreen'}
   429→   429→              >
   430→   430→                {isFullscreen ? <Minimize2 className="w-4 h-4" /> : <Maximize2 className="w-4 h-4" />}
   431→   431→              </button>
   432→   432→            )}
   433→   433→            {!isIOS && (
   434→   434→              <button
   435→   435→                onClick={onClose}
   436→   436→                className="w-9 h-9 rounded-full flex items-center justify-center text-white/60 hover:text-white hover:bg-white/10 transition-colors"
   437→   437→                aria-label="Close player"
   438→   438→              >
   439→   439→                <X className="w-5 h-5" />
   440→   440→              </button>
   441→   441→            )}
   442→   442→          </div>
   443→   443→        </div>
   444→   444→
   445→   445→        {/* iOS swipe hint */}
   446→   446→        {isIOS && (
   447→   447→          <motion.div
   448→   448→            initial={{ opacity: 1, y: 0 }}
   449→   449→            animate={{ opacity: 0, y: -10 }}
   450→   450→            transition={{ delay: 2.5, duration: 1 }}
   451→   451→            className="flex justify-center pb-1"
   452→   452→          >
   453→   453→            <div className="flex items-center gap-1.5 px-3 py-1 rounded-full bg-white/[0.08]">
   454→   454→              <div className="w-1 h-1 rounded-full bg-white/30" />
   455→   455→              <span className="text-white/50 text-[10px] font-medium">Swipe down to close</span>
   456→   456→              <div className="w-1 h-1 rounded-full bg-white/30" />
   457→   457→            </div>
   458→   458→          </motion.div>
   459→   459→        )}
   460→   460→
   461→   461→        {/* ==================== VIDEO AREA ==================== */}
   462→   462→        <div
   463→   463→          className="flex-1 flex items-center justify-center p-3 md:p-8 relative overflow-hidden"
   464→   464→        >
   465→   465→          <motion.div
   466→   466→            ref={containerRef}
   467→   467→            style={{
