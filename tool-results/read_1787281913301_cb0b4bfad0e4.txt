    27→    27→function useIOSFullscreenDetect(onReturn: () => void) {
    28→    28→  useEffect(() => {
    29→    29→    const h = () => onReturn();
    30→    30→    document.addEventListener('webkitendfullscreen', h);
    31→    31→    return () => document.removeEventListener('webkitendfullscreen', h);
    32→    32→  }, [onReturn]);
    33→    33→}

    35→    35→/* Animated equalizer bars for "now playing" indicator */
    36→    36→function PlayingBars({ color }: { color: string }) {
    37→    37→  return (
    38→    38→    <div className="flex items-end gap-[2px] h-3">
    39→    39→      {[0, 1, 2].map((i) => (
    40→    40→        <motion.div
    41→    41→          key={i}
    42→    42→          className="w-[3px] rounded-full"
    43→    43→          style={{ backgroundColor: color }}
    44→    44→          animate={{ height: ['40%', '100%', '60%', '100%', '40%'] }}
    45→    45→          transition={{
    46→    46→            duration: 1.2,
    47→    47→            repeat: Infinity,
    48→    48→            delay: i * 0.2,
    49→    49→            ease: 'easeInOut',
    50→    50→          }}
    51→    51→        />
    52→    52→      ))}
    53→    53→    </div>
    54→    54→  );
    55→    55→}
