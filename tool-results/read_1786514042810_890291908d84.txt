    53→function formatDuration(iso: string): string {
    54→  const m = iso.match(/PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?/);
    55→  if (!m) return '0:00';
    56→  const h = parseInt(m[1] || '0');
    57→  const min = parseInt(m[2] || '0');
    58→  const s = parseInt(m[3] || '0');
    59→  if (h > 0) return `${h}:${min.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
    60→  return `${min}:${s.toString().padStart(2, '0')}`;
    61→}
    62→
    63→function formatDurationSeconds(iso: string): number {
    64→  const m = iso.match(/PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?/);
    65→  if (!m) return 0;
    66→  const h = parseInt(m[1] || '0');
    67→  const min = parseInt(m[2] || '0');
    68→  const s = parseInt(m[3] || '0');
    69→  return h * 3600 + min * 60 + s;
    70→}
    71→
    72→function formatTime(sec: number): string {
    73→  const m = Math.floor(sec / 60);
    74→  const s = Math.floor(sec % 60);
    75→  return `${m}:${s.toString().padStart(2, '0')}`;
    76→}
    77→
    78→function formatViews(n: number): string {
    79→  if (n >= 1_000_000_000) return `${(n / 1_000_000_000).toFixed(1)}B`;
    80→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
    81→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`;
    82→  return String(n);
    83→}

   332→function FullScreenPlayer({
   333→  track,
   334→  onClose,
   335→  onNext,
   336→  onPrev,
   337→}: {
   338→  track: YoutubeVideoResult;
   339→  onClose: () => void;
   340→  onNext: () => void;
   341→  onPrev: () => void;
   342→}) {
   343→  const [imgError, setImgError] = useState(false);
   344→  const [progress, setProgress] = useState(0);
   345→  const totalSec = formatDurationSeconds(track.duration);
   346→  const [liked, setLiked] = useState(false);
   347→
   348→  useEffect(() => {
   349→    const interval = setInterval(() => {
   350→      setProgress((p) => {
   351→        if (p >= 100) return 100;
   352→        return p + (100 / Math.max(totalSec, 1)) * 0.5;
   353→      });
   354→    }, 500);
   355→    return () => clearInterval(interval);
   356→  }, [totalSec]);
   357→
   358→  const currentSec = (progress / 100) * totalSec;
   359→
   360→  return (
   361→    <motion.div
   362→      initial={{ y: '100%' }}
   363→      animate={{ y: 0 }}
   364→      exit={{ y: '100%' }}
   365→      transition={{ type: 'spring', stiffness: 300, damping: 30 }}

   367→    >
   368→      {/* Blurred background */}

   370→        {!imgError ? (
   371→          <img src={track.thumbnail} alt="" className="w-full h-full object-cover scale-150 blur-3xl opacity-30" onError={() => setImgError(true)} />
   372→        ) : (
   373→          <div className="w-full h-full bg-gradient-to-b from-zinc-900 to-black" />

   526→        {/* Controls */}

   528→          <button onClick={onPrev} className="w-10 h-10 flex items-center justify-center text-white/40 hover:text-white/70 active:text-white transition-colors">
   529→            <SkipBack className="w-4 h-4" />
   530→          </button>
   531→
   532→          {/* White play/pause circle */}
   533→          <button className="w-8 h-8 rounded-full bg-white flex items-center justify-center shadow-md shadow-white/10">
   534→            <Pause className="w-3.5 h-3.5 text-black fill-black" />
   535→          </button>
   536→
   537→          <button onClick={onNext} className="w-10 h-10 flex items-center justify-center text-white/40 hover:text-white/70 active:text-white transition-colors">
   538→            <SkipForward className="w-4 h-4" />
   539→          </button>
   540→        </div>
   541→
   542→        {/* Right actions */}
   543→        <div className="flex items-center gap-0.5 shrink-0">
   544→          <button className="w-8 h-8 flex items-center justify-center text-white/30 hover:text-white/60 transition-colors">
   545→            <MoreHorizontal className="w-4 h-4" />
   546→          </button>
   547→        </div>
   548→      </div>
   549→    </motion.div>
   550→  );
   551→}

   571→export function MusicPage() {
   572→  const [trending, setTrending] = useState<YoutubeVideoResult[]>([]);
   573→  const [searchResults, setSearchResults] = useState<YoutubeVideoResult[]>([]);
   574→  const [relatedResults, setRelatedResults] = useState<YoutubeVideoResult[]>([]);
   575→  const [loading, setLoading] = useState(true);
   576→  const [searchLoading, setSearchLoading] = useState(false);
   577→  const [searchQuery, setSearchQuery] = useState('');
   578→  const [activeMood, setActiveMood] = useState('');
   579→  const [currentTrack, setCurrentTrack] = useState<YoutubeVideoResult | null>(null);
   580→  const [showFullScreen, setShowFullScreen] = useState(false);
   581→  const [queue, setQueue] = useState<YoutubeVideoResult[]>([]);
   582→  const [queueIndex, setQueueIndex] = useState(0);
   583→  const searchTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   584→  const searchInputRef = useRef<HTMLInputElement>(null);
   585→
   586→  // Fetch trending on mount
   587→  useEffect(() => {
   588→    async function fetchTrending() {
   589→      try {
   590→        const res = await fetch('/api/youtube/trending');
   591→        const data: ApiResponse = await res.json();
   592→        setTrending(data.results || []);
   593→      } catch (err) {
   594→        console.error('Failed to fetch trending:', err);
   595→      } finally {
   596→        setLoading(false);
   597→      }
   598→    }
   599→    fetchTrending();
   600→  }, []);
   601→
   602→  // Search with debounce
   603→  const handleSearchInput = useCallback((value: string) => {
   604→    setSearchQuery(value);
   605→    setActiveMood('');
   606→    if (searchTimerRef.current) clearTimeout(searchTimerRef.current);
   607→    if (!value.trim()) {
   608→      setSearchResults([]);
   609→      setRelatedResults([]);
   610→      return;
   611→    }
   612→    searchTimerRef.current = setTimeout(async () => {
   613→      setSearchLoading(true);
   614→      try {
   615→        const res = await fetch(`/api/youtube/search?query=${encodeURIComponent(value)}`);
   616→        const data: ApiResponse = await res.json();
   617→        setSearchResults(data.results || []);
   618→        if (data.results?.[0]) {
   619→          const relatedRes = await fetch(`/api/youtube/related?videoId=${data.results[0].videoId}`);
   620→          const relatedData: ApiResponse = await relatedRes.json();
   621→          setRelatedResults(relatedData.results || []);
   622→        } else {
   623→          setRelatedResults([]);
   624→        }
   625→      } catch {
   626→        setSearchResults([]);
   627→        setRelatedResults([]);
   628→      } finally {
   629→        setSearchLoading(false);
   630→      }
   631→    }, 500);
   632→  }, []);
   633→
   634→  // Mood card click
   635→  const handleMoodClick = useCallback((mood: typeof MOOD_CATEGORIES[number]) => {
   636→    const isCurrentlyActive = activeMood === mood.label;
   637→    setActiveMood(isCurrentlyActive ? '' : mood.label);
   638→    if (!isCurrentlyActive) {
   639→      handleSearchInput(mood.query);
   640→      if (searchInputRef.current) searchInputRef.current.value = mood.query;
   641→    } else {
   642→      setSearchQuery('');
   643→      setSearchResults([]);
   644→      setRelatedResults([]);
   645→      if (searchInputRef.current) searchInputRef.current.value = '';
   646→    }
   647→  }, [activeMood, handleSearchInput]);
   648→
   649→  // Play a track
   650→  const handlePlay = useCallback((track: YoutubeVideoResult, trackList?: YoutubeVideoResult[]) => {
   651→    setCurrentTrack(track);
   652→    setShowFullScreen(true);
   653→    if (trackList) {
   654→      const idx = trackList.findIndex((t) => t.videoId === track.videoId);
   655→      setQueue(trackList);
   656→      setQueueIndex(idx >= 0 ? idx : 0);
   657→    }
   658→  }, []);
   659→
   660→  const handleNext = useCallback(() => {
   661→    if (queue.length === 0) return;
   662→    const nextIdx = (queueIndex + 1) % queue.length;
   663→    setQueueIndex(nextIdx);
   664→    setCurrentTrack(queue[nextIdx]);
   665→  }, [queue, queueIndex]);
   666→
   667→  const handlePrev = useCallback(() => {
   668→    if (queue.length === 0) return;
   669→    const prevIdx = (queueIndex - 1 + queue.length) % queue.length;
   670→    setQueueIndex(prevIdx);
   671→    setCurrentTrack(queue[prevIdx]);
   672→  }, [queue, queueIndex]);
   673→
   674→  const handleClosePlayer = useCallback(() => {
   675→    setShowFullScreen(false);
   676→  }, []);
   677→
   678→  const isSearching = searchQuery.trim().length > 0;
   679→  const hasPlayer = currentTrack !== null;
   680→
   681→  // Top 6 for ranked cards, rest for list
   682→  const rankedTracks = isSearching ? searchResults.slice(0, 6) : trending.slice(0, 6);
   683→  const listTracks = isSearching ? searchResults.slice(6) : trending.slice(6);
   684→
   685→  return (
   686→    <div className="relative min-h-screen">
   687→      {/* Ambient gradient glow */}
   688→      <div
   689→        className="fixed inset-0 pointer-events-none z-0"
   690→        style={{
   691→          background: 'radial-gradient(ellipse at 70% -10%, rgba(138,42,42,0.35) 0%, rgba(45,21,21,0.15) 30%, transparent 65%)',
   692→        }}
   693→      />
   694→
   695→      {/* ── Search Bar (sticky) ── */}
   696→      <div className="sticky top-0 z-30 bg-black/80 backdrop-blur-xl">
   697→        <div className="px-4 pt-12 md:pt-3 pb-3">
   698→          <div className="relative max-w-md">
   699→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-white/30" />
   700→            <input
   701→              ref={searchInputRef}
   702→              type="text"
   703→              placeholder="Search songs, artists..."
   704→              defaultValue={searchQuery}
   705→              onChange={(e) => handleSearchInput(e.target.value)}
   706→              className="w-full pl-10 pr-10 h-9 rounded-lg bg-[#1C1C1C] border-0 text-white placeholder:text-white/30 text-[15px] font-normal focus:outline-none focus:ring-1 focus:ring-white/20 transition-shadow"
   707→            />
   708→            {searchQuery && (
   709→              <button
   710→                onClick={() => {
   711→                  handleSearchInput('');
   712→                  if (searchInputRef.current) searchInputRef.current.value = '';
   713→                }}
   714→                className="absolute right-3 top-1/2 -translate-y-1/2 text-white/30 hover:text-white/60 transition-colors"
   715→              >
   716→                <X className="w-4 h-4" />
   717→              </button>
   718→            )}
   719→          </div>
   720→        </div>
   721→      </div>
   722→
   723→      {/* Content */}
   724→      <div className="relative z-10">
   725→        {/* ── Browse Categories (icon cards, no emoji) ── */}
   726→        {!isSearching && (
   727→          <div className="px-4 mb-8 mt-2">
   728→            <SectionHeader title="Browse" />
   729→            <div className="flex gap-3 overflow-x-auto pb-2 -mx-4 px-4 content-scroll">
   730→              {loading
   731→                ? Array.from({ length: 5 }).map((_, i) => <CategorySkeleton key={i} />)
   732→                : MOOD_CATEGORIES.map((mood) => (
   733→                    <CategoryCard
   734→                      key={mood.label}
   735→                      label={mood.label}
   736→                      icon={mood.icon}
   737→                      color={mood.color}
   738→                      onClick={() => handleMoodClick(mood)}
   739→                      isActive={activeMood === mood.label}
   740→                    />
   741→                  ))
   742→              }
   743→            </div>
   744→          </div>
   745→        )}
   746→
   747→        {/* ── Genre Pills (when searching) ── */}
   748→        {isSearching && (
   749→          <div className="px-4 mb-6 mt-2">
   750→            <div className="flex gap-2 overflow-x-auto pb-1 -mx-4 px-4 content-scroll">
   751→              {GENRE_PILLS.map((genre) => (
   752→                <button
   753→                  key={genre}
   754→                  onClick={() => {

   759→                >

   766→
   767→        {/* ── Hero stat text ── */}
   768→        {!isSearching && !loading && trending.length > 0 && (

   770→            <p className="text-[22px] md:text-[24px] font-normal text-[#CCCCCC] leading-relaxed">

   776→        {/* ── Ranked Cards (horizontal scroll) ── */}

   791→                    track={track}

   794→                    isActive={currentTrack?.videoId === track.videoId}

   796→                ))}
   797→              </div>

   799→          ) : null
   800→        ) : loading ? (
   801→          <div className="px-4 mb-8">
   802→            <SectionHeader title="Trending Now" />
   803→            <div className="flex gap-3 overflow-x-auto pb-2 -mx-4 px-4 content-scroll">
