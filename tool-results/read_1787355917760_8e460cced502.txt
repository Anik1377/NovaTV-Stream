    99→function proxyCover(url?: string): string {
   100→  if (!url) return '';
   101→  if (url.includes('gutenberg.org')) return `/api/novels/proxy?url=${encodeURIComponent(url)}`;
   102→  if (url.includes('readcomicsonline.lol')) return `/api/comics/proxy?url=${encodeURIComponent(url)}`;
   103→  return `/api/manga/proxy?url=${encodeURIComponent(url)}`;
   104→}

   117→/* ═══════════════════════════════════════════════════ */
   118→export function ReadPage() {
   119→  const { goHome, selectManga, selectNovel, selectComic } = useAppStore();
   120→  const [mainTab, setMainTab] = useState<MainTab>('manga');
   121→
   122→  // Manga state
   123→  const [mangaList, setMangaList] = useState<MangaItem[]>([]);
   124→  const [mangaSubTab, setMangaSubTab] = useState('all');
   125→  const [mangaLoading, setMangaLoading] = useState(false);
   126→  const [mangaError, setMangaError] = useState(false);
   127→  const [mangaSearch, setMangaSearch] = useState('');
   128→  const [mangaHasMore, setMangaHasMore] = useState(false);
   129→  const [mangaTotal, setMangaTotal] = useState(0);
   130→  const [mangaOffset, setMangaOffset] = useState(0);
   131→  const mangaDebounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   132→  const mangaAbortRef = useRef<AbortController | null>(null);
   133→
   134→  // Comics state
   135→  const [comicList, setComicList] = useState<any[]>([]);
   136→  const [comicPublisher, setComicPublisher] = useState('All');
   137→  const [comicGenre, setComicGenre] = useState('All');
   138→  const [comicSort, setComicSort] = useState('popular');
   139→  const [comicSearch, setComicSearch] = useState('');
   140→  const [comicLoading, setComicLoading] = useState(true);
   141→  const [comicError, setComicError] = useState(false);
   142→  const [comicTotal, setComicTotal] = useState(0);
   143→  const [availablePublishers, setAvailablePublishers] = useState<string[]>([]);
   144→  const [availableGenres, setAvailableGenres] = useState<string[]>([]);
   145→  const [showSortDropdown, setShowSortDropdown] = useState(false);
   146→  const comicDebounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   147→
   148→  // Novels state
   149→  const [novelList, setNovelList] = useState<NovelItem[]>([]);
   150→  const [novelGenre, setNovelGenre] = useState('All');
   151→  const [novelSearch, setNovelSearch] = useState('');
   152→  const [novelLoading, setNovelLoading] = useState(true);
   153→  const novelDebounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   154→
   155→  /* ── Manga fetch ── */
   156→  const fetchManga = useCallback(async (tab: string, search: string, offset: number, append = false) => {
   157→    mangaAbortRef.current?.abort();
   158→    const controller = new AbortController();
   159→    mangaAbortRef.current = controller;
   160→    if (!append) { setMangaLoading(true); setMangaError(false); }
   161→
   162→    try {
   163→      const params = new URLSearchParams();
   164→      if (search.trim()) {
   165→        params.set('q', search.trim());
   166→        if (offset > 0) params.set('page', String(Math.floor(offset / 20) + 1));
   167→        const res = await fetch(`/api/manga/search?${params}`, { signal: controller.signal });
   168→        if (!res.ok) throw new Error();
   169→        const data = await res.json();
   170→        if (controller.signal.aborted) return;
   171→        const items: MangaItem[] = (data.results || []).map((m: any) => ({
   172→          id: String(m.id), title: String(m.title || 'Untitled'),
   173→          coverUrl: String(m.coverUrl || ''), author: String(m.author || ''),
   174→          tags: Array.isArray(m.tags) ? m.tags.filter(Boolean).map(String) : [],
   175→        }));
   176→        setMangaList(append ? (prev) => [...prev, ...items] : items);
   177→        setMangaHasMore(!!data.hasMore);
   178→        setMangaTotal(data.total || 0);
   179→      } else {
   180→        if (tab === 'manga' || tab === 'manhwa' || tab === 'manhua') params.set('type', tab);
   181→        if (offset > 0) params.set('offset', String(offset));
   182→        const url = `/api/manga/trending${params.toString() ? `?${params}` : ''}`;
   183→        const res = await fetch(url, { signal: controller.signal });
   184→        if (!res.ok) throw new Error();
   185→        const data = await res.json();
   186→        if (controller.signal.aborted) return;
   187→        const items: MangaItem[] = (data.results || []).map((m: any) => ({
   188→          id: String(m.id), title: String(m.title || 'Untitled'),
   189→          coverUrl: String(m.coverUrl || ''), author: String(m.author || ''),
   190→          tags: Array.isArray(m.tags) ? m.tags.filter(Boolean).map(String) : [],
   191→        }));
   192→        setMangaList(append ? (prev) => [...prev, ...items] : items);
   193→        setMangaHasMore(!!data.hasMore);
   194→        setMangaTotal(data.total || 0);
   195→      }
   196→    } catch {
   197→      if (!append) setMangaError(true);
   198→    } finally {
   199→      if (!controller.signal.aborted) setMangaLoading(false);
   200→    }
   201→  }, []);
   202→
   203→  useEffect(() => {
   204→    setMangaOffset(0); setMangaSearch('');
   205→    fetchManga(mangaSubTab, '', 0);
   206→    return () => { mangaAbortRef.current?.abort(); };
   207→  }, [mangaSubTab, fetchManga]);
   208→
   209→  const handleMangaSearch = useCallback((value: string) => {
   210→    setMangaSearch(value);
   211→    if (mangaDebounceRef.current) clearTimeout(mangaDebounceRef.current);
   212→    if (!value.trim()) { setMangaOffset(0); fetchManga(mangaSubTab, '', 0); return; }
   213→    mangaDebounceRef.current = setTimeout(() => { setMangaOffset(0); fetchManga(mangaSubTab, value, 0); }, 300);
   214→  }, [mangaSubTab, fetchManga]);
   215→
   216→  /* ── Comics fetch ── */
   217→  const fetchComics = useCallback(async (search: string) => {
   218→    setComicLoading(true);
   219→    setComicError(false);
   220→    try {
   221→      const params = new URLSearchParams();
   222→      if (comicPublisher !== 'All') params.set('publisher', comicPublisher);
   223→      if (comicGenre !== 'All') params.set('genre', comicGenre);
   224→      if (comicSort) params.set('sort', comicSort);
   225→      if (search.trim()) params.set('q', search.trim());
   226→      const res = await fetch(`/api/comics/trending?${params}`);
   227→      if (!res.ok) throw new Error();
   228→      const data = await res.json();
   229→      setComicList(data.results || []);
   230→      setComicTotal(data.total || 0);
   231→      if (data.publishers) setAvailablePublishers(data.publishers);
   232→      if (data.genres) setAvailableGenres(data.genres);
   233→    } catch {
   234→      setComicError(true);
   235→    } finally {
   236→      setComicLoading(false);
   237→    }
   238→  }, [comicPublisher, comicGenre, comicSort]);
   239→
   240→  useEffect(() => {
   241→    fetchComics(comicSearch);
   242→  }, [fetchComics]);
   243→
   244→  const handleComicSearch = useCallback((value: string) => {
   245→    setComicSearch(value);
   246→    if (comicDebounceRef.current) clearTimeout(comicDebounceRef.current);
   247→    comicDebounceRef.current = setTimeout(() => {
   248→      fetchComics(value);
   249→    }, 300);
   250→  }, [fetchComics]);
   251→
   252→  /* ── Novels fetch ── */
   253→  useEffect(() => {
   254→    setNovelLoading(true);
   255→    fetch('/novels-data.json')
   256→      .then((r) => r.json())
   257→      .then((data: NovelItem[]) => setNovelList(data))
   258→      .catch(() => {})
   259→      .finally(() => setNovelLoading(false));
   260→  }, []);
   261→
   262→  const filteredNovels = useMemo(() => {
   263→    let list = novelList;
   264→    if (novelGenre !== 'All') list = list.filter((n) => n.genre.includes(novelGenre));
   265→    if (novelSearch.trim()) {
   266→      const q = novelSearch.toLowerCase();
   267→      list = list.filter(
   268→        (n) =>
   269→          n.title.toLowerCase().includes(q) ||
   270→          n.author.toLowerCase().includes(q) ||
   271→          n.description.toLowerCase().includes(q)
   272→      );
   273→    }
   274→    return list;
   275→  }, [novelList, novelGenre, novelSearch]);
   276→
   277→  const handleNovelSearch = useCallback((value: string) => {
   278→    setNovelSearch(value);
   279→    if (novelDebounceRef.current) clearTimeout(novelDebounceRef.current);
   280→  }, []);
   281→
   282→  /* ── Shared search input ── */
   283→  const searchPlaceholder = mainTab === 'manga'
   284→    ? 'Search manga, manhwa, manhua...'
   285→    : mainTab === 'comics'
   286→    ? 'Search Marvel, DC, Image comics...'
   287→    : 'Search classic novels...';
   288→
   289→  const currentSearch = mainTab === 'manga' ? mangaSearch : mainTab === 'comics' ? comicSearch : novelSearch;
   290→  const handleSearch = mainTab === 'manga' ? handleMangaSearch : mainTab === 'comics' ? handleComicSearch : handleNovelSearch;
   291→
   292→  /* ── Active color ── */
   293→  const activeColor = mainTab === 'manga' ? 'amber' : mainTab === 'comics' ? 'red' : 'emerald';
   294→
   295→  /* ── Render sub-tabs ── */
   296→  const subTabs: SubTab[] = mainTab === 'manga' ? MANGA_TABS : [];
   297→  const activeSubTab = mainTab === 'manga' ? mangaSubTab : '';
   298→  const setActiveSubTab = mainTab === 'manga' ? setMangaSubTab : () => {};
   299→
   300→  /* ── Item count ── */
   301→  const itemCount =
   302→    mainTab === 'manga' ? mangaTotal :
   303→    mainTab === 'comics' ? comicTotal :
   304→    filteredNovels.length;
   305→
   306→  /* ── Publisher icon helper ── */
   307→  const getPublisherIcon = (publisher: string) => {
   308→    switch (publisher) {
   309→      case 'Marvel': return <StarIcon className="w-3 h-3" />;
   310→      case 'DC': return <Crown className="w-3 h-3" />;
   311→      case 'Image': return <Eye className="w-3 h-3" />;
   312→      case 'Dark Horse': return <Skull className="w-3 h-3" />;
   313→      case 'IDW': return <Zap className="w-3 h-3" />;
   314→      default: return <Shield className="w-3 h-3" />;
   315→    }
   316→  };
   317→
   318→  return (
   319→    <div className="min-h-screen pb-10">
   320→      {/* Mobile back button */}
   321→      <button
   322→        onClick={goHome}
   323→        className="md:hidden fixed z-[90] flex items-center gap-1.5 text-white/60 active:text-white transition-colors"
   324→        style={{ top: 'max(env(safe-area-inset-top, 0px) + 8px, 8px)', left: 12 }}
   325→        aria-label="Go home"
   326→      >
   327→        <ArrowLeft className="w-5 h-5" />
   328→      </button>
   329→
   330→      <div className="pt-16 md:pt-8 pb-6 px-4 md:px-8">
   331→        {/* Desktop back */}
   332→        <button
   333→          onClick={goHome}
   334→          className="hidden md:flex items-center gap-2 text-white/70 hover:text-white mb-4 transition-colors"
   335→        >
   336→          <ArrowLeft className="w-4 h-4" />
   337→          <span className="text-sm">Back to Home</span>
   338→        </button>
   339→
   340→        {/* ── Hero header ── */}
   341→        <div className="relative mb-6">
   342→          <div className={`relative h-[20vh] min-h-[160px] md:min-h-[200px] rounded-2xl overflow-hidden`}>
   343→            <div className={`absolute inset-0 bg-gradient-to-br ${MAIN_TABS.find(t => t.key === mainTab)?.color || 'from-amber-500 to-orange-600'}`} />
   344→            <div className="absolute inset-0 opacity-20" style={{
   345→              backgroundImage: 'radial-gradient(circle at 20% 80%, rgba(255,255,255,0.15) 0%, transparent 50%), radial-gradient(circle at 80% 20%, rgba(255,255,255,0.1) 0%, transparent 50%)',
   346→            }} />
   347→            <div className="absolute inset-0 bg-gradient-to-t from-black/60 via-transparent to-black/20" />
   348→            <div className="relative h-full flex items-center justify-center text-center px-4">
   349→              <div>
   350→                <div className={`w-14 h-14 rounded-2xl bg-gradient-to-br ${MAIN_TABS.find(t => t.key === mainTab)?.color || ''} flex items-center justify-center shadow-lg mb-3`}>
   351→                  <BookOpen className="w-7 h-7 text-white" />
   352→                </div>
   353→                <h1 className="text-3xl md:text-4xl font-black text-white mb-1">
   354→                  Read
   355→                </h1>
   356→                <p className="text-white/50 text-xs md:text-sm max-w-md mx-auto">
   357→                  {mainTab === 'manga' && 'Manga, Manhwa & Manhua'}
   358→                  {mainTab === 'comics' && 'Marvel, DC, Image & More'}
   359→                  {mainTab === 'novels' && 'Classic Novels & Literature'}
   360→                  {' \u00B7 '}{itemCount > 0 && !mangaLoading && !comicLoading && !novelLoading ? `${itemCount.toLocaleString()} titles` : 'Free to read'}
   361→                </p>
   362→              </div>
   363→            </div>
   364→          </div>
   365→        </div>
   366→
   367→        {/* ── Main tabs ── */}
   368→        <div className="flex gap-1 mb-5">
   369→          {MAIN_TABS.map((tab) => {
   370→            const IconComp = tab.icon;
   371→            return (
   372→              <button
   373→                key={tab.key}
   374→                onClick={() => setMainTab(tab.key)}
   375→                className={`shrink-0 px-4 py-2.5 rounded-xl text-sm font-semibold transition-all relative ${
   376→                  mainTab === tab.key
   377→                    ? `bg-${activeColor}-500/15 text-${activeColor}-400 border border-${activeColor}-500/30`
   378→                    : 'bg-white/5 text-white/50 hover:bg-white/10 hover:text-white/70 border border-white/[0.08]'
   379→                }`}
   380→              >
   381→                <IconComp className="w-4 h-4" />
   382→                <span className="ml-1.5">{tab.label}</span>
   383→                {mainTab === tab.key && (
   384→                  <motion.div
   385→                    layoutId="read-main-tab"
   386→                    className={`absolute inset-0 rounded-xl bg-${activeColor}-500/15 border border-${activeColor}-500/30`}
   387→                    transition={{ type: 'spring', stiffness: 400, damping: 30 }}
   388→                  />
   389→                )}
   390→              </button>
   391→            );
   392→          })}
   393→        </div>
   394→
   395→        {/* ── Search ── */}
   396→        {mainTab !== 'novels' && (
   397→          <div className="relative mb-5">
   398→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-white/40" />
   399→            <input
   400→              type="text"
   401→              placeholder={searchPlaceholder}
   402→              value={currentSearch}
   403→              onChange={(e) => handleSearch(e.target.value)}
   404→              className={`w-full bg-white/5 border border-white/10 rounded-xl pl-10 pr-4 py-2.5 text-sm text-white placeholder:text-white/50 focus:outline-none focus:border-${activeColor}-500/50 focus:ring-1 focus:ring-${activeColor}-500/30 transition-colors`}
   405→            />
   406→          </div>
   407→        )}
   408→
   409→        {/* ── Sub-tabs (manga only) ── */}
   410→        {subTabs.length > 0 && (
   411→          <div className="flex gap-1 overflow-x-auto scrollbar-none pb-3 -mx-1 px-1">
   412→            {subTabs.map((tab) => (
   413→              <button
   414→                key={tab.key}
   415→                onClick={() => setActiveSubTab(tab.key)}
   416→                className={`shrink-0 px-3 py-1.5 rounded-lg text-xs font-medium transition-colors relative whitespace-nowrap ${
   417→                  activeSubTab === tab.key
   418→                    ? `text-${activeColor}-400`
   419→                    : 'text-white/40 hover:text-white/60'
   420→                }`}
   421→              >
   422→                {tab.flag && <span className="mr-1">{tab.flag}</span>}
   423→                {tab.label}
   424→                {activeSubTab === tab.key && (
   425→                  <motion.div
   426→                    layoutId="read-sub-tab"
   427→                    className={`absolute bottom-0 left-0 right-0 h-0.5 bg-${activeColor}-500 rounded-full`}
   428→                    transition={{ type: 'spring', stiffness: 400, damping: 30 }}
   429→                  />
   430→                )}
   431→              </button>
   432→            ))}
   433→          </div>
   434→        )}
   435→
   436→        {/* ── Comic publisher / genre / sort filters ── */}
   437→        {mainTab === 'comics' && (
   438→          <div className="space-y-3 mb-5">
   439→            {/* Publisher filter pills */}
   440→            <div className="flex gap-1.5 overflow-x-auto scrollbar-none -mx-1 px-1">
   441→              {COMIC_PUBLISHERS.map((pub) => (
   442→                <button
   443→                  key={pub}
   444→                  onClick={() => setComicPublisher(pub)}
   445→                  className={`shrink-0 flex items-center gap-1.5 px-3 py-1.5 rounded-full text-xs font-semibold transition-all border ${
   446→                    comicPublisher === pub
   447→                      ? (PUBLISHER_COLORS[pub] || 'bg-white/15 text-white/90 border-white/20')
   448→                      : 'bg-white/[0.04] text-white/40 border-white/[0.08] hover:bg-white/[0.08] hover:text-white/70'
   449→                  }`}
   450→                >
   451→                  {getPublisherIcon(pub)}
   452→                  {pub}
   453→                </button>
   454→              ))}
   455→            </div>
   456→
   457→            {/* Genre filter row */}
   458→            <div className="flex gap-1.5 overflow-x-auto scrollbar-none -mx-1 px-1">
   459→              {COMIC_GENRES.map((genre) => (
   460→                <button
   461→                  key={genre}
   462→                  onClick={() => setComicGenre(genre)}
   463→                  className={`shrink-0 px-2.5 py-1 rounded-full text-[11px] font-medium transition-all border ${
